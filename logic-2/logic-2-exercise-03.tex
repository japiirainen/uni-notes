\begin{exercise}
  Osoita, että kaavat \(\forall x\exists y(R(x,y) \land R(y,x))\) ja
  \(\forall x(\exists yR(x,y) \land \exists yR(y,x))\) eivät ole loogisesti
  ekvivalentteja. Onko jompi kumpi toisen looginen seuraus?
\end{exercise}

\begin{solution}
  Merkitään
  \[
    \begin{aligned}
      A &= \forall x\exists y(R(x,y) \land R(y,x)), \\
      B &= \forall x(\exists yR(x,y) \land \exists yR(y,x)).
    \end{aligned}
  \]
  Osoitetaan, että \(B\) on kaavan \(A\) looginen seuraus.
  Olkoon \(\mathcal M\) mielivaltainen malli ja \(s\) sen tulkintafunktio.
  Oletetaan, että \(\mathcal M \models_s A\).

  Olkoon \(b \in \dom(\mathcal M)\) mielivaltainen. Universaali- ja
  eksistenssikvanttorin totuusmääritelmien nojalla on olemassa
  \(a \in \dom(\mathcal M)\), jolla
  \[
    \mathcal M \models_{s(b/x)(a/y)}
    \bigl(R(x,y) \land R(y,x)\bigr).
  \]
  Tässä alkio \(a\) saa riippua valitusta alkiosta \(b\).
  Konjunktion totuusmääritelmän nojalla
  \[
    \mathcal M \models_{s(b/x)(a/y)} R(x,y)
    \quad\text{ja}\quad
    \mathcal M \models_{s(b/x)(a/y)} R(y,x).
  \]
  Sama alkio \(a\) kelpaa siis todistajaksi kumpaankin
  eksistenssikvantifioituun kaavaan, joten
  \[
    \mathcal M \models_{s(b/x)} \exists yR(x,y)
    \quad\text{ja}\quad
    \mathcal M \models_{s(b/x)} \exists yR(y,x).
  \]
  Näin ollen
  \[
    \mathcal M \models_{s(b/x)}
    \bigl(\exists yR(x,y) \land \exists yR(y,x)\bigr).
  \]
  Koska \(b\) oli mielivaltainen, universaalikvanttorin totuusmääritelmästä
  seuraa \(\mathcal M \models_s B\). Mallin \(\mathcal M\) ja
  tulkintafunktion \(s\) mielivaltaisuuden nojalla siis \(A \models B\).

  Käänteinen seuraussuhde ei päde. Tarkastellaan mallia \(\mathcal M\), jossa
  \[
    \dom(\mathcal M) = \{1,2,3\},
    \qquad
    R^{\mathcal M} = \{(1,3),(3,2),(2,1)\}.
  \]
  Olkoon \(s\) mielivaltainen \(\mathcal M\)-tulkintafunktio.
  Jokaiselle alkiolle \(b \in \dom(\mathcal M)\) voidaan valita alkiot
  \(a,c \in \dom(\mathcal M)\) seuraavasti:
  \[
    \begin{array}{c|cc}
      b & a & c \\
      \hline
      1 & 3 & 2 \\
      2 & 1 & 3 \\
      3 & 2 & 1
    \end{array}
  \]
  Tällöin \((b,a) \in R^{\mathcal M}\) ja \((c,b) \in R^{\mathcal M}\),
  joten
  \[
    \mathcal M \models_{s(b/x)}
    \bigl(\exists yR(x,y) \land \exists yR(y,x)\bigr).
  \]
  Koska tämä pätee jokaiselle \(b\), saadaan \(\mathcal M \models_s B\).

  Kaava \(A\) ei sen sijaan ole tosi tässä mallissa. Kun \(x\):n arvoksi
  valitaan \(1\), ainoa mahdollinen todistaja kaavalle \(R(x,y)\) on
  \(y = 3\). Koska \((3,1) \notin R^{\mathcal M}\), tämä alkio ei kuitenkaan
  toteuta kaavaa \(R(y,x)\). Siten mikään \(y\):n arvo ei toteuta molempia
  kaavoja yhtä aikaa, eli
  \[
    \mathcal M \not\models_{s(1/x)}
    \exists y\bigl(R(x,y) \land R(y,x)\bigr).
  \]
  Siis \(\mathcal M \not\models_s A\), joten \(B \not\models A\).
  Näin ollen kaavat \(A\) ja \(B\) eivät ole loogisesti ekvivalentteja:
  \(B\) on kaavan \(A\) looginen seuraus, mutta \(A\) ei ole kaavan \(B\)
  looginen seuraus.
\end{solution}

\begin{exercise}
  Osoita, että kaava \(\forall y\exists xA\) on kaavan
  \(\exists x\forall yA\) looginen seuraus (huomaa, että ei ole väliä,
  mikä kaava \(A\) on).
\end{exercise}

\begin{solution}
  Olkoon \(\mathcal M\) mielivaltainen malli ja \(s\) sen tulkintafunktio.
  Oletetaan, että
  \[
    \mathcal M \models_s \exists x\forall yA.
  \]
  Eksistenssi- ja universaalikvanttorin totuusmääritelmien nojalla on
  olemassa \(b \in \dom(\mathcal M)\), jolla kaikilla
  \(a \in \dom(\mathcal M)\) pätee
  \[
    \mathcal M \models_{s(b/x)(a/y)} A.
  \]
  Tässä sama alkio \(b\) toimii jokaisella \(a\):n arvolla.

  Olkoon nyt \(a \in \dom(\mathcal M)\) mielivaltainen. Koska \(x\) ja
  \(y\) ovat eri muuttujia, tulkintafunktion päivitykset voidaan tehdä
  kummassa tahansa järjestyksessä:
  \[
    s(b/x)(a/y) = s(a/y)(b/x).
  \]
  Siis \(\mathcal M \models_{s(a/y)(b/x)} A\), joten alkio \(b\)
  kelpaa eksistenssikvanttorin todistajaksi ja saadaan
  \[
    \mathcal M \models_{s(a/y)} \exists xA.
  \]
  Koska \(a\) oli mielivaltainen, universaalikvanttorin totuusmääritelmän
  nojalla \(\mathcal M \models_s \forall y\exists xA\).
  Mallin \(\mathcal M\) ja tulkintafunktion \(s\) mielivaltaisuudesta seuraa
  \[
    \exists x\forall yA \models \forall y\exists xA.
  \]
  Todistuksessa ei käytetty mitään oletuksia kaavan \(A\) rakenteesta,
  joten väite pätee kaikilla kaavoilla \(A\).\qedhere
\end{solution}

\begin{exercise}
  Mitkä muuttujien esiintymistä seuraavissa kaavoissa ovat vapaita ja mitkä
  sidottuja? Mitkä kaavoista ovat lauseita?
  \begin{enumerate}[label=(\alph*)]
    \item \(\forall x(P_0(x) \to P_1(y))\)
    \item \(\forall x(\forall yE(x,y) \lor \forall zE(y,z))\)
    \item \(\forall y(\exists x\,R(x,y) \lor \exists x\,R(y,x))\)
    \item \(\forall x(P(x) \to \exists yR(x,y))\)
    \item \(\forall xP_0(y)\)
    \item \(\forall y(\exists x\,x = y \lor R(y,x))\)
  \end{enumerate}
\end{exercise}

\begin{solution}
  Muuttujan esiintymä on sidottu, jos se on kyseistä muuttujaa sitovan
  kvanttorin vaikutusalassa, ja muuten vapaa. Kaava on lause, jos siinä
  ei ole vapaita muuttujien esiintymiä.
  \begin{enumerate}[label=(\alph*)]
    \item Kaavassa \(P_0(x)\) muuttujan \(x\) esiintymä on sidottu
      kvanttorilla \(\forall x\). Kaavassa \(P_1(y)\) muuttujan \(y\)
      esiintymä on vapaa, joten koko kaava ei ole lause.

    \item Kaavassa \(E(x,y)\) muuttujan \(x\) esiintymä on sidottu
      ulommalla kvanttorilla \(\forall x\) ja muuttujan \(y\) esiintymä
      kvanttorilla \(\forall y\). Kaavassa \(E(y,z)\) muuttujan \(z\)
      esiintymä on sidottu kvanttorilla \(\forall z\), mutta muuttujan
      \(y\) esiintymä on vapaa: kvanttorin \(\forall y\) vaikutusala
      rajoittuu vasempaan disjunktiin. Koko kaava ei siis ole lause.

    \item Muuttujan \(y\) molemmat esiintymät atomikaavoissa ovat sidottuja
      ulommalla kvanttorilla \(\forall y\). Kummankin atomikaavan
      \(x\)-esiintymä on sidottu oman disjunktinsa kvanttorilla
      \(\exists x\). Vapaita esiintymiä ei ole, joten kaava on lause.

    \item Muuttujan \(x\) esiintymät kaavoissa \(P(x)\) ja \(R(x,y)\)
      ovat sidottuja kvanttorilla \(\forall x\). Muuttujan \(y\) esiintymä
      kaavassa \(R(x,y)\) on sidottu kvanttorilla \(\exists y\).
      Vapaita esiintymiä ei ole, joten kaava on lause.

    \item Muuttujan \(y\) esiintymä kaavassa \(P_0(y)\) on vapaa.
      Kvanttori \(\forall x\) ei sido sitä, eikä sen vaikutusalassa ole
      yhtään muuttujan \(x\) esiintymää. Kaava ei siis ole lause.

    \item Yhtälössä \(x = y\) muuttujan \(x\) esiintymä on sidottu
      kvanttorilla \(\exists x\). Muuttujan \(y\) esiintymät sekä yhtälössä
      että kaavassa \(R(y,x)\) ovat sidottuja kvanttorilla \(\forall y\).
      Sen sijaan kaavassa \(R(y,x)\) muuttujan \(x\) esiintymä on vapaa,
      sillä kvanttorin \(\exists x\) vaikutusala on vain yhtälö \(x = y\).
      Koko kaava ei siis ole lause.
  \end{enumerate}
  Lauseita ovat siis täsmälleen kohtien \textbf{(c) ja (d)} kaavat.\qedhere
\end{solution}

\begin{exercise}
  Osoita, että sidotun muuttujan uudelleenkvantifiointi ei tuo kaavaan mitään
  uutta, eli osoita
  \begin{enumerate}[label=(\alph*)]
    \item \(\forall x_i\exists x_iA \Leftrightarrow \exists x_iA\),
    \item \(\exists x_i\forall x_iA \Leftrightarrow \forall x_iA\).
  \end{enumerate}
\end{exercise}

\begin{solution}
  Olkoon \(\mathcal M\) mielivaltainen malli, \(D = \dom(\mathcal M)\)
  ja \(s\) sen tulkintafunktio. Mallin perusjoukko \(D\) on epätyhjä.
  Kaikilla \(a,b \in D\) pätee
  \[
    s(a/x_i)(b/x_i) = s(b/x_i),
  \]
  sillä jälkimmäinen päivitys korvaa muuttujan \(x_i\) aiemman arvon.
  Kvanttorien totuusmääritelmistä saadaan siis:
  \begin{enumerate}[label=(\alph*)]
    \item
      \[
        \begin{aligned}
          &\mathcal M \models_s \forall x_i\exists x_iA \\
          &\iff (\forall a \in D)(\exists b \in D)\;
          \mathcal M \models_{s(a/x_i)(b/x_i)} A \\
          &\iff (\forall a \in D)(\exists b \in D)\;
          \mathcal M \models_{s(b/x_i)} A \\
          &\iff (\exists b \in D)\;
          \mathcal M \models_{s(b/x_i)} A \\
          &\iff \mathcal M \models_s \exists x_iA.
        \end{aligned}
      \]

    \item
      \[
        \begin{aligned}
          &\mathcal M \models_s \exists x_i\forall x_iA \\
          &\iff (\exists a \in D)(\forall b \in D)\;
          \mathcal M \models_{s(a/x_i)(b/x_i)} A \\
          &\iff (\exists a \in D)(\forall b \in D)\;
          \mathcal M \models_{s(b/x_i)} A \\
          &\iff (\forall b \in D)\;
          \mathcal M \models_{s(b/x_i)} A \\
          &\iff \mathcal M \models_s \forall x_iA.
        \end{aligned}
      \]
  \end{enumerate}
  Kummassakin kohdassa ehto \(\mathcal M \models_{s(b/x_i)} A\) on
  riippumaton alkiosta \(a\), joten ulompi kvanttori voidaan poistaa,
  koska \(D \ne \varnothing\). Näin molemmat loogiset ekvivalenssit
  pätevät.\qedhere
\end{solution}

\begin{exercise}
  Jos \(A\) on \(L\)-kaava, niin \(A\):n \emph{universaalisulkeuma} on
  lause, joka saadaan, kun \(A\):n kaikki vapaat muuttujat kvantifioidaan
  universaalikvanttoreilla. Esim. kaavan \(R(x_0,x_1) \land P(x_0)\)
  universaalisulkeuma on lause
  \(\forall x_0\forall x_1(R(x_0,x_1) \land P(x_0))\).
  \begin{enumerate}[label=(\alph*)]
    \item Osoita, että mielivaltaisessa \(L\)-mallissa \(\mathcal M\),
      \(A\) on tosi jos ja vain jos \(A\):n universaalisulkeuma on tosi.
    \item Anna esimerkki kaavasta \(A\), joka ei ole loogisesti ekvivalentti
      universaalisulkeumansa kanssa. Perustele.
  \end{enumerate}
\end{exercise}

\begin{solution}
  \textbf{(a)} Olkoon \(\mathcal M\) mielivaltainen \(L\)-malli.
  Merkitään kaavan \(A\) kaikkia vapaita muuttujia
  \(x_1,\ldots,x_k\) ja sen universaalisulkeumaa
  \(\overline A = \forall x_1\cdots\forall x_k A\).
  Jos vapaita muuttujia ei ole, niin \(\overline A = A\) ja väite on selvä.

  Oletetaan ensin, että \(\mathcal M \models A\), eli jokainen
  \(\mathcal M\)-tulkintafunktio toteuttaa kaavan \(A\).
  Olkoon \(s\) mielivaltainen tulkintafunktio. Kaikilla
  \(a_1,\ldots,a_k \in \dom(\mathcal M)\) myös
  \(s(a_1/x_1)\cdots(a_k/x_k)\) on tulkintafunktio, joten
  \[
    \mathcal M \models_{s(a_1/x_1)\cdots(a_k/x_k)} A.
  \]
  Universaalikvanttorin totuusmääritelmän nojalla
  \(\mathcal M \models_s \overline A\). Koska \(s\) oli mielivaltainen,
  saadaan \(\mathcal M \models \overline A\).

  Oletetaan kääntäen, että \(\mathcal M \models \overline A\), ja olkoon
  \(s\) mielivaltainen tulkintafunktio. Tällöin
  \(\mathcal M \models_s \forall x_1\cdots\forall x_k A\), joten
  valitsemalla \(a_j = s(x_j)\) kaikilla \(j=1,\ldots,k\) saadaan
  \[
    \mathcal M \models_{s(a_1/x_1)\cdots(a_k/x_k)} A.
  \]
  Koska \(s(a_1/x_1)\cdots(a_k/x_k) = s\), tästä seuraa
  \(\mathcal M \models_s A\). Tulkintafunktion \(s\) mielivaltaisuuden
  nojalla \(\mathcal M \models A\). Siis
  \(\mathcal M \models A \iff \mathcal M \models \overline A\).

  \medskip
  \textbf{(b)} Valitaan \(A = P(x)\), jonka universaalisulkeuma on
  \(\overline A = \forall xP(x)\). Tarkastellaan mallia \(\mathcal M\), jossa
  \[
    \dom(\mathcal M) = \{0,1\},
    \qquad P^{\mathcal M} = \{0\},
  \]
  ja tulkintafunktiota \(s\), jolla \(s(x)=0\). Tällöin
  \(\mathcal M \models_s P(x)\), koska \(0 \in P^{\mathcal M}\), mutta
  \(\mathcal M \not\models_s \forall xP(x)\), koska
  \(1 \notin P^{\mathcal M}\). Kaavoilla on siis eri totuusarvot samassa
  mallissa samalla tulkintafunktiolla, joten ne eivät ole loogisesti
  ekvivalentteja.\qedhere
\end{solution}

\begin{exercise}
  Todista materiaalin lause 4.7: Jos \(s\) ja \(s'\) ovat kaksi
  \(\mathcal M\)-tulkintafunktiota, joilla \(s(x_i) = s'(x_i)\) kaikilla
  kaavan \(A\) vapailla muuttujilla \(x_i\), niin
  \[
    \mathcal M \models_s A
    \quad\text{jos ja vain jos}\quad
    \mathcal M \models_{s'} A.
  \]
  Voit käyttää todistuksessasi seuraavaa aputulosta, joka on triviaali
  aakkostoille ilman funktiosymboleita, ja jonka voi myöhemmin yleistää
  kaikille aakkostoille:
  \begin{quote}
    Lemma. Jos \(s\) ja \(s'\) ovat kaksi \(\mathcal M\)-tulkintafunktiota,
    joilla \(s(x_i) = s'(x_i)\) kaikilla termissä \(t\) esiintyvillä
    muuttujilla \(x_i\), niin
    \(t^{\mathcal M}\langle s\rangle = t^{\mathcal M}\langle s'\rangle\).
  \end{quote}
\end{exercise}

\begin{solution}
  Todistetaan väite rakenneinduktiolla. Kiinnitetään malli \(\mathcal M\),
  merkitään \(D=\dom(\mathcal M)\) ja oletetaan, että
  \(s|_{\operatorname{FV}(A)}=s'|_{\operatorname{FV}(A)}\).
  Induktioväite koskee kaikkia tällaisia tulkintafunktiopareja.

  \textbf{Atomikaavat.} Jos \(A=R(t_1,\ldots,t_n)\), niin kaikki termien
  \(t_j\) muuttujat ovat vapaita kaavassa \(A\). Annetun lemman nojalla
  \(t_j^{\mathcal M}\langle s\rangle=t_j^{\mathcal M}\langle s'\rangle\)
  kaikilla \(j\). Sama alkiojono siis kuuluu relaatioon
  \(R^{\mathcal M}\) kummallakin tulkinnalla tai ei kummallakaan.
  Yhtälön \(t_1=t_2\) tapauksessa samat termiarvot antavat vastaavasti
  saman totuusarvon yhtäsuuruudelle.

  \textbf{Konnektiivit.} Esimerkiksi tapauksessa \(A=B\land C\)
  kaavojen \(B,C\) vapaat muuttujat ovat vapaita myös kaavassa \(A\).
  Induktio-oletuksen nojalla sekä \(B\):n että \(C\):n totuusarvo on
  sama tulkinnoilla \(s,s'\), joten myös niiden konjunktion totuusarvo
  on sama. Negaation ja muiden konnektiivien tapaukset seuraavat
  samalla tavalla induktio-oletuksesta ja totuustaulukoista.

  \textbf{Kvanttorit.} Olkoon \(A=QxB\), missä
  \(Q\in\{\exists,\forall\}\). Tällöin
  \(\operatorname{FV}(A)=\operatorname{FV}(B)\setminus\{x\}\).
  Siksi kaikilla \(a\in D\) pätee
  \[
    s(a/x)|_{\operatorname{FV}(B)}
    =s'(a/x)|_{\operatorname{FV}(B)}:
  \]
  molemmat antavat \(x\):lle arvon \(a\), ja muiden vapaiden muuttujien
  arvot yhtyvät oletuksen nojalla.
  Siten induktio-oletuksen nojalla kaikilla \(a\in D\)
  \[
    \mathcal M\models_{s(a/x)}B
    \iff\mathcal M\models_{s'(a/x)}B.
  \]
  Kvanttorien totuusmääritelmistä saadaan
  \[
    \begin{aligned}
      \mathcal M\models_s\exists xB
      &\iff (\exists a\in D)\;\mathcal M\models_{s(a/x)}B \\
      &\iff (\exists a\in D)\;\mathcal M\models_{s'(a/x)}B \\
      &\iff\mathcal M\models_{s'}\exists xB,
    \end{aligned}
  \]
  sekä
  \[
    \begin{aligned}
      \mathcal M\models_s\forall xB
      &\iff (\forall a\in D)\;\mathcal M\models_{s(a/x)}B \\
      &\iff (\forall a\in D)\;\mathcal M\models_{s'(a/x)}B \\
      &\iff\mathcal M\models_{s'}\forall xB.
    \end{aligned}
  \]
  Väite pätee siis rakenneinduktion nojalla kaikilla kaavoilla.\qedhere
\end{solution}

\begin{exercise}
  Minkä joukon kaava
  \(\exists x_1\exists x_2((E(x_0,x_1) \land E(x_1,x_2)) \land \neg x_0 = x_2)\)
  määrittelee allaolevassa verkossa?
  \begin{center}
    \includegraphics[width=0.4\linewidth]{figures/exercise-03-graph.pdf}
  \end{center}
\end{exercise}

\begin{solution}
  Merkitään kaavaa \(\varphi(x_0)\), vasemman komponentin päätekärkiä
  \(u,w\) ja niiden yhteistä naapuria \(v\). Kaava toteutuu kärjellä,
  jos siitä on kahden särmän kulku eri kärkeen. Kärjille \(u,w\)
  todistajiksi kelpaavat vastaavasti \((x_1,x_2)=(v,w)\) ja \((v,u)\).
  Kaikista muista kärjistä jokainen kahden särmän kulku palaa
  lähtökärkeen. Kaavan määrittelemä joukko on siis
  \[
    \boxed{\varphi^{\mathcal M}=\{u,w\}},
  \]
  eli vasemman komponentin kaksi päätekärkeä.\qedhere
\end{solution}

\begin{exercise}
  Alkio \(a\) on määriteltävä mallissa \(\mathcal M\) jos joukko \(\{a\}\)
  on määriteltävä \(\mathcal M\):ssä. Osoita, että \(2\) on määriteltävä
  mallissa \((\NN,<)\), missä \(<\) on luonnollisten lukujen luonnollinen
  järjestys.
\end{exercise}

\begin{solution}
  Olkoon \(\mathcal M=(\NN,<)\), missä \(\NN=\{0,1,2,\ldots\}\).
  Tarkastellaan kaavaa
  \[
    \varphi(x)=\exists y\exists z\Bigl(
      y<z\land z<x\land
      \forall w\bigl(w<x\to(w=y\lor w=z)\bigr)
    \Bigr).
  \]
  Ehto \(y<z<x\) takaa, että \(x\):ää pienempiä alkioita on vähintään
  kaksi, ja universaalikvantifioitu ehto, että muita ei ole. Siten
  \[
    \mathcal M\models_s\varphi(x)
    \iff \bigl|\{n\in\NN:n<s(x)\}\bigr|=2
    \iff s(x)=2.
  \]
  Siis \(\varphi\) määrittelee joukon \(\{2\}\), joten \(2\) on
  määriteltävä mallissa \(\mathcal M\).\qedhere
\end{solution}
