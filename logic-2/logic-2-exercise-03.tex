\begin{exercise}
  Osoita, että kaavat \(\forall x\exists y(R(x,y) \land R(y,x))\) ja
  \(\forall x(\exists yR(x,y) \land \exists yR(y,x))\) eivät ole loogisesti
  ekvivalentteja. Onko jompi kumpi toisen looginen seuraus?
\end{exercise}

\begin{solution}
  Merkitään
  \[
    \begin{aligned}
      A &= \forall x\exists y(R(x,y) \land R(y,x)), \\
      B &= \forall x(\exists yR(x,y) \land \exists yR(y,x)).
    \end{aligned}
  \]
  Osoitetaan, että \(B\) on kaavan \(A\) looginen seuraus.
  Olkoon \(\mathcal M\) mielivaltainen malli ja \(s\) sen tulkintafunktio.
  Oletetaan, että \(\mathcal M \models_s A\).

  Olkoon \(b \in \dom(\mathcal M)\) mielivaltainen. Universaali- ja
  eksistenssikvanttorin totuusmääritelmien nojalla on olemassa
  \(a \in \dom(\mathcal M)\), jolla
  \[
    \mathcal M \models_{s(b/x)(a/y)}
    \bigl(R(x,y) \land R(y,x)\bigr).
  \]
  Tässä alkio \(a\) saa riippua valitusta alkiosta \(b\).
  Konjunktion totuusmääritelmän nojalla
  \[
    \mathcal M \models_{s(b/x)(a/y)} R(x,y)
    \quad\text{ja}\quad
    \mathcal M \models_{s(b/x)(a/y)} R(y,x).
  \]
  Sama alkio \(a\) kelpaa siis todistajaksi kumpaankin
  eksistenssikvantifioituun kaavaan, joten
  \[
    \mathcal M \models_{s(b/x)} \exists yR(x,y)
    \quad\text{ja}\quad
    \mathcal M \models_{s(b/x)} \exists yR(y,x).
  \]
  Näin ollen
  \[
    \mathcal M \models_{s(b/x)}
    \bigl(\exists yR(x,y) \land \exists yR(y,x)\bigr).
  \]
  Koska \(b\) oli mielivaltainen, universaalikvanttorin totuusmääritelmästä
  seuraa \(\mathcal M \models_s B\). Mallin \(\mathcal M\) ja
  tulkintafunktion \(s\) mielivaltaisuuden nojalla siis \(A \models B\).

  Käänteinen seuraussuhde ei päde. Tarkastellaan mallia \(\mathcal M\), jossa
  \[
    \dom(\mathcal M) = \{1,2,3\},
    \qquad
    R^{\mathcal M} = \{(1,3),(3,2),(2,1)\}.
  \]
  Olkoon \(s\) mielivaltainen \(\mathcal M\)-tulkintafunktio.
  Jokaiselle alkiolle \(b \in \dom(\mathcal M)\) voidaan valita alkiot
  \(a,c \in \dom(\mathcal M)\) seuraavasti:
  \[
    \begin{array}{c|cc}
      b & a & c \\
      \hline
      1 & 3 & 2 \\
      2 & 1 & 3 \\
      3 & 2 & 1
    \end{array}
  \]
  Tällöin \((b,a) \in R^{\mathcal M}\) ja \((c,b) \in R^{\mathcal M}\),
  joten
  \[
    \mathcal M \models_{s(b/x)}
    \bigl(\exists yR(x,y) \land \exists yR(y,x)\bigr).
  \]
  Koska tämä pätee jokaiselle \(b\), saadaan \(\mathcal M \models_s B\).

  Kaava \(A\) ei sen sijaan ole tosi tässä mallissa. Kun \(x\):n arvoksi
  valitaan \(1\), ainoa mahdollinen todistaja kaavalle \(R(x,y)\) on
  \(y = 3\). Koska \((3,1) \notin R^{\mathcal M}\), tämä alkio ei kuitenkaan
  toteuta kaavaa \(R(y,x)\). Siten mikään \(y\):n arvo ei toteuta molempia
  kaavoja yhtä aikaa, eli
  \[
    \mathcal M \not\models_{s(1/x)}
    \exists y\bigl(R(x,y) \land R(y,x)\bigr).
  \]
  Siis \(\mathcal M \not\models_s A\), joten \(B \not\models A\).
  Näin ollen kaavat \(A\) ja \(B\) eivät ole loogisesti ekvivalentteja:
  \(B\) on kaavan \(A\) looginen seuraus, mutta \(A\) ei ole kaavan \(B\)
  looginen seuraus.
\end{solution}

\begin{exercise}
  Osoita, että kaava \(\forall y\exists xA\) on kaavan
  \(\exists x\forall yA\) looginen seuraus (huomaa, että ei ole väliä,
  mikä kaava \(A\) on).
\end{exercise}

\begin{solution}
  Olkoon \(\mathcal M\) mielivaltainen malli ja \(s\) sen tulkintafunktio.
  Oletetaan, että
  \[
    \mathcal M \models_s \exists x\forall yA.
  \]
  Eksistenssi- ja universaalikvanttorin totuusmääritelmien nojalla on
  olemassa \(b \in \dom(\mathcal M)\), jolla kaikilla
  \(a \in \dom(\mathcal M)\) pätee
  \[
    \mathcal M \models_{s(b/x)(a/y)} A.
  \]
  Tässä sama alkio \(b\) toimii jokaisella \(a\):n arvolla.

  Olkoon nyt \(a \in \dom(\mathcal M)\) mielivaltainen. Koska \(x\) ja
  \(y\) ovat eri muuttujia, tulkintafunktion päivitykset voidaan tehdä
  kummassa tahansa järjestyksessä:
  \[
    s(b/x)(a/y) = s(a/y)(b/x).
  \]
  Siis \(\mathcal M \models_{s(a/y)(b/x)} A\), joten alkio \(b\)
  kelpaa eksistenssikvanttorin todistajaksi ja saadaan
  \[
    \mathcal M \models_{s(a/y)} \exists xA.
  \]
  Koska \(a\) oli mielivaltainen, universaalikvanttorin totuusmääritelmän
  nojalla \(\mathcal M \models_s \forall y\exists xA\).
  Mallin \(\mathcal M\) ja tulkintafunktion \(s\) mielivaltaisuudesta seuraa
  \[
    \exists x\forall yA \models \forall y\exists xA.
  \]
  Todistuksessa ei käytetty mitään oletuksia kaavan \(A\) rakenteesta,
  joten väite pätee kaikilla kaavoilla \(A\).\qedhere
\end{solution}

\begin{exercise}
  Mitkä muuttujien esiintymistä seuraavissa kaavoissa ovat vapaita ja mitkä
  sidottuja? Mitkä kaavoista ovat lauseita?
  \begin{enumerate}[label=(\alph*)]
    \item \(\forall x(P_0(x) \to P_1(y))\)
    \item \(\forall x(\forall yE(x,y) \lor \forall zE(y,z))\)
    \item \(\forall y(\exists x\,R(x,y) \lor \exists x\,R(y,x))\)
    \item \(\forall x(P(x) \to \exists yR(x,y))\)
    \item \(\forall xP_0(y)\)
    \item \(\forall y(\exists x\,x = y \lor R(y,x))\)
  \end{enumerate}
\end{exercise}

\begin{exercise}
  Osoita, että sidotun muuttujan uudelleenkvantifiointi ei tuo kaavaan mitään
  uutta, eli osoita
  \begin{enumerate}[label=(\alph*)]
    \item \(\forall x_i\exists x_iA \Leftrightarrow \exists x_iA\),
    \item \(\exists x_i\forall x_iA \Leftrightarrow \forall x_iA\).
  \end{enumerate}
\end{exercise}

\begin{exercise}
  Jos \(A\) on \(L\)-kaava, niin \(A\):n \emph{universaalisulkeuma} on
  lause, joka saadaan, kun \(A\):n kaikki vapaat muuttujat kvaintifioidaan
  universaalikvanttoreilla. Esim. kaavan \(R(x_0,x_1) \land P(x_0)\)
  universaalisulkeuma on lause
  \(\forall x_0\forall x_1(R(x_0,x_1) \land P(x_0))\).
  \begin{enumerate}[label=(\alph*)]
    \item Osoita, että mielivaltaisessa \(L\)-mallissa \(\mathcal M\),
      \(A\) on tosi jos ja vain jos \(A\):n universaalisulkeuma on tosi.
    \item Anna esimerkki kaavasta \(A\), joka ei ole loogisesti ekvivalentti
      universaalisulkeumansa kanssa. Perustele.
  \end{enumerate}
\end{exercise}

\begin{exercise}
  Todista materiaalin lause 4.7: Jos \(s\) ja \(s'\) ovat kaksi
  \(\mathcal M\)-tulkintafunktiota, joilla \(s(x_i) = s'(x_i)\) kaikilla
  kaavan \(A\) vapailla muuttujilla \(x_i\), niin
  \[
    \mathcal M \models_s A
    \quad\text{jos ja vain jos}\quad
    \mathcal M \models_{s'} A.
  \]
  Voit käyttää todistuksessasi seuraavaa aputulosta, joka on triviaali
  aakkostoille ilman funktiosymboleita, ja jonka voi myöhemmin yleistää
  kaikille aakkostoille:
  \begin{quote}
    Lemma. Jos \(s\) ja \(s'\) ovat kaksi \(\mathcal M\)-tulkintafunktiota,
    joilla \(s(x_i) = s'(x_i)\) kaikilla termissä \(t\) esiintyvillä
    muuttujilla \(x_i\), niin
    \(t^{\mathcal M}\langle s\rangle = t^{\mathcal M}\langle s'\rangle\).
  \end{quote}
\end{exercise}
