  \end{align*}
\end{definition}

\subsubsection{Principle of mathematical induction}

Here we state the fundamental principle of mathematical induction without proof,
and use it freely thereafter.

\begin{theorem}[Principle of mathematical induction]\label{thm:induction}
  Let \(n_0 \in \mathbb{N}\), and let \(P(n)\) be a statement for each
  \(n \in \mathbb{N}\) with \(n \geq n_0\). Suppose that
  \begin{enumerate}
    \item \(P(n_0)\) is true, and
    \item for every \(n \geq n_0\), \(P(n)\) implies \(P(n+1)\).
  \end{enumerate}
  Then \(P(n)\) is true for every \(n \geq n_0\).
\end{theorem}

\begin{proof}
  Omitted.
\end{proof}

\begin{example}
  We prove by induction that
  \[
    \sum_{k=1}^{n} k = \frac{n(n+1)}{2}
  \]
  for every integer \(n \geq 1\).

  For \(n=1\), the identity holds because
  \[
    \sum_{k=1}^{1} k = 1 = \frac{1(1+1)}{2}.
  \]
  Now suppose that the identity holds for some integer \(n \geq 1\). Then
  \begin{align*}
    \sum_{k=1}^{n+1} k
    &= \sum_{k=1}^{n} k + (n+1) \\
    &= \frac{n(n+1)}{2} + (n+1) \\
    &= \frac{(n+1)(n+2)}{2}.
  \end{align*}
  Thus the identity also holds for \(n+1\), and hence for every integer
  \(n \geq 1\). \qed
\end{example}

\subsubsection{Divisibility of integers}

We start by defining the precise meaning of \emph{even} and \emph{odd} numbers.

\begin{definition}
  A number \(m \in \ZZ\) is \emph{even} if there exists \(k \in \ZZ\) such that
  \[
    m = 2k.
  \]
  On the other hand, \(m\) is \emph{odd} if there exists a \(k \in \ZZ\) such
  that
  \[
    m = 2k + 1.
  \]
\end{definition}

\begin{definition}[Divisibility]
  A number \(a \in \ZZ\) is said to be \emph{divisible} by a number
  \(b \in \ZZ, b \neq 0\), if there exists a \(k \in \ZZ\) such that
  \[
    a = kb.
  \]
  In this case we also say that \(b\) \emph{divides} \(a\), and write
  \[
    b \mid a.
  \]
  If \(b\) does \emph{not} divide \(a\), we write
  \[
    b \nmid a.
  \]
\end{definition}

\begin{definition}
  For \(n \in \ZZ \setminus \{0\}\), we denote the set of integers that are
  divisible by \(n\) by
  \[
    n\ZZ = \left\{z \in \ZZ : z = nk, \quad k \in \ZZ\right\}.
  \]
\end{definition}

\subsubsection{Strong induction}

In some proofs it is useful to assume all preceding cases rather than only the
immediately preceding case. This gives an equivalent form of the principle of
induction, called \emph{strong induction}.

\begin{theorem}[Strong induction]\label{thm:strong-induction}
  Let \(n_0 \in \NN\), and let \(P(n)\) be a statement for each
  \(n \in \NN\) with \(n \geq n_0\). Suppose that
  \begin{enumerate}
    \item \(P(n_0)\) is true, and
    \item for every \(n \geq n_0\), if \(P(k)\) is true for every \(k\) with
      \(n_0 \leq k \leq n\), then \(P(n + 1)\) is true.
  \end{enumerate}
  Then \(P(n)\) is true for every \(n \geq n_0\).
\end{theorem}

\begin{proof}
  Omitted.
\end{proof}

As an example where strong induction is natural, let us prove that all natural
numbers \(n \geq 2\) can be written as a product of \emph{prime} numbers.

\begin{theorem}
  Every natural number \(n \geq 2\) can be written as a product of
  prime numbers.
\end{theorem}

\begin{proof}
  When \(n = 2\), we have \(n \in \PP\), so the result holds trivially.
  Now suppose \(n \geq 2\) and the statement holds for all
  \(2 \leq k \leq n\). If \(n + 1 \in \PP\), the result holds trivially, so
  suppose \(n + 1 \notin \PP\). Then we have
  \[
    n + 1 = ab,
  \]
  for some \(2 \leq a,b \leq n\). By the induction hypothesis, \(a\) and
  \(b\) can be written as products of primes, so
  \[
    \begin{aligned}
      a &= p_1p_2\cdots p_i, \quad p_1,\ldots,p_i \in \PP, \\
      b &= q_1q_2\cdots q_j, \quad q_1,\ldots,q_j \in \PP.
    \end{aligned}
  \]
  Thus we have
  \[
    n + 1 = ab = p_1p_2\cdots p_iq_1q_2\cdots q_j.
  \]
  The result follows from Theorem \ref{thm:strong-induction}.
\end{proof}

\subsubsection{Sequences}

\begin{definition}
  A sequence of real numbers is a function \(z\colon \NN \rightarrow \RR\),
  whose terms are denoted by \(z_n = z(n)\). We write the sequence as
  \[
    (z_n)_{n \in \NN} = (z_0,z_1,z_2,\ldots).
  \]
\end{definition}

\begin{definition}
  A sequence of real numbers may also be specified recursively. First, for
  some \(k \in \ZZ^+\), we define the initial terms
  \(z_0,\ldots,z_{k-1}\). For each \(n \geq k\), we then define \(z_n\) in
  terms of earlier terms \(z_0,\ldots,z_{n-1}\).
\end{definition}

\subsubsection{Logical connectives}

\begin{definition}
  Let \(p\) and \(q\) be propositions. We define the following logical
  connectives:
  \begin{enumerate}
    \item The \emph{conjunction} \(p \land q\), read ``\(p\) and \(q\)'', is
      true exactly when both \(p\) and \(q\) are true.
    \item The \emph{disjunction} \(p \lor q\), read ``\(p\) or \(q\)'', is
      true exactly when at least one of \(p\) and \(q\) is true.
    \item The \emph{implication} \(p \implies q\), read ``\(p\) implies
      \(q\)'', is false exactly when \(p\) is true and \(q\) is false.
    \item The \emph{equivalence} \(p \iff q\), read ``\(p\) if and only if
      \(q\)'', is true exactly when \(p\) and \(q\) have the same truth value.
    \item The \emph{negation} \(\neg p\), read ``not \(p\)'', is true exactly
      when \(p\) is false.
  \end{enumerate}
\end{definition}

\begin{definition}
  Let \(p\) and \(q\) be propositions.
  \begin{enumerate}
    \item The \emph{negated implication} \(p \nRightarrow q\), read ``\(p\)
      does not imply \(q\)'', is true exactly when \(p\) is true and \(q\) is
      false.
    \item The \emph{negated equivalence} \(p \nLeftrightarrow q\), read
      ``\(p\) is not equivalent to \(q\)'', is true exactly when \(p\) and
      \(q\) have different truth values.
  \end{enumerate}
\end{definition}

\begin{definition}
  Let \(X\) be a set. A \emph{predicate} \(P(x)\) on \(X\) is a statement
  containing a variable \(x\) whose truth value depends on the chosen value of
  \(x \in X\). For each \(a \in X\), substituting \(a\) for \(x\) gives the
  proposition \(P(a)\), which is either true or false. For example,
  \(P(x) \coloneqq x > 0\) is a predicate on \(\RR\); \(P(2)\) is true, whereas
  \(P(-1)\) is false.
\end{definition}

\begin{definition}
  Let \(P(x)\) be a predicate on a set \(X\).
  \begin{enumerate}
    \item The \emph{universal quantification}
      \(\forall x \in X \colon P(x)\), read ``for every \(x \in X\),
      \(P(x)\)'', is true exactly when \(P(a)\) is true for every \(a \in X\).
    \item The \emph{existential quantification}
      \(\exists x \in X \colon P(x)\), read ``there exists an \(x \in X\) such
      that \(P(x)\)'', is true exactly when \(P(a)\) is true for at least one
      \(a \in X\).
  \end{enumerate}
\end{definition}

\begin{definition}
  The \emph{negated existential quantification}
  \(\nexists x \in X \colon P(x)\), read ``there does not exist an \(x \in X\)
  such that \(P(x)\)'', is true exactly when \(P(a)\) is false for every
  \(a \in X\). Equivalently,
  \[
    \nexists x \in X \colon P(x)
    \quad \iff \quad
    \forall x \in X \colon \neg P(x).
  \]
\end{definition}

\subsubsection{Basic set theory}

\begin{definition}
  A \emph{set} is a collection of distinct objects, called its
  \emph{elements}. If an object \(x\) is an element of a set \(A\), we write
  \(x \in A\); otherwise, we write \(x \notin A\). The order in which the
  elements are listed, and any repetition among them, do not affect the set.
\end{definition}

\begin{definition}
  Two sets \(A\) and \(B\) are \emph{equal}, written \(A = B\), if they have
  exactly the same elements. Equivalently,
  \[
    A = B \iff \forall x \colon (x \in A \iff x \in B).
  \]
\end{definition}

\begin{definition}
  The \emph{empty set} is the set containing no elements. It is denoted by
  \(\emptyset\).
\end{definition}

\begin{definition}
  Let \(A\) and \(B\) be are sets. \(A\) and \(B\) are said to be
  \emph{disjoint}, if \(A \cap B = \emptyset\).
\end{definition}

\begin{definition}
  A \emph{singleton} set is a set consisting of exactly \textbf{one} element.
\end{definition}

\begin{definition}
  Let \(P(x)\) be a predicate on a set \(X\). The \emph{set-builder notation}
  \[
    A \coloneqq \set{x \in X \colon P(x)}
  \]
  defines \(A\) to be the set of precisely those elements \(x \in X\) for which
  \(P(x)\) is true. Equivalently, for every \(a \in X\),
  \[
    a \in A \iff P(a).
  \]
\end{definition}

\begin{definition}
  A set \(A\) is a \emph{subset} of a set \(B\) if every element of \(A\) is
  also an element of \(B\). We then write \(A \subseteq B\). If \(A\) is not a
  subset of \(B\), we write \(A \nsubseteq B\); this means that some element of
  \(A\) is not an element of \(B\). If \(A \subseteq B\) and \(A \neq B\), then
  \(A\) is a \emph{proper subset} of \(B\), written \(A \subset B\).
  Two sets are equal exactly when both containments hold:
  \[
    A = B \iff A \subseteq B \land B \subseteq A.
  \]
\end{definition}

\begin{definition}
  Let \(A\) and \(B\) be subsets of a set \(X\).
  \begin{enumerate}
    \item The \emph{union} of \(A\) and \(B\) is
      \[
        A \cup B \coloneqq
        \set{x \in X \colon x \in A \lor x \in B}.
      \]
    \item The \emph{intersection} of \(A\) and \(B\) is
      \[
        A \cap B \coloneqq
        \set{x \in X \colon x \in A \land x \in B}.
      \]
    \item The \emph{difference} of \(A\) and \(B\) is
      \[
        A \setminus B \coloneqq
        \set{x \in X \colon x \in A \land x \notin B}.
      \]
    \item The \emph{complement} of \(A\) in \(X\) is
      \[
        \complement{A} \coloneqq X \setminus A
        = \set{x \in X \colon x \notin A}.
      \]
    \item The \emph{symmetric difference} of \(A\) and \(B\) is
      \[
        A \mathbin{\Delta} B \coloneqq
        (A \setminus B) \cup (B \setminus A).
      \]
  \end{enumerate}
\end{definition}

\subsubsection{Power sets}

\begin{definition}
  The \emph{power set} of a set \(A\), denoted by \(\powerset{A}\), is the set
  of all subsets of \(A\):
  \[
    \powerset{A} \coloneqq \set{B : B \subseteq A}.
  \]
  In particular, \(\emptyset \in \powerset{A}\) and
  \(A \in \powerset{A}\).
\end{definition}

\begin{definition}
  Let \(A\) and \(B\) be sets. Their \emph{Cartesian product} is the set of all
  ordered pairs whose first component belongs to \(A\) and whose second
  component belongs to \(B\):
  \[
    A \times B \coloneqq \set{(a,b) : a \in A \land b \in B}.
  \]
  More generally, for sets \(A_1,\ldots,A_n\), their Cartesian product is
  \[
    A_1 \times \cdots \times A_n
    \coloneqq
    \set{(a_1,\ldots,a_n) : a_i \in A_i
    \text{ for every } i \in \set{1,\ldots,n}}.
  \]
  If all the factors are the same set \(A\), one commonly writes
  \[
    A^n \coloneqq \underbrace{A \times \cdots \times A}_{n\text{ factors}}.
  \]
\end{definition}

\begin{definition}
  Let \(I\) be a countably infinite \emph{index set}. A family of subsets of a
  set \(X\), indexed by \(I\), is denoted by \((A_i)_{i \in I}\). Its
  \emph{union} and \emph{intersection} are defined by
  \begin{align*}
    \bigcup_{i \in I} A_i
    &\coloneqq \set{x \in X \colon \exists i \in I,\ x \in A_i}, \\
    \bigcap_{i \in I} A_i
    &\coloneqq \set{x \in X \colon \forall i \in I,\ x \in A_i}.
  \end{align*}
  Thus, an element belongs to the union if it belongs to at least one of the
  sets, and it belongs to the intersection if it belongs to every one of them.

  If the family itself is denoted by
  \(\mathcal{A} = \set{A_i : i \in I}\), one also writes
  \(\bigcup \mathcal{A}\) and \(\bigcap \mathcal{A}\).

  A common index set is \(I = \mathbb{N}\). Since \(0 \in \mathbb{N}\) by
  our convention, one writes
  \begin{align*}
    \bigcup_{n=0}^{\infty} A_n
    &= \bigcup_{n \in \mathbb{N}} A_n, &
    \bigcap_{n=0}^{\infty} A_n
    &= \bigcap_{n \in \mathbb{N}} A_n.
  \end{align*}
  For a family indexed by the positive integers, the corresponding notation is
  \(\bigcup_{n=1}^{\infty} A_n\) and
  \(\bigcap_{n=1}^{\infty} A_n\). When the index range is clear from context,
  these are also abbreviated as \(\bigcup_n A_n\) and \(\bigcap_n A_n\).
\end{definition}

\subsubsection{Laws of set algebra}

Let \(A\), \(B\), and \(C\) be subsets of a universal set \(X\). The union
and intersection operations are \emph{commutative}:
\begin{align*}
  A \cup B &= B \cup A, &
  A \cap B &= B \cap A.
\end{align*}
They are also \emph{associative}:
\begin{align*}
  (A \cup B) \cup C &= A \cup (B \cup C), &
  (A \cap B) \cap C &= A \cap (B \cap C).
\end{align*}
The \emph{distributive laws} are
\begin{align*}
  A \cap (B \cup C) &= (A \cap B) \cup (A \cap C), \\
  A \cup (B \cap C) &= (A \cup B) \cap (A \cup C).
\end{align*}
The \emph{De Morgan laws} relate complements to unions and intersections:
\begin{align*}
  \complement{(A \cup B)} &= \complement{A} \cap \complement{B}, \\
  \complement{(A \cap B)} &= \complement{A} \cup \complement{B}.
\end{align*}
Finally, set difference can be expressed using intersection and complement:
\[
  A \setminus B = A \cap \complement{B}.
\]
Equivalently, replacing \(B\) by \(C\) gives
\[
  A \setminus C = A \cap \complement{C}.
\]

\begin{theorem}
  Let \(A\) and \(B\) be subsets of a universal set \(X\). Then
  \[
    A \subseteq B \iff \complement{B} \subseteq \complement{A}.
  \]
\end{theorem}

\begin{proof}
  Suppose first that \(A \subseteq B\), and let
  \(x \in \complement{B}\). Then \(x \notin B\). Since every element of
  \(A\) belongs to \(B\), it follows that \(x \notin A\), and hence
  \(x \in \complement{A}\). Therefore
  \(\complement{B} \subseteq \complement{A}\).

  Conversely, suppose that
  \(\complement{B} \subseteq \complement{A}\), and let \(x \in A\). If
  \(x \notin B\), then \(x \in \complement{B}\), so the assumed inclusion
  would imply that \(x \in \complement{A}\), contradicting \(x \in A\).
  Thus \(x \in B\), and therefore \(A \subseteq B\).
\end{proof}

\subsubsection{Proof by counterexample}

To disprove a universal claim, it is enough to find a single case in which the
claim fails. Such a case is called a \emph{counterexample}.

\begin{example}
  Consider the claim that
  \[
    (a+b)^2 = a^2+b^2
  \]
  for all real numbers \(a\) and \(b\). Let \(a=b=1\). Then
  \[
    (a+b)^2 = (1+1)^2 = 4,
    \qquad
    a^2+b^2 = 1^2+1^2 = 2.
  \]
  Since \(4 \neq 2\), this is a counterexample, and therefore the claim is
  false.
\end{example}

\subsubsection{Proof by contradiction}

In a proof by contradiction, we assume that the claim is false and show that
this assumption leads to a contradiction.

\begin{example}
  Suppose that \(A \subseteq B\) and \(A \nsubseteq C\). We prove by
  contradiction that \(B \nsubseteq C\).

  Assume, to the contrary, that \(B \subseteq C\). Since
  \(A \nsubseteq C\), there exists an element \(x \in A\) such that
  \(x \notin C\). But \(A \subseteq B\) implies that \(x \in B\), and
  \(B \subseteq C\) then implies that \(x \in C\). This contradicts
  \(x \notin C\). Therefore \(B \nsubseteq C\).
\end{example}

\subsubsection{Relations}

\begin{definition}
  A \emph{relation} from a set \(A\) to a set \(B\) is a subset
  \[
    R \subseteq A \times B.
  \]
  If \((a,b) \in R\), we say that \(a\) is related to \(b\) and write
  \(aRb\). A relation on \(A\) is a relation from \(A\) to itself, that is, a
  subset of \(A \times A\).
\end{definition}

\begin{definition}
  Let \(R \subseteq A \times B\) be a relation. The \emph{inverse relation}
  of \(R\) is
  \[
    R^{-1} = \{(b,a) \in B \times A : (a,b) \in R\}.
  \]
  Equivalently, \(bR^{-1}a\) if and only if \(aRb\).
\end{definition}

\begin{definition}
  Let \(R\) be a relation on a set \(A\). The relation \(R\) is
  \begin{enumerate}
    \item \emph{reflexive} if \(xRx\) for every \(x \in A\);
    \item \emph{irreflexive} if \((x,x) \notin R\) for every \(x \in A\);
    \item \emph{symmetric} if, for all \(x,y \in A\),
      \(xRy\) implies \(yRx\);
    \item \emph{antisymmetric} if, for all \(x,y \in A\),
      \(xRy\) and \(yRx\) imply \(x=y\);
    \item \emph{asymmetric} if, for all \(x,y \in A\),
      \(xRy\) implies \((y,x) \notin R\);
    \item \emph{transitive} if, for all \(x,y,z \in A\),
      \(xRy\) and \(yRz\) imply \(xRz\); and
    \item \emph{total} if, for all \(x,y \in A\), either \(xRy\) or
      \(yRx\).
  \end{enumerate}
  A relation is an \emph{equivalence relation} if it is reflexive, symmetric,
  and transitive.
\end{definition}

\subsection{17.09.2026}

\subsubsection{Properties of inverse relations}

\begin{theorem}
  Let \(R,S \subseteq A \times B\) be relations. Then
  \begin{enumerate}[label=(\alph*)]
    \item \((R^{-1})^{-1} = R\);
    \item \((R \cup S)^{-1} = R^{-1} \cup S^{-1}\);
    \item \((R \cap S)^{-1} = R^{-1} \cap S^{-1}\);
    \item \((R \setminus S)^{-1} = R^{-1} \setminus S^{-1}\).
  \end{enumerate}
\end{theorem}

\begin{proof}
  For (a), let \((a,b) \in A \times B\). By the definition of the inverse
  relation,
  \[
    (a,b) \in (R^{-1})^{-1}
    \iff (b,a) \in R^{-1}
    \iff (a,b) \in R.
  \]
  For the remaining identities, let \((b,a) \in B \times A\). Then
  \begin{align*}
    (b,a) \in (R \cup S)^{-1}
    &\iff (a,b) \in R \cup S \\
    &\iff (a,b) \in R \lor (a,b) \in S \\
    &\iff (b,a) \in R^{-1} \lor (b,a) \in S^{-1} \\
    &\iff (b,a) \in R^{-1} \cup S^{-1},
  \end{align*}
  which proves (b). Similarly,
  \begin{align*}
    (b,a) \in (R \cap S)^{-1}
    &\iff (a,b) \in R \land (a,b) \in S \\
    &\iff (b,a) \in R^{-1} \land (b,a) \in S^{-1} \\
    &\iff (b,a) \in R^{-1} \cap S^{-1},
  \end{align*}
  proving (c). Finally,
  \begin{align*}
    (b,a) \in (R \setminus S)^{-1}
    &\iff (a,b) \in R \land (a,b) \notin S \\
    &\iff (b,a) \in R^{-1} \land (b,a) \notin S^{-1} \\
    &\iff (b,a) \in R^{-1} \setminus S^{-1},
  \end{align*}
  which proves (d). In each case, the two sets have exactly the same elements.
\end{proof}

\subsubsection{Composition of relations}

\begin{definition}
  Let \(R \subseteq X \times Y\) and \(S \subseteq Y \times Z\).
  Their \emph{composite relation} is the relation from \(X\) to \(Z\)
  defined by
  \[
    S \circ R \coloneqq
    \set{(x,z) \in X \times Z :
    \exists y \in Y,\ (x,y) \in R \land (y,z) \in S}.
  \]
  Equivalently,
  \[
    x(S \circ R)z
    \iff \exists y \in Y : xRy \land ySz.
  \]
  Thus \(S \circ R\) means that we first follow \(R\), then \(S\).
\end{definition}

\begin{theorem}[Associativity of composition]
  Let \(R \subseteq X \times Y\), \(S \subseteq Y \times Z\), and
  \(T \subseteq Z \times U\). Then
  \[
    T \circ (S \circ R) = (T \circ S) \circ R.
  \]
\end{theorem}

\begin{proof}
  Let \((x,u) \in X \times U\). If
  \((x,u) \in T \circ (S \circ R)\), there is a \(z \in Z\) such that
  \(x(S \circ R)z\) and \(zTu\). The first relation supplies a
  \(y \in Y\) with \(xRy\) and \(ySz\). Since \(ySz\) and \(zTu\),
  we have \(y(T \circ S)u\), and hence
  \((x,u) \in (T \circ S) \circ R\).

  Conversely, if \((x,u) \in (T \circ S) \circ R\), there is a
  \(y \in Y\) such that \(xRy\) and \(y(T \circ S)u\). Choose
  \(z \in Z\) with \(ySz\) and \(zTu\). Then \(x(S \circ R)z\),
  so \((x,u) \in T \circ (S \circ R)\). Both inclusions hold, proving
  the equality.
\end{proof}

\subsubsection{Equivalence relations and equivalence classes}

Recall that a relation \(\sim\) on a set \(X\) is an
\emph{equivalence relation} if it is reflexive, symmetric, and transitive.

\begin{problem}[Lecture exercise]
  Let \(A = \set{2,3,4,5,6}\), and define a relation \(R\) on \(A\) by
  \[
    xRy \iff x \text{ and } y \text{ have a common prime factor}.
  \]
  Is \(R\) an equivalence relation?
\end{problem}

\begin{solution}
  The relation is reflexive: each \(x \in A\) has a prime factor, and that
  prime divides both occurrences of \(x\). More explicitly, \(2\) divides
  \(2,4,6\), \(3\) divides \(3\), and \(5\) divides \(5\).

  It is also symmetric. If \(xRy\), some prime \(p\) divides both \(x\)
  and \(y\). The same prime divides both \(y\) and \(x\), so \(yRx\).

  However, it is not transitive. We have \(4R6\), since \(2\) divides
  both \(4\) and \(6\), and \(6R3\), since \(3\) divides both \(6\)
  and \(3\). But \(4\) and \(3\) have no common prime factor: their
  only prime factors are \(2\) and \(3\), respectively. Thus
  \((4,3) \notin R\). Consequently, \(R\) is not an equivalence relation.
\end{solution}

\begin{definition}
  Let \(\sim\) be an equivalence relation on a set \(X\), and let
  \(a \in X\). The \emph{equivalence class} of \(a\) is
  \[
    [a]_{\sim} \coloneqq \set{x \in X : x \sim a}.
  \]
  The element \(a\) is called a \emph{representative} of this class.
  The set of all equivalence classes is the \emph{quotient set}
  \[
    X/{\sim} \coloneqq \set{[a]_{\sim} : a \in X}.
  \]
\end{definition}

The following consequence explains why equivalence classes divide a set into
separate parts; it supplements the definitions in the lecture.

\begin{proposition}
  Let \(\sim\) be an equivalence relation on \(X\). For every \(a,b \in X\),
  \begin{enumerate}[label=(\alph*)]
    \item \(a \in [a]_{\sim}\);
    \item \(a \sim b\) if and only if \([a]_{\sim} = [b]_{\sim}\);
    \item the classes \([a]_{\sim}\) and \([b]_{\sim}\) are either equal
      or disjoint.
  \end{enumerate}
  Therefore the equivalence classes form a \emph{partition} of \(X\):
  they are nonempty, their union is \(X\), and distinct classes are disjoint.
\end{proposition}

\begin{proof}
  Reflexivity gives \(a \sim a\), proving (a).

  For (b), suppose \(a \sim b\). If \(x \in [a]_{\sim}\), then
  \(x \sim a\), so transitivity gives \(x \sim b\). Hence
  \([a]_{\sim} \subseteq [b]_{\sim}\). Symmetry gives \(b \sim a\),
  and the same argument gives the reverse inclusion. Conversely, if the
  classes are equal, then (a) gives \(a \in [a]_{\sim} = [b]_{\sim}\),
  which means \(a \sim b\).

  For (c), if the classes intersect, choose an element \(c\) in their
  intersection. Then \(c \sim a\) and \(c \sim b\). By symmetry,
  \(a \sim c\), so transitivity gives \(a \sim b\). Part (b) now gives
  equality of the classes. Thus distinct classes must be disjoint.
  Finally, every class is a subset of \(X\), and (a) shows that each
  element of \(X\) belongs to a nonempty class. Their union is therefore
  all of \(X\), proving the partition claim.
\end{proof}

\subsubsection{Functions}

\begin{definition}
  Let \(X\) and \(Y\) be nonempty sets. A \emph{function} (or
  \emph{mapping}) \(f\) from \(X\) to \(Y\) assigns to each element
  \(x \in X\) exactly one element \(y \in Y\). We write
  \[
    f\colon X \longrightarrow Y,
    \qquad y = f(x).
  \]
  The set \(X\) is the \emph{domain}, and \(Y\) is the \emph{codomain}.
\end{definition}

In terms of relations, the graph of \(f\) is
\[
  G_f = \set{(x,f(x)) : x \in X} \subseteq X \times Y.
\]
A relation \(F \subseteq X \times Y\) is the graph of a function precisely
when, for every \(x \in X\), there exists exactly one \(y \in Y\) such
that \((x,y) \in F\). Existence ensures that the function is defined at
every input, and uniqueness ensures that its value is unambiguous.

\subsection{21.09.2026}

\subsubsection{Inverse functions, images, and preimages}

The inverse relation of a function \(f\colon X \to Y\) is a relation from
\(Y\) to \(X\), but it is not generally a function. If this inverse
relation is a function from \(Y\) to \(X\), it is called the
\emph{inverse function} of \(f\), denoted by \(f^{-1}\).

\begin{definition}
  Let \(f\colon X \to Y\) be a function. The \emph{image} of a subset
  \(A \subseteq X\) is
  \[
    f(A) \coloneqq \set{f(x) : x \in A}.
  \]
  The \emph{preimage} of a subset \(B \subseteq Y\) is
  \[
    f^{-1}(B) \coloneqq \set{x \in X : f(x) \in B}.
  \]
\end{definition}

\begin{remark}
  The preimage \(f^{-1}(B)\) is defined for every function \(f\), even
  when \(f\) has no inverse function.
\end{remark}

\subsubsection{Images and preimages of unions and intersections}

\begin{theorem}
  Let \(f\colon X \to Y\). For all \(A,B \subseteq X\),
  \begin{enumerate}[label=(\roman*)]
    \item \(f(A \cup B) = f(A) \cup f(B)\);
    \item \(f(A \cap B) \subseteq f(A) \cap f(B)\).
  \end{enumerate}
\end{theorem}

\begin{theorem}
  Let \(f\colon X \to Y\). For all \(A,B \subseteq Y\),
  \begin{enumerate}[label=(\roman*)]
    \item \(f^{-1}(A \cup B) = f^{-1}(A) \cup f^{-1}(B)\);
    \item \(f^{-1}(A \cap B) = f^{-1}(A) \cap f^{-1}(B)\).
  \end{enumerate}
\end{theorem}

\begin{proof}
  Let \(A,B \subseteq Y\) and \(x \in X\). For (i),
  \begin{align*}
    x \in f^{-1}(A \cup B)
    &\iff f(x) \in A \lor f(x) \in B \\
    &\iff x \in f^{-1}(A) \lor x \in f^{-1}(B) \\
    &\iff x \in f^{-1}(A) \cup f^{-1}(B).
  \end{align*}
  Similarly, for (ii),
  \begin{align*}
    x \in f^{-1}(A \cap B)
    &\iff f(x) \in A \land f(x) \in B \\
    &\iff x \in f^{-1}(A) \land x \in f^{-1}(B) \\
    &\iff x \in f^{-1}(A) \cap f^{-1}(B).
  \end{align*}
  Thus both identities hold.
\end{proof}

\subsubsection{Injective, surjective, and bijective functions}

\begin{definition}
  A function \(f\colon X \to Y\) is \emph{injective} if distinct inputs
  have distinct images: for all \(x_1,x_2 \in X\),
  \[
    x_1 \neq x_2 \implies f(x_1) \neq f(x_2).
  \]
\end{definition}

\begin{definition}
  A function \(f\colon X \to Y\) is \emph{surjective} if, for every
  \(y \in Y\), there exists \(x \in X\) such that \(f(x)=y\).
  Equivalently, \(f(X)=Y\).
\end{definition}

\begin{definition}
  A function \(f\colon X \to Y\) is \emph{bijective} if it is both
  injective and surjective.
\end{definition}

\begin{theorem}
  A function \(f\colon X \to Y\) has an inverse function
  \(f^{-1}\colon Y \to X\) if and only if \(f\) is bijective.
\end{theorem}

\subsubsection{Lecture exercise: the floor function}

\begin{problem}[Lecture exercise]
  Let
  \[
    f(x) = \lfloor x \rfloor
    = \max\set{n \in \ZZ : n \leq x}.
  \]
  Determine whether \(f\) is injective, surjective, or bijective for each
  of the following choices of domain and codomain:
  \begin{enumerate}[label=(\alph*)]
    \item \(f\colon \RR \to \ZZ\);
    \item \(f\colon \ZZ \to \RR\);
    \item \(f\colon \ZZ \to \ZZ\);
    \item \(f\colon \RR \to \RR\).
  \end{enumerate}
\end{problem}

\begin{solution}
  \begin{enumerate}[label=(\alph*)]
    \item The function is not injective, since
      \(f(1)=f(1.5)=1\), although \(1 \neq 1.5\). It is surjective:
      for every \(y \in \ZZ\), any \(x \in [y,y+1)\) satisfies
      \(f(x)=y\). Therefore it is not bijective.
    \item For every \(x \in \ZZ\), we have \(f(x)=x\), so the function
      is injective. It is not surjective, since \(1/2 \in \RR\) is not
      the image of any integer. Therefore it is not bijective.
    \item Here \(f(x)=x\) for all \(x \in \ZZ\), so \(f\) is the
      identity function on \(\ZZ\). It is both injective and
      surjective, and hence bijective.
    \item The function is neither injective nor surjective:
      \(f(1)=f(1.5)\), and \(1/2\) is not in its image.
      Consequently, it is not bijective. \qedhere
  \end{enumerate}
\end{solution}

\subsubsection{Images followed by preimages}

\begin{theorem}
  Let \(f\colon X \to Y\), \(A \subseteq X\), and \(B \subseteq Y\).
  Then
  \begin{enumerate}[label=(\roman*)]
    \item \(A \subseteq f^{-1}(f(A))\);
    \item \(f(f^{-1}(B)) \subseteq B\).
  \end{enumerate}
\end{theorem}

\begin{proof}
  For (i), let \(x \in A\). Then \(f(x) \in f(A)\), so the
  definition of preimage gives \(x \in f^{-1}(f(A))\).

  For (ii), let \(y \in f(f^{-1}(B))\). There exists
  \(x \in f^{-1}(B)\) with \(y=f(x)\). Since \(x \in f^{-1}(B)\)
  means that \(f(x) \in B\), we obtain \(y \in B\).
\end{proof}

\begin{remark}
  These inclusions need not be equalities. For \(f\colon\RR\to\RR\),
  \(f(x)=x^2\), and \(A=\set{1}\), we have
  \(f^{-1}(f(A))=\set{-1,1}\). For the same function and
  \(B=\set{-1}\), we have \(f(f^{-1}(B))=\emptyset\).
  Here \(f^{-1}\) denotes a preimage, not an inverse function.
\end{remark}

\subsubsection{Composition of functions}

\begin{definition}
  Let \(f\colon X \to Y\) and \(g\colon Y \to Z\) be functions.
  Their \emph{composite function} is
  \[
    g\circ f\colon X\to Z,
    \qquad (g\circ f)(x)=g(f(x)) \quad (x\in X).
  \]
  Thus we apply \(f\) first and \(g\) second.
\end{definition}

\begin{theorem}\label{thm:composition-bijections}
  If \(f\colon X\to Y\) and \(g\colon Y\to Z\) are bijections,
  then \(g\circ f\colon X\to Z\) is a bijection.
\end{theorem}

\begin{proof}
  To prove injectivity, suppose that
  \((g\circ f)(x_1)=(g\circ f)(x_2)\). Then
  \(g(f(x_1))=g(f(x_2))\). Injectivity of \(g\) gives
  \(f(x_1)=f(x_2)\), and injectivity of \(f\) gives \(x_1=x_2\).

  To prove surjectivity, let \(z\in Z\). Since \(g\) is surjective,
  there is \(y\in Y\) with \(g(y)=z\). Since \(f\) is surjective,
  there is \(x\in X\) with \(f(x)=y\). Thus
  \((g\circ f)(x)=g(y)=z\).
\end{proof}

\subsubsection{Identity maps and inverse functions}

\begin{definition}
  The \emph{identity map} on a set \(X\) is the function
  \[
    \operatorname{id}_X\colon X\to X,
    \qquad \operatorname{id}_X(x)=x \quad (x\in X).
  \]
\end{definition}

\begin{theorem}\label{thm:inverse-identities}
  Let \(f\colon X\to Y\) and \(g\colon Y\to X\). Then \(g\)
  is the inverse function of \(f\) if and only if
  \[
    g\circ f=\operatorname{id}_X
    \qquad\text{and}\qquad
    f\circ g=\operatorname{id}_Y.
  \]
\end{theorem}

\begin{proof}
  Suppose first that \(g=f^{-1}\). For every \(x\in X\), the
  inverse sends \(f(x)\) back to \(x\), so \(g(f(x))=x\).
  For every \(y\in Y\), the value \(g(y)\) is the unique element
  mapped to \(y\) by \(f\), so \(f(g(y))=y\). These are the
  required identities.

  Conversely, suppose both identities hold. If \(f(x_1)=f(x_2)\),
  applying \(g\) gives
  \[
    x_1=g(f(x_1))=g(f(x_2))=x_2,
  \]
  so \(f\) is injective. For every \(y\in Y\), the element
  \(g(y)\in X\) satisfies \(f(g(y))=y\), so \(f\) is surjective.
  Thus each \(y\) has exactly one preimage, namely \(g(y)\), proving
  that \(g=f^{-1}\).
\end{proof}

\begin{corollary}\label{cor:inverse-bijection}
  The inverse of a bijection is a bijection.
\end{corollary}

\begin{proof}
  Let \(f\colon X\to Y\) be a bijection. If
  \(f^{-1}(y_1)=f^{-1}(y_2)\), applying \(f\) gives \(y_1=y_2\),
  so \(f^{-1}\) is injective. For any \(x\in X\), set \(y=f(x)\).
  Then \(f^{-1}(y)=x\), so \(f^{-1}\) is also surjective.
\end{proof}

\subsubsection{Infinite sets and their cardinalities}

\begin{definition}
  Sets \(A\) and \(B\) have \emph{equal cardinality} (or are
  \emph{equinumerous}) if there exists a bijection \(A\to B\).
  We write \(A\sim B\).
\end{definition}

\begin{definition}
  A set \(A\) is \emph{countably infinite} if \(A\sim\NN\).
  Equivalently, its elements can be listed as
  \(a_0,a_1,a_2,\ldots\), with every element appearing exactly once.
\end{definition}

\begin{remark}
  The Finnish term \emph{numeroituva} in this lecture is defined by
  \(A\sim\NN\). We use ``countably infinite'' for this precise meaning;
  ``countable'' is also commonly used to include finite sets.
  Recall that \(\NN=\set{0,1,2,\ldots}\) in these notes.
\end{remark}

For this discussion, we also allow functions with empty domain: the empty
relation is the unique function \(\emptyset\to B\) for any set \(B\).
In particular, \(\operatorname{id}_{\emptyset}\) is a bijection from
\(\emptyset\) to itself. This extends the earlier convention of discussing
functions between nonempty sets and allows us to include all subsets below.

\begin{theorem}
  Let \(X\) be a set. Equal cardinality is an equivalence relation on
  \(\powerset{X}\).
\end{theorem}

\begin{proof}
  We verify reflexivity, symmetry, and transitivity.

  \emph{Reflexivity.} Let \(A\subseteq X\). The identity map
  \(\operatorname{id}_A\colon A\to A\) is injective, since equal
  outputs are equal inputs, and surjective, since each \(a\in A\)
  is its own image. Hence \(A\sim A\), including when \(A=\emptyset\).

  \emph{Symmetry.} Let \(A,B\subseteq X\) and suppose \(A\sim B\).
  Choose a bijection \(f\colon A\to B\). By
  \cref{cor:inverse-bijection}, its inverse \(f^{-1}\colon B\to A\)
  is a bijection, so \(B\sim A\).

  \emph{Transitivity.} Let \(A,B,C\subseteq X\), and suppose
  \(A\sim B\) and \(B\sim C\). Choose bijections
  \(f\colon A\to B\) and \(g\colon B\to C\).
  By \cref{thm:composition-bijections},
  \(g\circ f\colon A\to C\) is a bijection. Therefore \(A\sim C\).

  Thus \(\sim\) is an equivalence relation on \(\powerset{X}\).
\end{proof}

\subsection{28.09.2026}

\subsubsection{Finite and countable sets}

We continue the discussion of cardinality. Throughout, \(\NN=\{0,1,2,\ldots\}\).
For \(n\in\NN\), write \([n]=\{1,\ldots,n\}\), with
\([0]=\emptyset\).

\begin{definition}
  A set \(A\) is \emph{finite} if there is a bijection \([n]\to A\)
  for some \(n\in\NN\). Otherwise it is \emph{infinite}.
  A set is \emph{countable} if it is finite or countably infinite, and
  \emph{uncountable} if it is not countable.
\end{definition}

\begin{remark}
  In today's lecture, the Finnish term \emph{numeroituva} includes finite
  sets. This is the meaning of ``countable'' used here; ``countably
  infinite'' continues to mean equinumerous with \(\NN\).
\end{remark}

\begin{theorem}
  The rational numbers are countably infinite: \(\QQ\sim\NN\).
\end{theorem}

\begin{proof}
  First list the positive rational numbers in the array
  \[
    \begin{array}{ccccc}
      1/1 & 2/1 & 3/1 & 4/1 & \cdots \\
      1/2 & 2/2 & 3/2 & 4/2 & \cdots \\
      1/3 & 2/3 & 3/3 & 4/3 & \cdots \\
      \vdots & \vdots & \vdots & \vdots & \ddots
    \end{array}
  \]
  Visit the entries \(p/q\), where \(p,q\geq1\), in increasing order of
  \(p+q\), and within each diagonal in increasing order of \(p\).
  Each diagonal has finitely many entries, so each pair \((p,q)\) is
  visited after finitely many steps. Keep an entry only if its rational
  value has not already appeared. Every positive rational has a
  representation \(p/q\), so it appears in the resulting list, and no
  value appears twice. The list is infinite because it contains every
  positive integer. Denote it by \(q_1,q_2,q_3,\ldots\).

  Define \(f\colon\NN\to\QQ\) by
  \[
    f(0)=0,\qquad f(2n-1)=q_n,\qquad f(2n)=-q_n
    \quad(n\geq1).
  \]
  Every rational number is zero, positive, or negative, so \(f\) is
  surjective. The positive values are distinct, as are the negative
  values, and values from different sign classes cannot be equal.
  Thus \(f\) is injective as well, proving the claim.
\end{proof}

\subsubsection{Uncountability of the real numbers}

\begin{theorem}[Cantor's diagonal argument]
  The interval \([0,1]\), and consequently \(\RR\), is uncountable.
\end{theorem}

\begin{proof}
  Suppose first that \([0,1]\) is countable. It is infinite, since it
  contains the distinct numbers \(1,1/2,1/3,\ldots\). Hence we could
  list all its elements as \(x_1,x_2,x_3,\ldots\).

  For each \(x_n<1\), choose its decimal expansion that is not
  eventually all \(9\)'s; use trailing zeros for terminating decimals.
  Represent \(1\) by \(1.000\ldots\). Write
  \[
    x_n=a_n.d_{n1}d_{n2}d_{n3}\ldots,
    \qquad d_{nj}\in\{0,1,\ldots,9\},
  \]
  where \(a_n\) is the integer part. Define
  \[
    b_n=
    \begin{cases}
      2,&d_{nn}=1,\\
      1,&d_{nn}\neq1,
    \end{cases}
    \qquad y=\sum_{n=1}^{\infty}b_n10^{-n}=0.b_1b_2b_3\ldots.
  \]
  Since \(1\leq b_n\leq2\), the geometric series gives
  \(1/9\leq y\leq2/9\), so \(y\in[0,1]\).

  We check that the decimal comparison is valid even for numbers with
  two decimal representations. Fix \(n\). If \(x_n=1\), then
  \(x_n\neq y\). Otherwise, the decimal digits of \(x_n\) and \(y\)
  differ at position \(n\), so they have a first differing position
  \(k\leq n\). The difference at position \(k\) has absolute value
  at least \(10^{-k}\), while the difference of the remaining tails
  has absolute value at most
  \[
    \sum_{j>k}9\cdot10^{-j}=10^{-k}.
  \]
  Equality in this tail bound would require every later pair of digits
  to differ by \(9\), which is impossible because every digit of
  \(y\) is \(1\) or \(2\). The first difference therefore cannot be
  cancelled by the tail. Thus \(y\neq x_n\) for every \(n\), contrary
  to the assumption that the list contains all of \([0,1]\).

  Finally, suppose \(\RR\) were countable. It is infinite, so it would
  have a listing without repetitions. Retain just the entries in
  \([0,1]\). Infinitely many entries remain, and every element of
  \([0,1]\) occurs at a finite position in the original list, hence
  also in the retained list. This would enumerate \([0,1]\), a
  contradiction. Therefore \(\RR\) is uncountable.
\end{proof}

\subsubsection{The size of a finite set}

\begin{lemma}[Pigeonhole principle]\label{lem:finite-pigeonhole}
  If \(n,k\in\NN\) and \(k<n\), there is no injection \([n]\to[k]\).
\end{lemma}

\begin{proof}
  We use induction on \(k\), proving the assertion for every \(n>k\).
  For \(k=0\), no function from the nonempty set \([n]\) to the empty
  set exists.

  Suppose the assertion holds for \(k\), and let \(n>k+1\). Assume
  there is an injection \(f\colon[n]\to[k+1]\). Set \(a=f(n)\), and
  let \(\tau\colon[k+1]\to[k+1]\) interchange \(a\) and \(k+1\),
  fixing all other elements. If \(a=k+1\), take \(\tau\) to be the
  identity. In either case \(g=\tau\circ f\) is injective and
  \(g(n)=k+1\). Injectivity implies \(g(i)\neq k+1\) for every
  \(i\in[n-1]\). Thus the restriction of \(g\) is an injection
  \([n-1]\to[k]\). Since \(n-1>k\), this contradicts the induction
  hypothesis. The result follows by induction.
\end{proof}

\begin{theorem}
  For every finite set \(A\), there is a unique \(n\in\NN\) such
  that \([n]\sim A\).
\end{theorem}

\begin{proof}
  Existence is the definition of finiteness. Suppose there are
  bijections \(f\colon[n]\to A\) and \(g\colon[k]\to A\), with
  \(n\neq k\). By interchanging the names if necessary, assume
  \(k<n\). Then \(g^{-1}\circ f\colon[n]\to[k]\) is a bijection,
  hence an injection, contradicting \cref{lem:finite-pigeonhole}.
  Therefore \(n=k\).
\end{proof}

\begin{definition}
  The unique integer above is the \emph{cardinality} (or \emph{size})
  of the finite set \(A\), denoted by \(\abs{A}\) or \(N(A)\).
  In particular, \(\abs{\emptyset}=0\).
\end{definition}

The following elementary fact ensures that the intersections and differences
in our counting formulas are finite.

\begin{lemma}\label{lem:finite-subsets}
  Every subset \(S\) of a finite set \(A\) is finite, and
  \(\abs{S}\leq\abs{A}\).
\end{lemma}

\begin{proof}
  Choose a bijection \(f\colon[n]\to A\). Scan the indices
  \(1,\ldots,n\) in order, retaining precisely those for which
  \(f(i)\in S\). This finite procedure produces a list
  \(i_1<\cdots<i_r\), where \(0\leq r\leq n\). The map
  \(j\mapsto f(i_j)\) is a bijection \([r]\to S\): injectivity
  follows from that of \(f\), and surjectivity follows because every
  element of \(S\) has an index that was retained. This also covers
  \(S=\emptyset\), using the empty list. Hence \(\abs{S}=r\leq n\).
\end{proof}

\subsubsection{Unions and the inclusion--exclusion principle}

\begin{lemma}[Disjoint unions]\label{lem:disjoint-counting}
  If \(A\) and \(B\) are finite and disjoint, then \(A\cup B\) is
  finite and \(\abs{A\cup B}=\abs{A}+\abs{B}\).
\end{lemma}

\begin{proof}
  Let \(m=\abs{A}\), \(n=\abs{B}\), and choose bijections
  \(f\colon[m]\to A\) and \(g\colon[n]\to B\). Define
  \[
    h\colon[m+n]\to A\cup B,\qquad
    h(i)=
    \begin{cases}
      f(i),&1\leq i\leq m,\\
      g(i-m),&m<i\leq m+n.
    \end{cases}
  \]
  Each branch is injective, and values from different branches lie in
  disjoint sets, so \(h\) is injective. Every element of \(A\cup B\)
  is reached by one of the branches, so \(h\) is surjective. Thus
  \(h\) is a bijection and the cardinality is \(m+n\). The same
  construction works when either set is empty.
\end{proof}

\begin{theorem}\label{thm:two-set-counting}
  Let \(A,B\) be finite sets. Then \(A\cup B\) is finite and
  \begin{enumerate}[label=(\alph*)]
    \item \(\abs{A\cup B}=\abs{A}+\abs{B}-\abs{A\cap B}\);
    \item if \(A\cap B=\emptyset\), then
      \(\abs{A\cup B}=\abs{A}+\abs{B}\);
    \item \(\abs{A\cup B}=\abs{A}+\abs{B\setminus A}\);
    \item if \(A\subseteq B\), then
      \(\abs{B}=\abs{A}+\abs{B\setminus A}\).
  \end{enumerate}
\end{theorem}

\begin{proof}
  By \cref{lem:finite-subsets}, both \(B\setminus A\) and
  \(A\cap B\) are finite. The decompositions
  \[
    A\cup B=A\cup(B\setminus A),\qquad
    B=(A\cap B)\cup(B\setminus A)
  \]
  are disjoint unions. Applying \cref{lem:disjoint-counting} proves
  finiteness of \(A\cup B\), part (c), and the identity
  \[
    \abs{B}=\abs{A\cap B}+\abs{B\setminus A}.
  \]
  Substitute \(\abs{B\setminus A}=\abs{B}-\abs{A\cap B}\) into
  (c) to obtain (a). Part (b) follows from (a) and
  \(\abs{\emptyset}=0\). If \(A\subseteq B\), then \(A\cup B=B\),
  so (c) gives (d).
\end{proof}

\begin{theorem}[Inclusion--exclusion for three sets]
  Let \(A,B,C\) be finite sets. Then
  \[
    \begin{aligned}
      \abs{A\cup B\cup C}
      ={}&\abs{A}+\abs{B}+\abs{C}\\
      &-\abs{A\cap B}-\abs{A\cap C}-\abs{B\cap C}\\
      &+\abs{A\cap B\cap C}.
    \end{aligned}
  \]
\end{theorem}

\begin{proof}
  Apply \cref{thm:two-set-counting}(a) to \(A\cup B\) and \(C\):
  \[
    \abs{A\cup B\cup C}
    =\abs{A\cup B}+\abs{C}-\abs{(A\cup B)\cap C}.
  \]
  Distributivity gives
  \((A\cup B)\cap C=(A\cap C)\cup(B\cap C)\), so another
  application of the same formula yields
  \[
    \abs{(A\cup B)\cap C}
    =\abs{A\cap C}+\abs{B\cap C}-\abs{A\cap B\cap C}.
  \]
  Substituting this and
  \(\abs{A\cup B}=\abs{A}+\abs{B}-\abs{A\cap B}\) into the first
  equality gives the stated formula.
\end{proof}

\subsubsection{Cardinality of Cartesian products}

\begin{theorem}\label{thm:finite-product-counting}
  If \(A,B\) are finite sets, then \(A\times B\) is finite and
  \[
    \abs{A\times B}=\abs{A}\abs{B}.
  \]
\end{theorem}

\begin{proof}
  If either factor is empty, the product is empty and both sides are
  zero. Otherwise, write \(A=\{a_1,\ldots,a_m\}\) with distinct
  elements. Then
  \[
    A\times B=\bigcup_{i=1}^{m}(\{a_i\}\times B).
  \]
  These sets are pairwise disjoint: an ordered pair in both the
  \(i\)th and \(j\)th sets would have first coordinate both \(a_i\)
  and \(a_j\), forcing \(i=j\). For each \(i\), the map
  \(B\to\{a_i\}\times B\), \(b\mapsto(a_i,b)\), is a bijection,
  with inverse \((a_i,b)\mapsto b\). Thus each such set is finite
  and has cardinality \(\abs{B}\).

  Repeated use of \cref{lem:disjoint-counting} shows that a union of
  finitely many pairwise disjoint finite sets is finite and its size
  is the sum of their sizes. Formally, the assertion for one set is
  immediate; adding the next set preserves disjointness from the
  preceding union, so the two-set lemma proves the induction step.
  Consequently,
  \[
    \abs{A\times B}
    =\sum_{i=1}^{m}\abs{\{a_i\}\times B}
    =m\abs{B}=\abs{A}\abs{B}.\qedhere
  \]
\end{proof}

\begin{theorem}
  Let \(n\geq1\), and let \(A_1,\ldots,A_n\) be finite sets.
  Their Cartesian product is finite and
  \[
    \abs{A_1\times\cdots\times A_n}
    =\prod_{i=1}^{n}\abs{A_i}.
  \]
\end{theorem}

\begin{proof}
  We use induction on \(n\). For \(n=1\), the identification
  \((a_1)\mapsto a_1\) is a bijection, so the formula holds.
  Suppose it holds for \(n\) factors. The map
  \[
    (a_1,\ldots,a_n,a_{n+1})
    \longmapsto ((a_1,\ldots,a_n),a_{n+1})
  \]
  is a bijection from \(A_1\times\cdots\times A_{n+1}\) to
  \((A_1\times\cdots\times A_n)\times A_{n+1}\), since its inverse
  simply removes the inner parentheses. The first factor is finite
  by the induction hypothesis. Hence \cref{thm:finite-product-counting}
  and the induction hypothesis give
  \[
    \abs{A_1\times\cdots\times A_{n+1}}
    =\abs{A_1\times\cdots\times A_n}\abs{A_{n+1}}
    =\prod_{i=1}^{n+1}\abs{A_i}.
  \]
  This proves both finiteness and the formula by induction, including
  cases with an empty factor.
\end{proof}

\subsubsection{The pigeonhole principle}

\begin{theorem}[Pigeonhole principle]
  If at least \(k+1\) objects are placed in \(k\) boxes, where
  \(k\geq1\), then at least one box contains at least two objects.
\end{theorem}

\begin{proof}
  If each box contained at most one object, the \(k\) boxes would
  contain at most \(k\) objects altogether, a contradiction.
\end{proof}

\begin{theorem}[Generalized pigeonhole principle]
  If \(n\in\NN\) objects are placed in \(k\geq1\) boxes, then at
  least one box contains at least \(\lceil n/k\rceil\) objects.
  Here \(\lceil x\rceil\) denotes the smallest integer greater than
  or equal to \(x\).
\end{theorem}

\begin{proof}
  Put \(r=\lceil n/k\rceil\). If every box contained fewer than
  \(r\) objects, each would contain at most \(r-1\). The total
  number of objects would then satisfy
  \[
    n\leq k(r-1)<k\frac{n}{k}=n,
  \]
  where the strict inequality follows from \(r-1<n/k\).
  This is a contradiction.
\end{proof}

\begin{example}
  There are \(524\) participants. How many are guaranteed to share
  a birthday, or a birth month?

  There are \(366\) possible birthdays, allowing for February 29.
  By the generalized pigeonhole principle, at least
  \[
    \left\lceil\frac{524}{366}\right\rceil=2
  \]
  participants share the same birthday (day and month). There are
  \(12\) possible birth months, so at least
  \[
    \left\lceil\frac{524}{12}\right\rceil=44
  \]
  participants were born in the same month.
\end{example}

\subsubsection{Algebraic properties of complex numbers}

\begin{theorem}
  The complex numbers form an abelian group under addition, with
  identity \(0=(0,0)\). The nonzero complex numbers
  \(\CC^{\times}=\CC\setminus\{0\}\) form an abelian group under
  multiplication, with identity \(1=(1,0)\). Moreover,
  \((\CC,+,\cdot,0,1)\) is a field.
\end{theorem}

\begin{proof}
  We check the group laws and distributivity directly from the pair
  definitions. Let \(z=(a,b)\), \(w=(c,d)\), and \(u=(e,f)\).

  Addition is closed in \(\CC\), and the corresponding laws in
  \(\RR\) give
  \[
    z+w=(a+c,b+d)=(c+a,d+b)=w+z
  \]
  and
  \[
    (z+w)+u=(a+c+e,b+d+f)=z+(w+u).
  \]
  Also, \(z+(0,0)=z\), and
  \(z+(-a,-b)=(0,0)\). Thus \((\CC,+,0)\) is an abelian group.

  Multiplication is closed in \(\CC\) and commutative, since
  \[
    zw=(ac-bd,ad+bc)=(ca-db,cb+da)=wz.
  \]
  Expanding both parenthesizations gives
  \begin{align*}
    (zw)u
    &=(ace-bde-adf-bcf,\;acf-bdf+ade+bce),\\
    z(wu)
    &=(ace-adf-bcf-bde,\;acf+ade+bce-bdf).
  \end{align*}
  The coordinates agree, so multiplication is associative.
  Its identity is \((1,0)\), because \((a,b)(1,0)=(a,b)\).
  If \(z\neq0\), then \(a^2+b^2>0\), and
  \[
    z^{-1}=\left(\frac{a}{a^2+b^2},\frac{-b}{a^2+b^2}\right)
  \]
  is nonzero and satisfies
  \[
    zz^{-1}
    =\left(\frac{a^2+b^2}{a^2+b^2},
    \frac{-ab+ba}{a^2+b^2}\right)=(1,0).
  \]
  Products of nonzero elements are nonzero: if \(zw=0\) and
  \(z\neq0\), associativity gives
  \(w=(z^{-1}z)w=z^{-1}(zw)=z^{-1}0=0\).
  Thus multiplication restricts to an abelian group on \(\CC^{\times}\).

  Finally,
  \begin{align*}
    z(w+u)
    &=(a(c+e)-b(d+f),\;a(d+f)+b(c+e))\\
    &=(ac-bd,ad+bc)+(ae-bf,af+be)\\
    &=zw+zu.
  \end{align*}
  Commutativity gives the other distributive law. Since
  \((0,0)\neq(1,0)\), these properties show that \(\CC\) is a field.
\end{proof}

\subsubsection{Real and imaginary parts}

We identify each \(a\in\RR\) with \((a,0)\in\CC\). This identification
preserves addition and multiplication, since
\[
  (a,0)+(c,0)=(a+c,0),\qquad (a,0)(c,0)=(ac,0).
\]

\begin{definition}
  For \(z=(a,b)\in\CC\), its \emph{real part} and
  \emph{imaginary part} are
  \[
    \operatorname{Re}z=a,\qquad \operatorname{Im}z=b.
  \]
  If \(b=0\), then \(z\) is real. If \(a=0\), then \(z\) is
  \emph{purely imaginary} (including zero under this convention).
\end{definition}

\begin{definition}
  The complex number \(i=(0,1)\) is the \emph{imaginary unit}.
\end{definition}

By the multiplication rule,
\[
  i^2=(0,1)(0,1)=(-1,0)=-1.
\]
Every complex number can therefore be written as
\[
  z=(a,b)=(a,0)+(0,b)=(a,0)+(b,0)(0,1)=a+bi.
\]
Thus \(\CC=\set{a+bi:a,b\in\RR}\).

\begin{example}
  Let \(z=a+bi\) and \(w=c+di\). Then
  \begin{align*}
    z+w&=(a+bi)+(c+di)=(a+c)+(b+d)i,\\
    zw&=(a+bi)(c+di)\\
    &=ac+adi+bci+bdi^2\\
    &=(ac-bd)+(ad+bc)i.
  \end{align*}
\end{example}

\subsubsection{Complex conjugation and modulus}

\begin{definition}
  The \emph{complex conjugate} of \(z=a+bi\in\CC\) is
  \[
    \overline{z}=a-bi.
  \]
\end{definition}

\begin{definition}
  The \emph{modulus} of \(z=a+bi\in\CC\) is
  \[
    \abs{z}=\sqrt{a^2+b^2}.
  \]
  It is the distance of \((a,b)\) from the origin \((0,0)\) in the
  complex plane.
\end{definition}

\begin{proposition}
  For every \(z\in\CC\),
  \[
    z\overline{z}=\abs{z}^2,\qquad
    \abs{z}=\sqrt{z\overline{z}}.
  \]
\end{proposition}

\begin{proof}
  Write \(z=a+bi\). Then
  \[
    z\overline{z}=(a+bi)(a-bi)=a^2-b^2i^2=a^2+b^2=\abs{z}^2.
  \]
  Taking the nonnegative real square root gives the second identity.
\end{proof}

\subsubsection{Properties of conjugation and modulus}

\begin{proposition}
  For all \(z,w\in\CC\),
  \begin{enumerate}[label=(\alph*)]
    \item \(\overline{z+w}=\overline z+\overline w\);
    \item \(\overline{zw}=\overline z\,\overline w\);
    \item \(\overline{\overline z}=z\);
    \item \(\operatorname{Im}z=0\) if and only if \(\overline z=z\).
  \end{enumerate}
\end{proposition}

\begin{proof}
  Write \(z=a+bi\) and \(w=c+di\), where \(a,b,c,d\in\RR\).
  Then
  \begin{align*}
    \overline{z+w}
    &=\overline{(a+c)+(b+d)i}
    =(a+c)-(b+d)i=\overline z+\overline w,\\
    \overline{zw}
    &=\overline{(ac-bd)+(ad+bc)i}
    =(ac-bd)-(ad+bc)i\\
    &=(a-bi)(c-di)=\overline z\,\overline w.
  \end{align*}
  Also, \(\overline{\overline z}=\overline{a-bi}=a+bi=z\).
  Finally, \(\overline z=z\) holds precisely when
  \(a-bi=a+bi\), which is equivalent to \(b=0\), or
  \(\operatorname{Im}z=0\).
\end{proof}

\begin{proposition}
  For all \(z,w\in\CC\),
  \begin{enumerate}[label=(\alph*)]
    \item \(\abs{z}=\sqrt{z\overline z}\);
    \item \(\abs{zw}=\abs{z}\abs{w}\);
    \item \(zw=0\) if and only if \(z=0\) or \(w=0\);
    \item \(\abs{z}=0\) if and only if \(z=0\).
  \end{enumerate}
\end{proposition}

\begin{proof}
  Part (a) was proved at the end of the previous lecture.
  Using conjugation of products and commutativity,
  \[
    \abs{zw}^{2}
    =(zw)\overline{zw}
    =(z\overline z)(w\overline w)
    =\abs{z}^{2}\abs{w}^{2}.
  \]
  Moduli are nonnegative, so taking square roots proves (b).
  If \(z=a+bi\), then \(\abs{z}^{2}=a^{2}+b^{2}\).
  Since both squares are nonnegative, \(\abs{z}=0\) exactly when
  \(a=b=0\), proving (d). Now (b) and (d) show that
  \(zw=0\) exactly when \(\abs{z}\abs{w}=0\), which is equivalent
  to \(z=0\) or \(w=0\). This proves (c).
\end{proof}

\subsubsection{Reciprocals and quotients}

\begin{definition}
  If \(z\in\CC\setminus\{0\}\), its \emph{reciprocal} is
  \[
    z^{-1}=\frac1z=\frac{\overline z}{\abs{z}^{2}}.
  \]
  For \(w\neq0\), division is defined by
  \[
    \frac zw=zw^{-1}=\frac{z\overline w}{\abs{w}^{2}}.
  \]
\end{definition}

\begin{proof}[Verification]
  Because \(z\neq0\), the preceding proposition gives
  \(\abs{z}^{2}>0\), so the formula is defined. Moreover,
  \[
    z\frac{\overline z}{\abs{z}^{2}}
    =\frac{z\overline z}{\abs{z}^{2}}=1.
  \]
  Thus it agrees with the multiplicative inverse established earlier.
  The quotient formula follows by substituting the reciprocal of \(w\).
\end{proof}

\begin{example}
  If \(z=a+bi\neq0\), then
  \[
    \frac1{a+bi}
    =\frac{a-bi}{a^{2}+b^{2}}
    =\frac{a}{a^{2}+b^{2}}-\frac{b}{a^{2}+b^{2}}i.
  \]
  More generally, for \(w=c+di\neq0\),
  \[
    \frac{a+bi}{c+di}
    =\frac{(a+bi)(c-di)}{c^{2}+d^{2}}
    =\frac{ac+bd}{c^{2}+d^{2}}
    +\frac{bc-ad}{c^{2}+d^{2}}i.
  \]
\end{example}

\begin{proposition}
  If \(z\neq0\), then \(\abs{z^{-1}}=\abs{z}^{-1}=1/\abs{z}\).
\end{proposition}

\begin{proof}
  Since \(zz^{-1}=1\), multiplicativity of the modulus gives
  \(\abs{z}\abs{z^{-1}}=\abs{1}=1\). Divide by \(\abs{z}>0\).
\end{proof}

\subsubsection{Polar form of a complex number}

\begin{definition}
  A \emph{polar form} of \(z\in\CC\) is an expression
  \[
    z=\abs{z}(\cos\varphi+i\sin\varphi).
  \]
  For \(z\neq0\), the angle \(\varphi\) is an \emph{argument} of
  \(z\), determined modulo \(2\pi\). If \(z=a+bi\), then
  \(\cos\varphi=a/\abs{z}\) and \(\sin\varphi=b/\abs{z}\).
  Consequently, \(\tan\varphi=b/a\) when \(a\neq0\), but this
  ratio alone does not determine the quadrant. For \(z=0\), every
  angle gives a polar form; its argument is left undefined.
\end{definition}

\begin{example}
  The numbers \(z=1+i\) and \(w=\sqrt3-i\) have moduli
  \(\sqrt2\) and \(2\), respectively. Their locations in the
  complex plane give arguments \(\pi/4\) and \(-\pi/6\). Thus
  \[
    1+i=\sqrt2\left(\cos\frac\pi4+i\sin\frac\pi4\right),
    \qquad
    \sqrt3-i=2\left(\cos\!\left(-\frac\pi6\right)
    +i\sin\!\left(-\frac\pi6\right)\right).
  \]
\end{example}

\begin{proposition}
  If \(z=\abs{z}(\cos\varphi+i\sin\varphi)\), then
  \[
    \overline z=\abs{z}(\cos(-\varphi)+i\sin(-\varphi)).
  \]
  If, moreover, \(z\neq0\), then
  \[
    z^{-1}=\frac1{\abs{z}}
    (\cos(-\varphi)+i\sin(-\varphi)).
  \]
\end{proposition}

\begin{proof}
  Conjugation changes the sign of the imaginary part, so
  \[
    \overline z=\abs{z}(\cos\varphi-i\sin\varphi)
    =\abs{z}(\cos(-\varphi)+i\sin(-\varphi)),
  \]
  using that cosine is even and sine is odd. For \(z\neq0\),
  the reciprocal formula gives
  \[
    z^{-1}=\frac{\overline z}{\abs{z}^{2}}
    =\frac1{\abs{z}}
    (\cos(-\varphi)+i\sin(-\varphi)).
  \]
\end{proof}

\begin{proposition}
  Let \(z=\abs{z}(\cos\alpha+i\sin\alpha)\) and
  \(w=\abs{w}(\cos\beta+i\sin\beta)\). Then
  \[
    zw=\abs{z}\abs{w}
    (\cos(\alpha+\beta)+i\sin(\alpha+\beta)).
  \]
  If \(w\neq0\), then
  \[
    \frac zw=\frac{\abs{z}}{\abs{w}}
    (\cos(\alpha-\beta)+i\sin(\alpha-\beta)).
  \]
\end{proposition}

\begin{proof}
  Expanding the product and using the angle addition identities,
  \begin{align*}
    zw
    &=\abs{z}\abs{w}
    \bigl[(\cos\alpha\cos\beta-\sin\alpha\sin\beta)\\
      &\hspace{43mm}
    +i(\sin\alpha\cos\beta+\cos\alpha\sin\beta)\bigr]\\
    &=\abs{z}\abs{w}
    (\cos(\alpha+\beta)+i\sin(\alpha+\beta)).
  \end{align*}
  If \(w\neq0\), its reciprocal has modulus \(1/\abs{w}\)
  and angle \(-\beta\) by the preceding proposition. Applying
  the product formula to \(zw^{-1}\) gives the quotient formula.
\end{proof}
