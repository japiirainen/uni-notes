\begin{exercise}
  Osoita, että äärettömien bittijonojen joukko on ylinumeroituva.
\end{exercise}

\begin{solution}
  Olkoon \(\mathcal{B}\) äärettömien bittijonojen joukko. Tehdään
  vastaoletus, että \(\mathcal{B}\) on numeroituva. Tällöin sen alkiot
  voidaan luetella jonona \(b_1, b_2, b_3, \dots\), jossa jokainen
  joukon \(\mathcal{B}\) alkio esiintyy.

  Muodostetaan ääretön bittijono \(s\) seuraavasti: jokaisella
  \(n \geq 1\) valitaan jonon \(s\) \(n\):nneksi bitiksi \(0\), jos
  jonon \(b_n\) \(n\):s bitti on \(1\), ja \(1\), jos se on \(0\).
  Näin \(s\) eroaa jokaisesta jonosta \(b_n\) sen \(n\):nnessä
  kohdassa, joten \(s\) ei esiinny luettelossa.

  Koska luettelo sisältää vastaoletuksen mukaan kaikki joukon
  \(\mathcal{B}\) alkiot eikä \(s\) esiinny luettelossa, saadaan
  \(s \notin \mathcal{B}\). Toisaalta \(s\) on ääretön bittijono, joten
  \(s \in \mathcal{B}\). Tämä on ristiriita. Siis \(\mathcal{B}\) on
  ylinumeroituva.
\end{solution}

\begin{exercise}
  Eräässä 150 henkilön ryhmässä 45 harrastaa uintia, 40 pyöräilyä ja 50 juoksua.
  Lisäksi 32 henkilöä harrastaa juoksua mutta ei pyöräilyä, 27 henkilöä
  harrastaa juoksua ja uintia ja 10 henkilöä harrastaa kaikkia kolmea lajia.
\begin{enumerate}[label=\alph*), nosep]
  \item Kuinka moni henkilöistä harrastaa juoksua mutta ei uintia
    eikä pyöräilyä?
  \item Jos 21 henkilöä harrastaa sekä pyöräilyä että uintia, niin kuinka
    moni ei harrasta mitään kyseisistä kolmesta lajista?
\end{enumerate}
Käytä apuna Vennin kaavioita.
\end{exercise}

\begin{solution}
Olkoon \(X\) kaikkien 150 henkilön joukko. Merkitään uintia harrastavien
joukkoa \(U\), pyöräilyä harrastavien joukkoa \(P\) ja juoksua harrastavien
joukkoa \(J\), missä \(U,P,J \subseteq X\). Tehtävänannon mukaan
\[
  \begin{gathered}
    \abs{X}=150, \qquad \abs{U}=45, \qquad \abs{P}=40, \qquad \abs{J}=50, \\
    \abs{J\setminus P}=32, \qquad \abs{J\cap U}=27, \qquad
    \abs{U\cap P\cap J}=10.
  \end{gathered}
\]
\begin{enumerate}[label=\alph*), nosep]
\item Halutaan laskea \(\abs{J\setminus(U\cup P)}\).
  Koska \(J\setminus P\) ja \(J\cap P\) muodostavat joukon \(J\)
  erillisen jaon, saadaan
  \[
    \abs{J\cap P}=\abs{J}-\abs{J\setminus P}=50-32=18.
  \]
  Juoksijoista uintia tai pyöräilyä harrastaa
  \[
    \begin{aligned}
      \abs{J\cap(U\cup P)}
      &=\abs{J\cap U}+\abs{J\cap P}-\abs{J\cap U\cap P} \\
      &=27+18-10=35.
    \end{aligned}
  \]
  Kaikkia kolmea lajia harrastavat vähennetään kerran, koska heidät
  laskettiin mukaan molempiin leikkauksiin. Näin ollen
  \[
    \abs{J\setminus(U\cup P)}=50-35=15.
  \]
  Siis juoksua mutta ei uintia eikä pyöräilyä harrastaa 15 henkilöä.

  \begin{center}
    \begin{tikzpicture}[
        x=1.05cm, y=1.05cm,
        every node/.style={font=\normalsize, inner sep=2pt},
        set boundary/.style={line width=0.65pt}
      ]
      \draw[set boundary] (-1,0.55) circle[radius=1.75];
      \draw[set boundary] (1,0.55) circle[radius=1.75];
      \draw[set boundary] (0,-1.15) circle[radius=1.75];

      \node[above] at (-1.45,2.4) {\(U\): uinti};
      \node[above] at (1.45,2.4) {\(P\): pyöräily};
      \node[below] at (0,-3.05) {\(J\): juoksu};

      \node at (-1,-0.8) {\(17\)};
      \node at (1,-0.8) {\(8\)};
      \node at (0,-0.2) {\(10\)};
      \node[font=\large\bfseries] at (0,-2.05) {15};
    \end{tikzpicture}

    \small Luvut osoittavat juoksijoiden määrät neljällä erillisellä alueella.
  \end{center}
\item Oletetaan lisäksi, että \(\abs{U\cap P}=21\).
  Kohdassa (a) saatiin \(\abs{J\cap P}=18\). Sisällyttämis- ja
  poissulkemisperiaatteen nojalla vähintään yhtä näistä lajeista harrastaa
  \[
    \begin{aligned}
      \abs{U\cup P\cup J}
      &=\abs{U}+\abs{P}+\abs{J} \\
      &\quad-\abs{U\cap P}-\abs{U\cap J}-\abs{P\cap J}
      +\abs{U\cap P\cap J} \\
      &=45+40+50-21-27-18+10 \\
      &=79
    \end{aligned}
  \]
  henkilöä. Koska \(U\cup P\cup J\subseteq X\), saadaan
  \[
    \abs{X\setminus(U\cup P\cup J)}
    =\abs{X}-\abs{U\cup P\cup J}=150-79=71.
  \]
  Siis 71 henkilöä ei harrasta mitään näistä kolmesta lajista.

  \begin{center}
    \begin{tikzpicture}[
        x=1.05cm, y=1.05cm,
        every node/.style={font=\normalsize, inner sep=2pt},
        set boundary/.style={line width=0.65pt}
      ]
      \draw[line width=0.4pt] (-3.35,-3.65) rectangle (4.35,3.4);
      \node[anchor=north west, font=\small] at (-3.1,3.2)
      {\(X\): kaikki 150 henkilöä};

      \draw[set boundary] (-1,0.55) circle[radius=1.75];
      \draw[set boundary] (1,0.55) circle[radius=1.75];
      \draw[set boundary] (0,-1.15) circle[radius=1.75];
      \node[above] at (-1.45,2.4) {\(U\): uinti};
      \node[above] at (1.45,2.4) {\(P\): pyöräily};
      \node[below] at (0,-3.05) {\(J\): juoksu};

      \node at (-1.65,0.65) {\(7\)};
      \node at (1.65,0.65) {\(11\)};
      \node at (0,1.2) {\(11\)};
      \node at (-1,-0.8) {\(17\)};
      \node at (1,-0.8) {\(8\)};
      \node at (0,-0.2) {\(10\)};
      \node at (0,-2.05) {\(15\)};

      \node[font=\large\bfseries] at (3,-2.05) {71};

    \end{tikzpicture}

    \small Ympyröiden sisällä on yhteensä 79 henkilöä ja ulkopuolella 71.
  \end{center}
\end{enumerate}
\end{solution}

\begin{exercise}
Yleistä summaperiaate \(n\) joukolle ja todista se induktiolla.
\end{exercise}

\begin{solution}
\textbf{Väite.} Olkoon \(n \geq 1\) kokonaisluku ja olkoot
\(A_1, \ldots, A_n\) pareittain erillisiä äärellisiä joukkoja, eli
\(A_i \cap A_j = \emptyset\), kun \(1 \leq i,j \leq n\) ja \(i \neq j\).
Tällöin
\[
\abs{A_1 \cup \cdots \cup A_n} = \abs{A_1} + \cdots + \abs{A_n}.
\]
\textbf{Todistus.} Edetään induktiolla \(n\):n suhteen.
Kun \(n = 1\), väite on yhtälö \(\abs{A_1} = \abs{A_1}\), joten se pätee.

Oletetaan, että väite pätee jollakin \(n \geq 1\) kaikille \(n\):n
pareittain erillisen äärellisen joukon kokoelmille. Olkoot
\(A_1, \ldots, A_{n+1}\) pareittain erillisiä äärellisiä joukkoja.
Merkitään \(B = A_1 \cup \cdots \cup A_n\). Joukko \(B\) on äärellinen,
ja pareittaisen erillisyyden nojalla
\[
B \cap A_{n+1}
= \bigcup_{i=1}^{n}(A_i \cap A_{n+1})
= \emptyset.
\]
Voimme siis soveltaa kahden joukon summaperiaatetta joukkoihin
\(B\) ja \(A_{n+1}\). Sen ja induktio-oletuksen perusteella
\[
\begin{aligned}
  \abs{A_1 \cup \cdots \cup A_{n+1}}
  &= \abs{B \cup A_{n+1}} \\
  &= \abs{B} + \abs{A_{n+1}} \\
  &= \abs{A_1} + \cdots + \abs{A_n} + \abs{A_{n+1}}.
\end{aligned}
\]
Väite pätee siis myös \(n+1\) joukolle. Induktioperiaatteen nojalla
se pätee kaikilla kokonaisluvuilla \(n \geq 1\).
\end{solution}

\begin{exercise}
Tarkastellaan kompleksilukuja \(z = (2, -3)\) ja \(w = (4, 1)\).
\begin{enumerate}[label=\alph*), nosep]
\item Esitä \(z\) ja \(w\) imaginaariyksikön avulla muodossa \(x + iy\).
\item Mikä on kompleksiluvun \(z\) liittoluku?
\item Mikä on kompleksiluvun \(w\) vastaluku?
\item Laske kompleksilukujen \(z\) ja \(w\) summa ja erotus.
\item Mikä on kompleksiluvun \(z\) itseisarvo eli moduli?
\end{enumerate}
Esitä määrittämäsi kompleksiluvut piirtämällä niitä vastaavat
kompleksitason pisteet koordinaatistoon.
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item \(z = 2 - 3i\), \(w = 4 + i\).
\item Luvun \(z\) liittoluku on \(\overline{z} = \overline{2
- 3i} = 2 + 3i\).
\item Luvun \(w\) vastaluku on \(-w = -(4 + i) = -4 - i\).
\item Lukujen \(z\) ja \(w\) summa ja erotus ovat
\[
\begin{aligned}
z + w &= (2 - 3i) + (4 + i) = (2 + 4) + (-3 + 1)i = 6 - 2i, \\
z - w &= (2 - 3i) - (4 + i) = (2 - 4) + (-3 - 1)i = -2 - 4i.
\end{aligned}
\]
\item Luvun \(z\) itseisarvo eli moduli on
\[
\abs{z} = \sqrt{2^2 + (-3)^2} = \sqrt{4 + 9} = \sqrt{13}.
\]
\end{enumerate}

\begin{exercise}
Osoita, että kompleksilukujen kertolasku on \emph{vaihdannainen} eli
\(z \cdot w = w \cdot z\) kaikilla \(z, w \in \CC\).
Mikä on kompleksilukujen kertolaskun \emph{neutraalialkio} eli sellainen
kompleksiluku \(a\), jolla \(a \cdot z = z \cdot a = z\) kaikilla
\(z \in \CC\)?
\end{exercise}

\begin{solution}
Olkoot \(z = a + bi\) ja \(w = c + di\) mielivaltaisia kompleksilukuja,
missä \(a, b, c, d \in \RR\). Kertolaskun määritelmän sekä reaalilukujen
yhteen- ja kertolaskun vaihdannaisuuden perusteella
\[
\begin{aligned}
z \cdot w &= (a + bi)(c + di) \\
&= (ac - bd) + (ad + bc)i \\
&= (ca - db) + (cb + da)i \\
&= (c + di)(a + bi) = w \cdot z.
\end{aligned}
\]
Siis kompleksilukujen kertolasku on vaihdannainen.

Kertolaskun \emph{neutraalialkio} on \(1 = 1 + 0i\), eli parina
\((1, 0)\). Nimittäin jokaisella \(z = a + bi \in \CC\) pätee
\[
(1 + 0i)z = (1 \cdot a - 0 \cdot b) + (1 \cdot b + 0 \cdot a)i
= a + bi = z.
\]
Juuri todistetun vaihdannaisuuden nojalla myös \(z(1 + 0i) = z\).
\end{solution}
