\begin{exercise}
  Tarkastellaan (taas) kompleksilukuja \(z = (2, -3)\) ja \(w = (4, 1)\).
\begin{enumerate}[label=\alph*), nosep]
  \item Laske kompleksilukujen \(z\) ja \(w\) tulo \(zw\).
  \item Mikä on kompleksiluvun \(w\) käänteisluku?
  \item Laske kompleksilukujen \(z\) ja \(w\) osamäärä \(\frac{z}{w}\).
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Laskemalla saadaan
  \[
    \begin{aligned}
      zw &= (2 - 3i)(4 + i) \\
      &= 8 + 2i - 12i - 3i^2 \\
      &= 8 + 3 - 10i \\
      &= 11 - 10i.
    \end{aligned}
  \]
\item Koska \(w \neq 0\), käänteisluvun kaavasta seuraa
  \[
    \begin{aligned}
      w^{-1} &= \frac{\overline{w}}{\abs{w}^2}
      = \frac{4 - i}{4^2 + 1^2} \\
      &= \frac{4 - i}{17}
      = \frac{4}{17} - \frac{1}{17}i.
    \end{aligned}
  \]
\item Kertomalla osoittaja ja nimittäjä luvulla \(4 - i\) saadaan
  \[
    \begin{aligned}
      \frac{z}{w} &= \frac{2 - 3i}{4 + i} \\
      &= \frac{(2 - 3i)(4 - i)}{(4 + i)(4 - i)} \\
      &= \frac{8 - 2i - 12i + 3i^2}{17} \\
      &= \frac{5 - 14i}{17} \\
      &= \frac{5}{17} - \frac{14}{17}i.
    \end{aligned}
  \]
\end{enumerate}
\end{solution}

\begin{exercise}
Olkoot \(z = -3 + 3i\) ja \(w = \sqrt{3} + i\).
\begin{enumerate}[label=\alph*), nosep]
\item Määritä kompleksilukujen \(z\) ja \(w\) argumentit eli vaihekulmat.
\item Määritä kompleksiluvun \(z\) napaesitys.
\item Määritä kompleksiluvun \(w\) eksponenttiesitys.
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Luku \(z=-3+3i\) on kompleksitason toisessa neljänneksessä.
Sen viitekulma on \(\pi/4\), joten sen pääargumentti on \(3\pi/4\).
Luku \(w=\sqrt{3}+i\) on ensimmäisessä neljänneksessä, ja sen
viitekulma on \(\pi/6\). Siis pääargumentit ovat
\[
\arg z = \frac{3\pi}{4}, \qquad \arg w = \frac{\pi}{6}.
\]
Muut argumentit saadaan lisäämällä näihin kulmiin \(2k\pi\), missä
\(k \in \ZZ\).
\item Koska \(\abs{z}=\sqrt{(-3)^2+3^2}=3\sqrt{2}\), luvun \(z\)
napaesitys on
\[
z=3\sqrt{2}\left(\cos\frac{3\pi}{4}
+i\sin\frac{3\pi}{4}\right).
\]
\item Koska \(\abs{w}=\sqrt{(\sqrt{3})^2+1^2}=2\), luvun \(w\)
eksponenttiesitys on
\[
w=2e^{i\pi/6}.
\]
\end{enumerate}
\end{solution}

\begin{exercise}
Olkoot \(z,w \in \CC\), \(w \neq 0\). Osoita, että osamäärän liittoluku on
liittolukujen osamäärä eli
\[
\overline{\left(\frac{z}{w}\right)} = \frac{\overline{z}}{\overline{w}}.
\]
\end{exercise}

\begin{solution}
Olkoon \(z,w \in \CC\) ja \(w \neq 0\). Tällöin
\(\abs{w}^2=w\overline{w}>0\), joten
\(\frac{z}{w}=z\overline{w}\cdot\frac{1}{\abs{w}^2}\).
Koska \(\frac{1}{\abs{w}^2}\) on reaalinen, tulon liittoluvun kaavasta
ja yhtälöstä \(\overline{\overline{w}}=w\) seuraa
\[
\begin{aligned}
\overline{\left(\frac{z}{w}\right)}
&=\overline{\left(z\overline{w}\cdot\frac{1}{\abs{w}^2}\right)} \\
&=\overline{z}\,w\cdot\frac{1}{\abs{w}^2} \\
&=\frac{\overline{z}\,w}{w\overline{w}} \\
&=\frac{\overline{z}}{\overline{w}}.
\end{aligned}
\]
\end{solution}

\begin{exercise}
Laske de Moivren kaavan avulla \((1 + \sqrt{3}\,i)^5\).
\end{exercise}

\begin{solution}
Olkoon \(z=1+\sqrt{3}\,i\). Koska \(z\) on ensimmäisessä neljänneksessä,
sen pääargumentti on \(\arg z=\pi/3\). Lisäksi
\(\abs{z}=\sqrt{1^2+(\sqrt{3})^2}=2\). Siis luvun \(z\) napaesitys on
\[
z=2\left(\cos\frac{\pi}{3}+i\sin\frac{\pi}{3}\right).
\]
De Moivren kaavan avulla saadaan
\[
\begin{aligned}
z^5
&=\left[2\left(\cos\frac{\pi}{3}+i\sin\frac{\pi}{3}\right)\right]^5 \\
&=2^5\left(\cos\frac{5\pi}{3}+i\sin\frac{5\pi}{3}\right) \\
&=32\left(\frac{1}{2}-\frac{\sqrt{3}}{2}i\right) \\
&=16-16\sqrt{3}\,i.
\end{aligned}
\]
\end{solution}

\begin{exercise}
Määritä
\begin{enumerate}[label=\alph*), nosep]
\item kompleksiluvun \(-4\) neljännet juuret,
\item kompleksiluvun \(i\) kuutiojuuret.
\end{enumerate}
Esitä kummankin kohdan (a) ja (b) juuret kompleksitasossa.
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Kirjoitetaan ensin \(-4\) napamuodossa:
\[
-4 = 4(\cos\pi + i\sin\pi).
\]
\newpage
Halutaan ratkaista yhtälö \(w^4 = -4\). Ratkaisut saadaan kaavasta
\[
w_k = 4^{1/4}\left(\cos{\frac{\pi + 2\pi k}{4}} + i\sin{\frac{\pi +
2\pi k}{4}}\right), \quad k \in \set{0, 1, 2, 3}.
\]
Ratkaisut ovat
\[
\begin{aligned}
w_0 &= \sqrt{2}\left(\cos{\frac{\pi}{4}} +
i\sin{\frac{\pi}{4}}\right) = 1 + i, \\
w_1 &= \sqrt{2}\left(\cos{\frac{3\pi}{4}} +
i\sin{\frac{3\pi}{4}}\right) = -1 + i, \\
w_2 &= \sqrt{2}\left(\cos{\frac{5\pi}{4}} +
i\sin{\frac{5\pi}{4}}\right) = -1 - i, \\
w_3 &= \sqrt{2}\left(\cos{\frac{7\pi}{4}} +
i\sin{\frac{7\pi}{4}}\right) = 1 - i.
\end{aligned}
\]
Kompleksiluvun \(-4\) neljännet juuret ovat täten
\[
1 + i, -1 + i, -1 - i, 1 - i.
\]
\item Kirjoitetaan ensin \(i\) napamuodossa:
\[
i = \cos\frac{\pi}{2} + i\sin\frac{\pi}{2}.
\]
Halutaan ratkaista yhtälö \(w^3 = i\). Ratkaisut saadaan kaavasta
\[
w_k = \cos{\frac{\pi/2 + 2\pi k}{3}} + i\sin{\frac{\pi/2 + 2\pi
k}{3}}, \quad k \in \set{0, 1, 2}.
\]
Ratkaisut ovat
\[
\begin{aligned}
w_0 &= \cos{\frac{\pi}{6}} + i\sin{\frac{\pi}{6}} = \frac{\sqrt{3}}{2}
+ \frac{1}{2}i, \\
w_1 &= \cos{\frac{5\pi}{6}} + i\sin{\frac{5\pi}{6}} = -\frac{\sqrt{3}}{2}
+ \frac{1}{2}i, \\
w_2 &= \cos{\frac{3\pi}{2}} + i\sin{\frac{3\pi}{2}} = -i.
\end{aligned}
\]
Kompleksiluvun \(i\) kuutiojuuret ovat täten
\[
\frac{\sqrt{3}}{2} + \frac{1}{2}i, -\frac{\sqrt{3}}{2} + \frac{1}{2}i, -i.
\]
\end{enumerate}

Juuret kompleksitasossa (vaaka-akseli on reaaliosa ja pystyakseli
imaginaariosa). Katkoviiva kuvaa yksikköympyrää: kohdan (a) juurten
itseisarvo on \(\sqrt{2}\) ja kohdan (b) juurten itseisarvo on \(1\).

\begin{exercise}
Ratkaise kompleksilukujen joukossa yhtälöt
\begin{enumerate}[label=\alph*), nosep]
\item \(2x^2 - 2x + 1 = 0\),
\item \(x^4 - 2x^2 + 4 = 0\).
\end{enumerate}
Esitä kummankin kohdan (a) ja (b) ratkaisut kompleksitasossa.
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Ratkaisukaavan avulla saadaan
\[
x=\frac{2\pm\sqrt{(-2)^2-4\cdot2\cdot1}}{2\cdot2}
=\frac{2\pm2i}{4}=\frac{1\pm i}{2}.
\]
Merkitään kohdan (a) ratkaisuja \(x_{a,1}=(1+i)/2\) ja
\(x_{a,2}=(1-i)/2\).

\item Asetetaan \(y=x^2\). Tällöin yhtälö on \(y^2-2y+4=0\), joten
\[
y=\frac{2\pm\sqrt{(-2)^2-4\cdot4}}{2}
=1\pm\sqrt3\,i.
\]
Molempien \(y\):n arvojen itseisarvo on \(2\). Niiden argumentit ovat
\(\pi/3\) ja \(-\pi/3\), joten napaesitykset ovat
\[
1\pm\sqrt3\,i
=2\left(\cos\left(\pm\frac{\pi}{3}\right)
+i\sin\left(\pm\frac{\pi}{3}\right)\right).
\]
Neliöjuurten itseisarvo on siis \(\sqrt2\), ja niiden kulmat saadaan
puolittamalla argumentit sekä lisäämällä \(\pi\) toiselle juurelle.
Näin saadaan neljä ratkaisua:
\[
\begin{aligned}
x_{b,1}&=\frac{\sqrt6}{2}+\frac{\sqrt2}{2}i,
&x_{b,2}&=-\frac{\sqrt6}{2}-\frac{\sqrt2}{2}i,\\
x_{b,3}&=\frac{\sqrt6}{2}-\frac{\sqrt2}{2}i,
&x_{b,4}&=-\frac{\sqrt6}{2}+\frac{\sqrt2}{2}i.
\end{aligned}
\]
Tässä \(x_{b,1}^2=x_{b,2}^2=1+\sqrt3\,i\) ja
\(x_{b,3}^2=x_{b,4}^2=1-\sqrt3\,i\).
\end{enumerate}

Ratkaisut kompleksitasossa. Katkoviiva kuvaa yksikköympyrää;
kohdan (a) juurten itseisarvo on \(1/\sqrt2\) ja kohdan (b) \(\sqrt2\).
