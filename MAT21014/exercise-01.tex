  lauseita:

    p_0&: \text{Sataa.} \\
    p_1&: \text{Tuulee.} \\
    p_2&: \text{On Kylmä.}
  \end{align*}

  \[\left((p_0 \land \neg p_1) \rightarrow \neg p_2\right).\]
