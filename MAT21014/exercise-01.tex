\begin{exercise}
  Merkitään propositiosymboleilla \(p_0\), \(p_1\) ja \(p_2\) seuraavia
  lauseita:
  \begin{align*}
    p_0&: \text{Sataa.} \\
    p_1&: \text{Tuulee.} \\
    p_2&: \text{On Kylmä.}
  \end{align*}
  Esitä luonnollisella kielellä lause
  \[\left((p_0 \land \neg p_1) \rightarrow \neg p_2\right).\]
\end{exercise}

\begin{solution}
  ``Jos sataa ja \emph{ei} tuule, niin \emph{ei} ole kylmä.``, tai mahdollisesti
  hieman tyylikkäämmin: ``Jos sataa \emph{eikä} tuule, niin \emph{ei}
  ole kylmä.``
\end{solution}

\begin{exercise}
  Annetaan propositiosymboleille merkitykset tehtävän 1 tapaan. Käännä
  seuraavat lauseet propositiologiikan kielelle:
\begin{enumerate}[label=\alph*), nosep]
  \item Jos on kylmä, mutta ei sada, niin tuulee.
  \item Jos ei sada, niin ei ole kylmä, paitsi jos tuulee.
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item \(((p_2 \land \neg p_0) \rightarrow p_1).\)
\item \((\neg p_0 \rightarrow (\neg p_2 \lor p_1)).\)
\end{enumerate}
\end{solution}

\begin{exercise}
Tarkastellaan lausetta ``Jos Niilo syö valtavasti aina, kun Unicafessa
on linssibolognesea, niin joko Niilo on kova syömään tai linssibolognese on
hänen lempiruokaansa.``
\begin{enumerate}[label=\alph*), nosep]
\item Mitkä ovat lauseen sisältämät atomilauseet?
\item Formalisoi lause propositiolauseella. (Huomaa, että joudut korvaamaan
  luonnollisen kielen vivahteikkaampia ilmaisuja propositiologiikan
konnektiiveilla.)
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item ``Niilo syö valtavasti'', ``Unicafessa on linssibolognesea'',
``Niilo on kova syömään'', ``linssibolognese on hänen lempiruokaansa''.

\item Merkitään propositiosymboleilla \(p_0,p_1,p_2,p_3\) kohdassa (a)
löytyneet atomilauseet. Olkoon
\[
\begin{aligned}
  p_0 &= \text{``Niilo syö valtavasti''},\\
  p_1 &= \text{``Unicafessa on linssibolognesea''},\\
  p_2 &= \text{``Niilo on kova syömään''},\\
  p_3 &= \text{``linssibolognese on hänen lempiruokaansa''}.
\end{aligned}
\]
Tällöin lauseen formaali muoto on
\[
(p_1 \to p_0) \to (p_2 \lor p_3).
\]
\end{enumerate}
\end{solution}

\begin{exercise}
Keksi luonnollisen kielen lause, jonka formalisointi on muotoa
\[(((p_0 \rightarrow (p_1 \lor p_2)) \land \neg p_2) \rightarrow (p_0
\rightarrow p_1)).\]
\end{exercise}

\begin{solution}
Olkoon
\[
\begin{aligned}
p_0 &= \text{``Nalle menee ulos''}, \\
p_1 &= \text{``Nalle ottaa sateenvarjon''}, \\
p_2 &= \text{``Nalle ottaa sadetakin''}.
\end{aligned}
\]
``Rakennetaan'' pitkähkö lause \emph{sisältä ulos}:
\[
\begin{aligned}
(p_1 \lor p_2) &= \text{``Nalle ottaa sateenvarjon tai sadetakin''}, \\
(p_0 \rightarrow (p_1 \lor p_2)) &= \text{``Jos Nalle menee ulos, hän
ottaa sateenvarjon tai sadetakin''}, \\
\neg p_2 &= \dots \text{``ja Nalle ei ota sadetakkia.''}, \\
(p_0 \rightarrow p_1) &= \text{``Jos Nalle menee ulos, niin hän ottaa
sateenvarjon.''}
\end{aligned}
\]
Täten, yksi mahdollinen luonnollisen kielen lause on:

``Jos pätee, että jos Nalle menee ulos, hän ottaa sateenvarjon tai
sadetakin, ja Nalle ei ota sadetakkia, niin jos Nalle menee ulos,
hän ottaa sateenvarjon.''
\end{solution}

\begin{exercise}
Mitkä seuraavista merkkijonoista ovat propositiolauseita? Perustele.
\begin{enumerate}[label=\alph*), nosep]
\item \(((p_0 \land p_0) \land \neg p_0)\)
\item \(p_0 \rightarrow p_1 \rightarrow p_2\)
\item \(p_{32110}\)
\item \((p_0) \lor (\neg p_0)\)
\item \((\neg(\neg p_{222}))\)
\item \())p_1 \lor \rightarrow p_0\)
\item \((p_5 \rightarrow (p_0 \rightarrow \neg \neg p_0))\)
\item \((p_0 \neg \land p_2)\)
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Rakennetaan merkkijono propositiolauseen määritelmän mukaisesti. Ensiksi,
\(A = p_0\) on propositiolause, sillä se on propositiosymboli.
Propositiolauseen määritelmän mukaan, merkkijonot
\(B = (A \land A)\) ja \(C = \neg A\) ovat propositiolauseita. Jälleen,
propositiolauseen määritelmän nojalla, merkkijon \((B \land C)\) on
propositiolause. Koska
\[(B \land C) = ((p_0 \land p_0) \land \neg p_0),\]
niin \(((p_0 \land p_0) \land \neg p_0)\) on propositiolause.
\item Merkkijono \(p_0 \rightarrow p_1 \rightarrow p_2\) ei ole propositiolause.
Propositiolauseen määritelmän mukaan, jos \(A\) ja \(B\) ovat
propositiolauseita, implikaation avulla muodostettu propositiolause on muotoa
\((A \rightarrow B)\). Merkkijonossa \(p_0 \rightarrow p_1 \rightarrow p_2\)
implikaatioiden ympäriltä puuttuivat määritelmän edellyttävät sulkeet, joten
sitä ei voida muodostaa propositiolauseen määritelmän mukaisesti.
\item \(p_{32110}\) on propositiosymboli, joten propositiolauseen määritelmän
mukaan, se on propositiolause.
\item Merkkijono \((p_0) \lor (\neg p_0)\) ei ole propositiolause, sillä
propositiolauseen määritelmän mukaan disjunktio on muotoa \((A \lor B)\),
jossa \(A\) ja \(B\) ovat propositiolauseita. Annettu merkkijono ei ole
tätä muotoa. Lisäksi osamerkkijonoa \((p_0)\) ei voida muodostaa
propositiolauseen mukaisesti.
\item Merkkijono \((\neg(\neg p_{222}))\) ei ole propositiolause.
Propositiolauseen määritelmän mukaan sulkeita käytetään\emph{vain}
binäärisillä konnektiiveilla muodostetuissa lauseissa, jotka ovat muotoa
\[(A \land B), (A \lor B), (A \rightarrow B), (A \leftrightarrow B),\]
missä \(A\) ja \(B\) on propositiolauseita. Merkkijono \((\neg(\neg p_{222}))\)
ei ole mitään näistä muodoista, joten se ei ole propositiolause.
\item \())p_1\land \rightarrow p_0\) ei ole propositiolause, sillä
propositiolauseen määritelmän mukaan mikään propositiolause ei ola
oikealla sululla \()\).
\item Olkoon \(A = p_0\), \(B = \neg A\), \(C = \neg B\). Propositiolauseen
määritelmän mukaan \(A,B,C\) ovat propositiolauseita. Määritelmän mukaan myös
\(D = (A \rightarrow C)\) on propositiolause. Nyt \(E = p_5\) on
propositiolause, sillä \(p_5\) on propositiosymboli, joten myös
\[(E \rightarrow D) = (p_5 \rightarrow (p_0 \rightarrow \neg \neg p_0))\]
on propositiolause.
\item \((p_0 \neg \land p_2)\) ei ole propositiolause. Propositiolauseen
määritelmässä negaatio esiintyy aina muodossa \(\neg A\), missä \(A\)
on propositiolause. Propositiolause ei voi alkaa konnektiivilla \(\land\),
joten merkkijono \(\neg \land\) ei voi esiintyä propositiolauseen
alimerkkijonona. Koska \((p_0 \neg \land p_2)\) sisältää tämän
alimerkkijonon, se ei ole propositiolause.
\end{enumerate}
\end{solution}

\begin{exercise}
Mitkä ovat seuraavien propositiolauseiden pääkonnektiivit ja välittömät
alilauseet?
\begin{enumerate}[label=\alph*), nosep]
\item \((\neg p_0 \rightarrow p_1)\)
\item \((\neg \neg (p_0 \rightarrow p_1) \rightarrow (p_1 \land (p_0
\lor \neg p_1)))\)
\item \(((\neg(p_0 \rightarrow p_1) \rightarrow p_1) \land (p_2 \lor
(p_1 \rightarrow p_2)))\)
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Pääkonnektiivi on \(\rightarrow\). Välittömät alilauseet ovat täten
\(\neg p_0\) ja \(p_1\).
\item Pääkonnektiivi on \(\rightarrow\). Välittömät alilauseet ovat täten
\(\neg \neg(p_0 \rightarrow p_1)\) ja \((p_1 \land (p_0 \lor \neg p_1))\).
\item Pääkonnektiivi on \(\land\). Välittömät alilauseet ovat täten
\((\neg(p_0 \rightarrow p_1) \rightarrow p_1)\) ja
\((p_2 \lor (p_1 \rightarrow p_2))\).
\end{enumerate}
\end{solution}

\begin{exercise}
Mitkä ovat lauseen
\((((p_0 \land \neg p_1) \rightarrow p_2) \leftrightarrow \neg(p_1
\rightarrow (p_2 \land p_3)))\)
alilauseet?
\end{exercise}

\begin{solution}
Alilauseet ovat \(p_0\), \(p_1\), \(p_2\), \(p_3\), \(\neg p_1\),
\((p_0 \land \neg p_1)\), \((p_2 \land p_3)\), \((p_1 \rightarrow
(p_2 \land p_3))\), \(\neg(p_1 \rightarrow (p_2 \land p_3))\),
sekä \((((p_0 \land \neg p_1) \rightarrow p_2) \leftrightarrow \neg(p_1
\rightarrow (p_2 \land p_3)))\).
\end{solution}

\begin{exercise}
Onko olemassa propositiolausetta, jolla on tasan
\begin{enumerate}[label=\alph*), nosep]
\item 0
\item 1
\item 2
\item 3
\end{enumerate}
alilausetta? Anna esimerkki, tai perustele, miksei sellaista ole.
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item ``Pienin'' mahdollinen propositiolause koostuu yhdestä
propositiosymbolista, jolla on yksi alilause, kyseinen propositiosymboli
itse. Ei siis ole olemassa propositiolausetta, jolla on \(0\) alilausetta.
\item Propositiolauseella \(p_n\), jossa \(n \in \NN\) on yksi alilausetta,
nimittäin \(p_n\).
\item Olkoon \(p_n\), missä \(n \in \NN\) propositiosymboli. Propositiolauseella
\(\neg p_n\) on \(2\) alilausetta, nimittäin \(p_n\) ja \(\neg p_n\).
\item Olkoon \(p_0\) ja \(p_1\) propositiosymboleita. Propositiolauseessa
\((p_0 \rightarrow p_1)\) on \(3\) alilausetta, nimittäin
\(p_0\), \(p_1\), ja \((p_0 \rightarrow p_1)\).
\end{enumerate}
\end{solution}

\begin{exercise}
Onko olemassa propositiolausetta, jolla on tasan
\begin{enumerate}[label=\alph*), nosep]
\item 0
\item 1
\item 2
\item 3
\end{enumerate}
välitöntä alilausetta? Anna esimerkki, tai perustele, miksei sellaista ole.
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
\item Propositiosymbolilla \(p_n\), missä \(n \in \mathbb{N}\), ei ole
yhtään välitöntä alilausetta. Siis on olemassa propositiolause, jolla on
täsmälleen \(0\) välitöntä alilausetta.

\item Olkoon \(A\) propositiolause. Tällöin propositiolauseella
\[
\neg A
\]
on täsmälleen yksi välitön alilause, nimittäin \(A\).

\item Esimerkiksi propositiolauseella
\[
(p_0 \to p_1)
\]
on täsmälleen kaksi välitöntä alilausetta, nimittäin \(p_0\) ja \(p_1\).

\item Ei ole olemassa propositiolausetta, jolla olisi täsmälleen kolme
välitöntä alilausetta. Propositiolause on määritelmän mukaan joko
propositiosymboli, muotoa \(\neg A\) oleva propositiolause tai muotoa
\[
(A \mathbin{\circ} B)
\]
oleva propositiolause, missä \(\circ\) on binäärikonnektiivi. Näillä on
vastaavasti \(0\), \(1\) tai \(2\) välitöntä alilausetta. Siten
propositiolauseella ei voi olla täsmälleen \(3\) välitöntä alilausetta.
\end{enumerate}
\end{solution}

\begin{exercise}
Todista induktiolla, että propositiolauseella, jossa on \(n\) konnektiivia,
on korkeintaan \(2n + 1\) alilausetta.
\end{exercise}

\begin{solution}
Todistetaan väite induktiolla rakenteen suhteen. Merkitään \(\sigma(A)\):lla
\(A\):n alilauseiden lukumäärää ja \(\lambda(A)\):lla \(A\):ssa esiintyvien
konnektiivien lukumäärää. Jos \(A\) on propositiosymboli, niin
\(\lambda(A) = 0\). Tässä tapauksessa, \(\sigma(A) = 1\), joten
\(\sigma(A) = 1 \leq 2\lambda(A) + 1 = 2\cdot 0 + 1\). Induktioaskel:
\begin{enumerate}[label=\alph*), nosep]
\item Jos \(A = \neg B\) ja
\[\sigma(B) \leq 2\lambda(B) + 1,\]
niin
\[
\begin{aligned}
\sigma(A) &= \sigma(B) + 1 \\
&\leq 2\lambda(B) + 2 \\
&= 2(\lambda(A) - 1) + 2 \\
&= 2\lambda(A) \\
&\leq 2\lambda(A) + 1.
\end{aligned}
\]
\item Olkoon \(\circ \in \left\{\land, \lor, \rightarrow,
\leftrightarrow \right\}\).
Jos \(C = (A \circ B)\) ja
\[\sigma(A) \leq 2\lambda(A) + 1,\]
sekä
\[\sigma(B) \leq 2\lambda(B) + 1.\]
Tällöin
\[
\begin{aligned}
\sigma(C) &\leq \sigma(A) + \sigma(B) + 1 \\
&\leq (2\lambda(A) + 1) + (2\lambda(B) + 1) + 1 \\
&= 2\lambda(A) + 2\lambda(B) + 3 \\
&= 2(\lambda(A) + \lambda(B)) + 3 \\
&= 2\lambda(C) + 1.
\end{aligned}
\]
sillä \(\lambda(C) = \lambda(A) + \lambda(B) + 1\).
\end{enumerate}
Näin väite pätee propositiosymboleille sekä säilyy muodostaessa
uusia propositiolauseita negaation ja binäärikonnektiivien avulla.
Rakenteellisen induktioperiaatteen nojalla
\[\sigma(A) \leq 2\lambda(A) + 1\]
pätee kaikille propositiolauseilla \(A\).
\end{solution}

\begin{lemma}\label{lem:subformula}
Jokainen propositiolauseessa \(A\) esiintyvä propositiosymboli,
negaatiomerkki \(\neg\) tai vasen sulkumerkki \((\) on jonkin
\(A\):n alilauseen ensimmäinen merkki.
\end{lemma}

\begin{proof}
Todistetaan väite induktiolla rakenteen suhteen. Jos \(A\) on propositiosymboli,
\(A = p_n\), niin ainoa väitteessä tarkasteltava merkki on \(p_n\), joka
on \(A\):n ensimmäinen merkki. Koska \(A\) on itsensä alilause, väite pätee.

Oletetaan sitten, että väite pätee propositiolauseella \(B\).
Jos \(A = \neg B\), niin ensimmäinen merkki \(\neg\) aloittaa alilauseen \(A\).
Lisäksi jokainen \(B\):ssä esiintyvä propositiosymboli, negaatiomerkki tai
vasen sulkumerkki aloittaa induktio-oletuksen nojalla jonkin \(B\):n alilauseen.
Jokainen \(B\):n alilause on \(A\):n alilause, joten väite pätee \(A\):lle.

Oletetaan lopuksi, että väite pätee propositiolauseilla \(B\) ja \(C\).
Jos
\[
A = (B \circ C), \qquad
\circ \in \{\land, \lor, \rightarrow, \leftrightarrow\},
\]
niin ensimmäinen merkki \((\) aloittaa alilauseen \(A\). Muut väitteessä
tarkasteltavat merkit esiintyvät joko \(B\):ssä tai \(C\):ssä.
Induktio-oletuksen nojalla ne aloittavat jonkin \(B\):n tai \(C\):n alilauseen
ja siten jonkin \(A\):n alilauseen.

Rakenteellisen induktioperiaatteen nojalla väite pätee kaikilla
propositiolauseilla \(A\).
\end{proof}

\begin{lemma}\label{lem:start-seg}
Jos \(B\) ja \(C\) ovat propositiolauseita, ja \(B\) on \(C\):n alusegmentti,
niin \(B = C\).
\end{lemma}

\begin{proof}
Olkoon \(B\) ja \(C\) propositiolauseita, ja \(B\) \(C\):n alkusegmentti,
eli \(C = BZ\), jollain merkkijonolla \(Z\). Merkitään tyhjää
merkkijonoa symbolilla \(\epsilon\). On näytettävä \(Z = \epsilon\).
Edetään induktiolla rakenteen suhteen. Ensiksi, jos \(C = p_n\) niin
\(C\):n ainoa merkki on \(p_n\), joten \(C = B = p_n\), eli \(Z = \epsilon\).

Oletetaan sitten, että väite pätee propositiolauseella \(A\). Jos
\(C = \neg A\), niin \(B\) on oltava muotoa \(B = \neg B^{'}\).
Koska \(B\) on \(C\):n alkusegmentti, \(B^{'}\) on \(A\):n alkusegmentti.
Induktio-oletuksen nojalla \(A = B^{'}\), joten \(B = C\).

Oletetaan lopuksi, että väite pätee propositiolauseilla \(D\) ja \(E\). Olkoon
\[
C = (D \circ E), \qquad
\circ \in \{\land, \lor, \rightarrow, \leftrightarrow\}.
\]
Määritellään seuraavat funktiot merkkijonolle \(Y\)
\[
\begin{aligned}
O(Y) &= \text{``\(Y\) \textbf{oikeiden} sulkumerkkien lukumäärä''}, \\
V(Y) &= \text{``\(Y\) \textbf{vasenten} sulkumerkkien lukumäärä''}.
\end{aligned}
\]
\begin{lemma}\label{lem:helper}
Jos \(Y\) on propositiolauseen \(A\) alkusegmentti, niin
\(V(Y) \geq O(Y)\).
\end{lemma}
\begin{proof}
Olkoon \(Y\) propositiolauseen \(A\) alkusegmentti. Jokainen \(Y\):ssä
esiintyvä oikea sulkumerkki \()\) päättää jonkun binäärisen alilauseen
\[B \circ C,\]
joten sitä vastaava vasen sulkumerkki \((\) esiintyy ennen sitä ja kuuluu
siten myös \(Y\):hyn. Eri oikeita sulkumerkkejä vastaavat eri vasemmat
sulkumerkit. Näin ollen \(Y\):ssä on vähintään yhtä monta vasenta
sulkumerkkiä kuin oikeaa sulkumerkkiä, eli
\[V(Y) \geq  O(Y).\]
\end{proof}
Oletetaan vastoin väitettä, että \(B \neq C\). Tällöin \(B\) on
\(C\):n aito alkusegmentti. Koska \(C=(D\circ E)\), \(B\) sisältää
\(C\):n ensimmäisen vasemman sulkumerkin, mutta ei sitä vastaavaa
viimeistä oikeaa sulkumerkkiä. Lemman (\ref{lem:helper}) nojalla muut \(B\):ssä
esiintyvät sulkumerkit eivät voi tuottaa oikeiden sulkumerkkien
ylijäämää. Siis
\[
V(B)>O(B).
\]
Toisaalta \(B\) on propositiolause, joten
\[
V(B)=O(B),
\]
mikä on ristiriita. Siis \(B=C\).

Väite seuraa rakenteellisesta induktioperiaatteesta.
\end{proof}

\begin{exercise}[Haastetehtävä]
Osoita, että kaikilla propositiolauseilla \(A\) pätee:

Propositiolauseen \(A\) alimerkkijono on \(A\):n alilause, jos ja vain jos
se on itsessään propositiolause.
\end{exercise}

\begin{solution}
Olkoon \(X\) propositiolauseen \(A\) alimerkkijono.

\((\Longrightarrow)\) Olkoon \(X\) on \(A\):n alilause. Halutaan osoittaa,
että \(X\) on itsessään propositiolause. Alilauseen (rekursiivisen) määritelmän
mukaan, jokainen \(A\):n alilause on itsessään propositiolause, joten
\(X\) on propositiolause.

\((\Longleftarrow)\) Olkoon \(X\) on propositiolause. Halutaan osoittaa,
että \(X\) on \(A\):n alilause. Tiedetään Lemman (\ref{lem:subformula}) nojalla,
että kohdassa, josta \(X\) alkaa merkkijonossa \(A\), alkaa jokin \(A\):n
alilause \(B\). Koska \(X\) ja \(B\) alkavat samasta kohdasta,
jompikumpi on oltava toisen alkusegmentti. Koska sekä \(X\) että \(B\) ovat
propositiolauseita, Lemman (\ref{lem:start-seg}) nojalla, \(X = B\).
Siis \(X\) on \(A\):n alilause.
\end{solution}
