\begin{exercise}
  Olkoot \(a\), \(b\), \(c\) ja \(d\) kokonaislukuja sekä \(n\)
  positiivinen kokonaisluku. Todista seuraavat väitteet.
  \begin{enumerate}[label=(\alph*), nosep]
    \item Jos \(a \mid b\) ja \(a \mid c\), niin \(a \mid xb + yc\)
      aina, kun \(x\) ja \(y\) ovat kokonaislukuja.
    \item Jos \(a \mid b\) ja \(b \mid c\), niin \(a \mid c\).
    \item Jos \(a \equiv b \pmod{n}\) ja \(b \equiv c \pmod{n}\),
      niin \(a \equiv c \pmod{n}\).
  \end{enumerate}
\end{exercise}

\begin{solution}
  Olkoot \(a, b, c \in \ZZ\) ja \(n \in \ZZ_+\).
  \begin{enumerate}[label=(\alph*), nosep]
    \item Oletetaan, että \(a \mid b\) ja \(a \mid c\). Tällöin on olemassa
      \(k, \ell \in \ZZ\) siten, että
      \[
        b = ka, \quad c = \ell a.
      \]
      Tällöin
      \[
        b + c = ka + \ell a = a(k + \ell),
      \]
      eli \(a \mid b + c\). Olkoon \(x, y \in \ZZ\). Tällöin
      \[
        xb + yc = a(xk + y\ell),
      \]
      eli \(a \mid xb + yc\).
    \item Oletetaan, että \(a \mid b\) ja \(b \mid c\). Tällöin on olemassa
      \(k, \ell \in \ZZ\) siten, että
      \[
        b = ka, \quad c = \ell b.
      \]
      Tällöin
      \[
        c = \ell b = \ell(ka) = a(\ell k),
      \]
      eli \(a \mid c\).
    \item Oletetaan, että \(a \equiv b \pmod{n}\) ja \(b \equiv c \pmod{n}\).
      Kongruenssin määritelmän mukaan
      \[
        n \mid a - b, \quad n \mid b - c.
      \]
      Täten on olemassa \(k, \ell \in \ZZ\) siten, että
      \[
        a - b = kn, \quad b - c = \ell n.
      \]
      Tällöin
      \[
        (a - b) + (b - c) = a - c = n(k + \ell),
      \]
      eli \(n \mid a - c\). Siis
      \[
        a \equiv c \pmod{n}.
      \]
  \end{enumerate}
\end{solution}

\begin{exercise}
  Todista, että luku \(20 \cdot 14^k + 6\) on jaollinen luvulla \(13\)
  kaikilla \(k \in \ZZ_+\)
  \begin{enumerate}[label=(\alph*), nosep]
    \item induktiolla,
    \item kongruenssin avulla.
  \end{enumerate}
\end{exercise}

\begin{solution}
  \leavevmode\par
  \begin{enumerate}[label=(\alph*), nosep]
    \item Laskemalla saadaan
      \[
        20\cdot 14^1 + 6 = 286 = 22\cdot 13,
      \]
      eli väite pätee, kun \(k = 1\). Olkoon \(n \in \ZZ_+\) ja oletetaan, että
      \[
        20 \cdot 14^n + 6 = 13m
      \]
      jollain \(m \in \ZZ\). Tällöin
      \[
        \begin{aligned}
          20 \cdot 14^{n + 1} + 6 &= 14\cdot 20 \cdot 14^n + 6 \\
          &= 14(20\cdot 14^n + 6) - 78 \\
          &= 14(13m) - 13\cdot 6 \\
          &= 13(14m - 6).
        \end{aligned}
      \]
      Koska \(14m - 6 \in \ZZ\), saadaan \(13 \mid 20 \cdot 14^{n + 1} + 6\).
      Induktioperiaatteen nojalla väite pätee kaikilla \(k \in \ZZ_+\).
    \item Ensiksi,
      \[
        14^1 \equiv 1 \pmod{13}.
      \]
      Olkoon \(n \in \ZZ_+\) ja oletetaan
      \[
        14^n \equiv 1 \pmod{13}.
      \]
      Tällöin
      \[
        14^{n + 1} = 14\cdot 14^n \equiv 14 \equiv 1 \pmod{13}.
      \]
      Täten
      \[
        14^n \equiv 1 \pmod{13}
      \]
      kaikilla \(n \in \ZZ_+\). Täten
      \[
        20 \cdot 14^k + 6 \equiv 26 \equiv 0 \pmod{13},
      \]
      eli \(13 \mid 20 \cdot 14^k + 6\) kaikilla \(k \in \ZZ_+\).
  \end{enumerate}
\end{solution}

\begin{exercise}
  Olkoon \(k\) positiivinen kokonaisluku. Osoita, että on olemassa \(k\)
  peräkkäistä positiivista kokonaislukua, joista yksikään ei ole alkuluku.
  (Vinkki: Luvusta \((k + 1)! + 2\) on hyötyä. Muista, että
  \(n! = 1 \cdot 2 \cdot 3 \cdot \ldots \cdot (n - 1) \cdot n\).)
\end{exercise}

\begin{solution}
  Olkoon \(k \in \ZZ_+\) ja \(f \colon \NN \to \NN\),
  \[
    f(n) = (k + 1)! + 2 + n.
  \]
  Kun \(0 \leq n < k\), pätee \(2 \leq n + 2 \leq k + 1\), joten
  \(n + 2 \mid (k + 1)!\). Näin ollen
  \[
    n + 2 \mid f(n).
  \]
  Koska \(1 < n + 2 < f(n)\), luku \(f(n)\) ei ole alkuluku.
  Luvut
  \[
    f(0), f(1), \ldots, f(k-1)
  \]
  ovat siis \(k\) peräkkäistä positiivista kokonaislukua, joista
  yksikään ei ole alkuluku.
\end{solution}

\begin{exercise}
  Mikä on luvun \(7\) potenssi luvun \(100!\) alkutekijähajotelmassa?
\end{exercise}

\begin{solution}
  Tulon \(100! = 1 \cdot 2 \cdots 100\) tekijöistä
  \(\lfloor 100/7 \rfloor = 14\) on jaollisia luvulla \(7\).
  Kukin näistä tuo alkutekijähajotelmaan yhden tekijän \(7\).
  Lisäksi \(\lfloor 100/49 \rfloor = 2\) tekijää, luvut \(49\) ja \(98\),
  ovat jaollisia luvulla \(7^2 = 49\), joten kumpikin tuo vielä yhden
  tekijän \(7\). Koska \(7^3 = 343 > 100\), muita tekijöitä \(7\)
  ei tule. Näin ollen luvun \(7\) eksponentti luvun \(100!\)
  alkutekijähajotelmassa on \(14 + 2 = 16\), eli kysytty potenssi on
  \(7^{16}\).
\end{solution}

\begin{exercise}
  \leavevmode\par
  \begin{enumerate}[label=(\alph*), nosep]
    \item Olkoot \(a, b \in \ZZ\), \(d, n \in \ZZ_+\) ja \(d \mid n\).
      Oletetaan lisäksi, että \(ad \equiv bd \pmod{n}\). Osoita, että
      \(a \equiv b \pmod{\frac{n}{d}}\).
    \item Pääteekö aina, että jos \(ad \equiv bd \pmod{n}\), niin
      \(a \equiv b \pmod{n}\)?
  \end{enumerate}
\end{exercise}

\begin{exercise}
  Olkoon luvun \(n\) kymmenjärjestelmäesitys
  \[
    n = a_k 10^k + a_{k-1} 10^{k-1} + \cdots a_1 10 + a_0,
  \]
  missä \(0 \leq a_i \leq 9\) kaikilla \(i = 0, 1, \ldots, k\).

  Todista seitsemän jaollisuussääntö:

  Luku \(n\) on jaollinen luvulla \(7\) täsmälleen silloin, kun luku
  \[
    a_k 10^{k-1} + a_{k-1} 10^{k-2} + \cdots a_2 10 + a_1 - 2a_0
  \]
  on jaollinen luvulla \(7\).
\end{exercise}
