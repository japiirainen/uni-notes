\begin{exercise}
  Tutki, onko yhtälöryhmällä
  \[
    \text{(a)}\;
    \begin{linearsystem}{2}
      2x_1 &{}-{}& x_2 &= 3 \\
      -4x_1 &{}+{}& 2x_2 &= -6
    \end{linearsystem}
    \quad
    \text{(b)}\;
    \begin{linearsystem}{2}
      -x_1 &{}+{}& 3x_2 &= -2 \\
      2x_1 &{}+{}& x_2 &= -3
    \end{linearsystem}
    \quad
    \text{(c)}\;
    \begin{linearsystem}{2}
      -2x_1 &{}+{}& 6x_2 &= -1 \\
      x_1 &{}-{}& 3x_2 &= 1
    \end{linearsystem}
  \]
  tasan yksi, ei yhtään vai ääretön määrä ratkaisuja. Tulkitse jokaisen
  yhtälöparin ratkaiseminen graafisesti \(x_1x_2\)-tasossa.
\end{exercise}

\begin{solution}
  \leavevmode\par
\begin{enumerate}[label=\alph*), nosep]
  \item Kerrotaan toinen yhtälö luvulla \(-\frac12\):
    \[
      \begin{aligned}
        &
        \begin{linearsystem}{2}
          2x_1-x_2 &= 3\\
          -4x_1+2x_2 &= -6 &\quad& \rowop{\cdot(-1/2)}
        \end{linearsystem}\\[1ex]
        \sim{}&
        \begin{linearsystem}{2}
          2x_1-x_2 &= 3\\
          2x_1-x_2 &= 3
        \end{linearsystem}
      \end{aligned}
    \]
    Molemmat yhtälöt kuvaavat siis samaa suoraa \(x_2 = 2x_1 - 3\).
    Suorat ovat päällekkäiset, ja jokainen tämän suoran piste on
    yhtälöryhmän ratkaisu. Ratkaisuja on siten äärettömän monta.

    Valitaan \(x_1=t\in\RR\). Tällöin \(x_2=2t-3\), joten
    \[
      (x_1,x_2)
      =
      (t,2t-3)
      =
      (0,-3)
      + t
      (1,2).
    \]
    Ratkaisujoukko on siis
    \[
      \left\{
        (0,-3)
        + t
        (1,2)
        \colon\ t \in \RR
      \right\}.
    \]
    \begin{center}
      \begin{tikzpicture}
        \begin{axis}[
            width=7cm, height=6cm,
            axis lines=middle, axis equal image,
            axis line style={thin, gray},
            xmin=-1, xmax=4, ymin=-4, ymax=4,
            xlabel={$x_1$}, ylabel={$x_2$},
            xtick={1.5}, xticklabels={$\frac32$}, ytick={-3},
            tick label style={font=\small},
            domain=-0.5:3.5, samples=2,
            clip=false,
          ]
          \addplot[black, semithick] {2*x-3};
          \node[anchor=west, font=\small] at (axis cs:2.7,2.4)
          {$x_2=2x_1-3$};
        \end{axis}
      \end{tikzpicture}
    \end{center}
  \item Tehdään seuraavat rivioperaatiot:
    \[
      \begin{aligned}
        &
        \begin{linearsystem}{2}
          -x_1+3x_2 &= -2\\
          2x_1+x_2 &= -3 &\quad& \rowop{+2R_1}
        \end{linearsystem}\\[1ex]
        \sim{}&
        \begin{linearsystem}{2}
          -x_1+3x_2 &= -2\\
          7x_2 &= -7 &\quad& \rowop{\cdot(1/7)}
        \end{linearsystem}\\[1ex]
        \sim{}&
        \begin{linearsystem}{2}
          -x_1+3x_2 &= -2 &\quad& \rowop{-3R_2}\\
          x_2 &= -1
        \end{linearsystem}\\[1ex]
        \sim{}&
        \begin{linearsystem}{2}
          -x_1 &= 1 &\quad& \rowop{\cdot(-1)}\\
          x_2 &= -1
        \end{linearsystem}\\[1ex]
        \sim{}&
        \begin{linearsystem}{2}
          x_1 &= -1\\
          x_2 &= -1
        \end{linearsystem}
      \end{aligned}
    \]
    Yhtälöryhmällä on siten täsmälleen yksi ratkaisu, ja ratkaisujoukko on
    \[
      \left\{
      (-1,-1)\right\}.
    \]
    \begin{samepage}
      Graafisesti yhtälöt kuvaavat suoria
      \[
        x_2=\frac13x_1-\frac23
        \qquad\text{ja}\qquad
        x_2=-2x_1-3.
      \]
      Suorien kulmakertoimet ovat erisuuret, joten ne leikkaavat toisensa
      täsmälleen yhdessä pisteessä \((-1,-1)\).
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[
              width=8cm, height=6cm,
              axis lines=middle, axis equal image,
              axis line style={thin, gray},
              xmin=-3, xmax=2, ymin=-4, ymax=2,
              xlabel={$x_1$}, ylabel={$x_2$},
              xtick=\empty, ytick=\empty,
              samples=2, clip=false,
            ]
            \addplot[black, semithick, domain=-3:2] {x/3-2/3};
            \addplot[black, semithick, domain=-2.5:0.5] {-2*x-3};
            \addplot[only marks, mark=*, mark size=1.5pt, black]
            coordinates {(-1,-1)};
            \node[anchor=south east, font=\small] at (axis cs:-1,-1)
            {$(-1,-1)$};
            \node[anchor=north west, font=\small] at (axis cs:1,-0.3333)
            {$x_2=\frac13x_1-\frac23$};
            \node[anchor=east, font=\small] at (axis cs:-2.2,1.4)
            {$x_2=-2x_1-3$};
          \end{axis}
        \end{tikzpicture}
      \end{center}
    \end{samepage}
  \item Lisätään ensimmäiseen yhtälöön kaksi kertaa toinen:
    \[
      \begin{aligned}
        &
        \begin{linearsystem}{2}
          -2x_1+6x_2 &= -1 &\quad& \rowop{+2R_2}\\
          x_1-3x_2 &= 1
        \end{linearsystem}\\[1ex]
        \sim{}&
        \begin{linearsystem}{2}
          0 &= 1\\
          x_1-3x_2 &= 1
        \end{linearsystem}
      \end{aligned}
    \]
    Yhtälö \(0=1\) on ristiriita. Yhtälöryhmällä ei siis ole ratkaisuja,
    ja ratkaisujoukko on \(\varnothing\).

    \begin{samepage}
      Graafisesti yhtälöt kuvaavat suoria
      \[
        x_2=\frac13x_1-\frac16
        \qquad\text{ja}\qquad
        x_2=\frac13x_1-\frac13.
      \]
      Suorilla on sama kulmakerroin mutta eri vakiotermit. Ne ovat siis
      erilliset yhdensuuntaiset suorat, eikä niillä ole yhteisiä pisteitä.
      \begin{center}
        \begin{tikzpicture}
          \begin{axis}[
              width=8cm, height=4.5cm,
              axis lines=middle, axis equal image,
              axis line style={thin, gray},
              xmin=-2, xmax=2, ymin=-1.2, ymax=1,
              xlabel={$x_1$}, ylabel={$x_2$},
              xtick=\empty, ytick=\empty,
              domain=-2:2, samples=2, clip=false,
            ]
            \addplot[black, semithick] {x/3-1/6};
            \addplot[black, semithick] {x/3-1/3};
            \node[anchor=west, font=\small] at (axis cs:2,0.5)
            {$x_2=\frac13x_1-\frac16$};
            \node[anchor=north west, font=\small] at (axis cs:-2,-1)
            {$x_2=\frac13x_1-\frac13$};
          \end{axis}
        \end{tikzpicture}
      \end{center}
    \end{samepage}
\end{enumerate}
\end{solution}

\begin{exercise}
Kemiallisia reaktioita kuvaavat yhtälöt pitää tasapainottaa niin, että
niissä on kummallakin puolella sama määrä kunkin alkuaineen atomeja.
Määritä seuraavien reaktioyhtälöiden kertoimet \(x_1\), \(x_2\), \(x_3\)
ja \(x_4\), joiksi etsitään mahdollisimman pienet
positiiviset kokonaisluvut:
\begin{enumerate}[label=(\alph*)]
  \item \(x_1\mathrm{NO}_2 + x_2\mathrm{H}_2\mathrm{O}
    \to x_3\mathrm{HNO}_2 + x_4\mathrm{HNO}_3\),
  \item \(x_1\mathrm{C}_6\mathrm{H}_6 + x_2\mathrm{O}_2
    \to x_3\mathrm{C} + x_4\mathrm{H}_2\mathrm{O}\).
\end{enumerate}
\end{exercise}

\begin{solution}
\leavevmode\par
\begin{enumerate}[label=\alph*)]
\item Kunkin alkuaineen atomien lukumäärän tulee olla sama reaktion
  molemmilla puolilla. Typen, hapen ja vedyn atomien lukumäärien
  säilymisestä saadaan tässä järjestyksessä yhtälöryhmä, jota
  muokataan rivioperaatioilla:
  \[
    \begin{aligned}
      &
      \begin{linearsystem}{2}
        x_1-x_3-x_4 &= 0\\
        2x_1+x_2-2x_3-3x_4 &= 0 &\quad& \rowop{-2R_1}\\
        2x_2-x_3-x_4 &= 0
      \end{linearsystem}\\[1ex]
      \sim{}&
      \begin{linearsystem}{2}
        x_1-x_3-x_4 &= 0\\
        x_2-x_4 &= 0\\
        2x_2-x_3-x_4 &= 0 &\quad& \rowop{-2R_2}
      \end{linearsystem}\\[1ex]
      \sim{}&
      \begin{linearsystem}{2}
        x_1-x_3-x_4 &= 0 &\quad& \rowop{-R_3}\\
        x_2-x_4 &= 0 &\quad& \rowop{-R_3}\\
        -x_3+x_4 &= 0
      \end{linearsystem}\\[1ex]
      \sim{}&
      \begin{linearsystem}{2}
        x_1-2x_3 &= 0\\
        x_2-x_3 &= 0\\
        -x_3+x_4 &= 0
      \end{linearsystem}
    \end{aligned}
  \]
  Valitaan \(x_3=t\). Tällöin kaikki ratkaisut ovat muotoa
  \[
    (x_1,x_2,x_3,x_4)=(2t,t,t,t), \qquad t\in\RR.
  \]
  Kertoimet ovat positiivisia kokonaislukuja täsmälleen silloin, kun
  \(t\in\ZZ_+\). Pienin mahdollinen arvo on \(t=1\), joten pienimmät
  positiiviset kokonaislukukertoimet ovat
  \[
    (x_1,x_2,x_3,x_4)=(2,1,1,1).
  \]
  Tasapainotettu reaktioyhtälö on siis
  \[
    2\mathrm{NO}_2+\mathrm{H}_2\mathrm{O}
    \longrightarrow \mathrm{HNO}_2+\mathrm{HNO}_3.
  \]
\item Esitetään atomien lukumäärät vektoreina järjestyksessä hiili,
  vety ja happi. Atomien lukumäärien säilyminen antaa vektoriyhtälön
  \[
    x_1
    (6,6,0)
    +x_2
    (0,0,2)
    =x_3
    (1,0,0)
    +x_4
    (0,2,1).
  \]
  Tämä vastaa yhtälöryhmää
  \[
    \begin{linearsystem}{1}
      6x_1-x_3 &= 0 \\
      6x_1-2x_4 &= 0 \\
      2x_2-x_4 &= 0.
    \end{linearsystem}
  \]
  Ensimmäisestä yhtälöstä saadaan \(x_3=6x_1\), toisesta
  \(x_4=3x_1\) ja kolmannesta \(x_4=2x_2\).
  Siis \(3x_1=2x_2\), joten
  \[
    x_1=\frac23x_2, \qquad
    x_3=6\cdot\frac23x_2=4x_2, \qquad
    x_4=2x_2.
  \]
  Valitaan \(x_2=t\). Tällöin kaikki ratkaisut ovat muotoa
  \[
    (x_1,x_2,x_3,x_4)=\left(\frac23t,t,4t,2t\right),
    \qquad t\in\RR.
  \]
  Koska \(t=x_2\) on positiivinen kokonaisluku ja myös \(\frac23t\)
  on oltava kokonaisluku, luvun \(t\) tulee olla jaollinen kolmella.
  Pienin mahdollinen arvo on siis \(t=3\), jolloin kaikki kertoimet
  ovat positiivisia kokonaislukuja. Pienimmät kertoimet ovat siten
  \[
    (x_1,x_2,x_3,x_4)=(2,3,12,6).
  \]
  Tasapainotettu reaktioyhtälö on siis
  \[
    2\mathrm{C}_6\mathrm{H}_6+3\mathrm{O}_2
    \longrightarrow 12\mathrm{C}+6\mathrm{H}_2\mathrm{O}.
  \]
\end{enumerate}
\end{solution}

\begin{exercise}
Etsi kertoimet \(a\), \(b\) ja \(c\) siten, että funktion
\(f(x) = ax^2 + bx + c\) kuvaaja kulkee pisteiden \((1,2)\),
\((-1,6)\) ja \((2,3)\) kautta.
\end{exercise}

\begin{solution}
Kuvaaja kulkee annettujen pisteiden kautta täsmälleen silloin, kun
\(f(1)=2\), \(f(-1)=6\) ja \(f(2)=3\). Näistä ehdoista saadaan yhtälöryhmä
\[
\begin{aligned}
  &
  \begin{linearsystem}{2}
    a+b+c &= 2\\
    a-b+c &= 6 &\quad& \rowop{-R_1}\\
    4a+2b+c &= 3 &\quad& \rowop{-4R_1}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    a+b+c &= 2\\
    -2b &= 4 &\quad& \rowop{\cdot(-1/2)}\\
    -2b-3c &= -5
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    a+b+c &= 2 &\quad& \rowop{-R_2}\\
    b &= -2\\
    -2b-3c &= -5 &\quad& \rowop{+2R_2}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    a+c &= 4\\
    b &= -2\\
    -3c &= -9 &\quad& \rowop{\cdot(-1/3)}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    a+c &= 4 &\quad& \rowop{-R_3}\\
    b &= -2\\
    c &= 3
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    a &= 1\\
    b &= -2\\
    c &= 3
  \end{linearsystem}
\end{aligned}
\]
Kertoimet ovat siis yksikäsitteisesti
\[
(a,b,c)=(1,-2,3),
\]
ja vastaava funktio on \(f(x)=x^2-2x+3\). Tarkistetaan vielä, että
\[
\begin{aligned}
  f(1) &= 1-2+3=2, \\
  f(-1) &= 1+2+3=6, \\
  f(2) &= 4-4+3=3.
\end{aligned}
\]
Kuvaaja kulkee siis kaikkien kolmen annetun pisteen kautta.
\end{solution}

\begin{exercise}
Ratkaise yhtälöryhmä
\[
\begin{linearsystem}{4}
  x_1 &{}+{}& x_2 &{}+{}& x_3 &{}+{}& x_4 &= 0 \\
  2x_1 &{}+{}& 3x_2 &{}-{}& x_3 &{}-{}& x_4 &= 2 \\
  3x_1 &{}+{}& 2x_2 &{}+{}& x_3 &{}+{}& x_4 &= 5 \\
  3x_1 &{}+{}& 6x_2 &{}-{}& x_3 &{}-{}& x_4 &= 4.
\end{linearsystem}
\]
\end{exercise}

\begin{solution}
Koska tuntemattomien \(x_3\) ja \(x_4\) kertoimet ovat jokaisessa
yhtälössä samat, merkitään \(s=x_3+x_4\). Yhtälöryhmä saa muodon
\[
\begin{aligned}
  &
  \begin{linearsystem}{2}
    x_1+x_2+s &= 0\\
    2x_1+3x_2-s &= 2 &\quad& \rowop{-2R_1}\\
    3x_1+2x_2+s &= 5 &\quad& \rowop{-3R_1}\\
    3x_1+6x_2-s &= 4 &\quad& \rowop{-3R_1}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+x_2+s &= 0\\
    x_2-3s &= 2\\
    -x_2-2s &= 5 &\quad& \rowop{+R_2}\\
    3x_2-4s &= 4 &\quad& \rowop{-3R_2}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+x_2+s &= 0\\
    x_2-3s &= 2\\
    -5s &= 7\\
    5s &= -2 &\quad& \rowop{+R_3}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+x_2+s &= 0\\
    x_2-3s &= 2\\
    -5s &= 7\\
    0 &= 5
  \end{linearsystem}
\end{aligned}
\]
Yhtälö \(0=5\) on ristiriita. Alkuperäisellä yhtälöryhmällä ei siis
ole ratkaisuja,
ja sen ratkaisujoukko on \(\varnothing\).
\end{solution}

\begin{exercise}
Ratkaise yhtälöryhmä
\[
\begin{linearsystem}{6}
  x_1 &{}+{}& 3x_2 &{}-{}& 2x_3 & & &{}+{}&
  2x_5 & & &= 0 \\
  2x_1 &{}+{}& 6x_2 &{}-{}& 5x_3 &{}-{}& 2x_4 &{}+{}&
  4x_5 &{}-{}& 3x_6 &= -1 \\
  & & & & 5x_3 &{}+{}& 10x_4 & &
  &{}+{}& 15x_6 &= 5 \\
  2x_1 &{}+{}& 6x_2 & & &{}+{}& 8x_4 &{}+{}&
  4x_5 &{}+{}& 18x_6 &= 6.
\end{linearsystem}
\]
\end{exercise}

\begin{solution}
Tehdään seuraavat rivioperaatiot:
\[
\begin{aligned}
  &
  \begin{linearsystem}{2}
    x_1+3x_2-2x_3+2x_5 &= 0\\
    2x_1+6x_2-5x_3-2x_4+4x_5-3x_6 &= -1 &\quad& \rowop{-2R_1}\\
    5x_3+10x_4+15x_6 &= 5 &\quad& \rowop{\cdot(1/5)}\\
    2x_1+6x_2+8x_4+4x_5+18x_6 &= 6 &\quad& \rowop{-2R_1}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+3x_2-2x_3+2x_5 &= 0\\
    -x_3-2x_4-3x_6 &= -1 &\quad& \rowop{+R_3}\\
    x_3+2x_4+3x_6 &= 1\\
    4x_3+8x_4+18x_6 &= 6 &\quad& \rowop{-4R_3}
  \end{linearsystem}
\end{aligned}
\]
\[
\begin{aligned}
  \sim{}&
  \begin{linearsystem}{2}
    x_1+3x_2-2x_3+2x_5 &= 0\\
    0 &= 0\\
    x_3+2x_4+3x_6 &= 1\\
    6x_6 &= 2 &\quad& \rowop{\cdot(1/6)}
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+3x_2-2x_3+2x_5 &= 0\\
    0 &= 0\\
    x_3+2x_4+3x_6 &= 1 &\quad& \rowop{-3R_4}\\
    x_6 &= 1/3
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+3x_2-2x_3+2x_5 &= 0 &\quad& \rowop{+2R_3}\\
    0 &= 0\\
    x_3+2x_4 &= 0\\
    x_6 &= 1/3
  \end{linearsystem}\\[1ex]
  \sim{}&
  \begin{linearsystem}{2}
    x_1+3x_2+4x_4+2x_5 &= 0\\
    0 &= 0\\
    x_3+2x_4 &= 0\\
    x_6 &= 1/3
  \end{linearsystem}
\end{aligned}
\]
Siis \(x_6=\frac13\), \(x_3=-2x_4\) ja
\(x_1=-3x_2-4x_4-2x_5\).
Kaikki alkuperäiset yhtälöt toteutuvat näillä ehdoilla: sijoitusten
jälkeen toinen ja neljäs yhtälö ovat ensimmäisen monikertoja ja
kolmas toteutuu identtisesti. Muuttujat \(x_2\), \(x_4\) ja \(x_5\)
voidaan siis valita vapaasti.

Valitaan \(x_2=s\), \(x_4=t\) ja \(x_5=u\), missä \(s,t,u\in\RR\).
Ratkaisujoukon parametriesitys on
\[
\begin{aligned}
  \Bigl\{&
    \bigl(0,0,0,0,0,\frac13\bigr)
    +s
    (-3,1,0,0,0,0)
    +t
    (-4,0,-2,1,0,0)
    \\
    &\quad
    +u
    (-2,0,0,0,1,0)
    \colon\ s,t,u\in\RR
  \Bigr\}.
\end{aligned}
\]
\end{solution}

\begin{exercise}
Olkoon \(n \in \ZZ_+\). Todista monisteen Lemma 1.3.2:
yhtälöryhmillä
\[
\begin{linearsystem}{1}
  a_{11}x_1 + \dots + a_{1n}x_n &= b_1 \\
  a_{21}x_1 + \dots + a_{2n}x_n &= b_2
\end{linearsystem}
\]
ja
\[
\begin{linearsystem}{1}
  a_{11}x_1 + \dots + a_{1n}x_n &= b_1 \\
  (a_{21} + a_{11})x_1 + \dots + (a_{2n} +
  a_{1n})x_n &= b_2 + b_1
\end{linearsystem}
\]
on sama ratkaisujoukko.
\end{exercise}

\begin{solution}
Merkitään alkuperäisen yhtälöryhmän ratkaisujoukkoa kirjaimella \(R\)
ja jälkimmäisen ratkaisujoukkoa symbolilla \(R_+\). Siis
\[
\begin{aligned}
  R ={}& \bigl\{(x_1,\ldots,x_n)\in\RR^n
  \colon a_{11}x_1+\dots+a_{1n}x_n=b_1\bigr\} \\
  &\cap \bigl\{(x_1,\ldots,x_n)\in\RR^n
  \colon a_{21}x_1+\dots+a_{2n}x_n=b_2\bigr\}
\end{aligned}
\]
ja
\[
\begin{aligned}
  R_+ ={}& \bigl\{(x_1,\ldots,x_n)\in\RR^n
  \colon a_{11}x_1+\dots+a_{1n}x_n=b_1\bigr\} \\
  &\cap \bigl\{(x_1,\ldots,x_n)\in\RR^n \\
    &\qquad\colon (a_{21}+a_{11})x_1+\dots+
  (a_{2n}+a_{1n})x_n=b_2+b_1\bigr\}.
\end{aligned}
\]
Osoitetaan, että \(R\subseteq R_+\) ja \(R_+\subseteq R\).

Olkoon \((x_1,\ldots,x_n)\in R\). Tällöin
\[
\begin{aligned}
  a_{11}x_1+\dots+a_{1n}x_n &= b_1, \\
  a_{21}x_1+\dots+a_{2n}x_n &= b_2 \quad\rowop{+R_1}
\end{aligned}
\]
Laskemalla yhtälöt puolittain yhteen saadaan
\[
(a_{21}+a_{11})x_1+\dots+(a_{2n}+a_{1n})x_n=b_2+b_1.
\]
Koska myös ensimmäinen yhtälö toteutuu, vektori
\((x_1,\ldots,x_n)\) kuuluu joukkoon \(R_+\). Siis \(R\subseteq R_+\).

Olkoon kääntäen \((x_1,\ldots,x_n)\in R_+\). Tällöin
\[
\begin{aligned}
  a_{11}x_1+\dots+a_{1n}x_n &= b_1, \\
  (a_{21}+a_{11})x_1+\dots+(a_{2n}+a_{1n})x_n &= b_2+b_1 \quad\rowop{-R_1}
\end{aligned}
\]
Vähentämällä jälkimmäisestä yhtälöstä ensimmäinen puolittain saadaan
\[
\bigl((a_{21}+a_{11})-a_{11}\bigr)x_1+\dots+
\bigl((a_{2n}+a_{1n})-a_{1n}\bigr)x_n=(b_2+b_1)-b_1,
\]
eli
\[
a_{21}x_1+\dots+a_{2n}x_n=b_2.
\]
Ensimmäinen yhtälö toteutuu jo oletuksen nojalla, joten
\((x_1,\ldots,x_n)\in R\). Siis \(R_+\subseteq R\).

Näin ollen \(R=R_+\), eli yhtälöryhmillä on sama ratkaisujoukko.
\end{solution}
