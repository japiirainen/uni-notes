\subsection{27.10.2026: lineaariset yhtälöryhmät}

Lineaarisen yhtälöryhmän ratkaisu toteuttaa kaikki ryhmän yhtälöt
samanaikaisesti. Geometrisesti ratkaiseminen tarkoittaa yhtälöiden
ratkaisujoukkojen yhteisten pisteiden etsimistä.

\subsubsection*{Suorat tasossa ja niiden leikkauspisteet}

Yhtälö $y=ax+b$, missä $a,b\in\RR$, esittää tason suoraa.
Suora voidaan ajatella pistejoukkona
\[
  L=\{(x,y)\in\RR^2:y=ax+b\}.
\]
Kahden suoran $y=ax+b$ ja $y=a'x+b'$ yhteiset pisteet muodostavat
joukon $L\cap L'$. Ne saadaan ratkaisemalla molemmat yhtälöt yhtä aikaa.

\begin{example}[Kahden suoran leikkaus]
  Ratkaistaan yhtälöryhmä
  \[
    \begin{cases}
      y=2x+3,\\
      y=-x+2.
    \end{cases}
  \]
  Koska molemmat oikeat puolet ovat yhtä suuret kuin $y$, saadaan
  \[
    2x+3=-x+2
    \quad\Longleftrightarrow\quad
    3x=-1
    \quad\Longleftrightarrow\quad
    x=-\frac13.
  \]
  Sijoittamalla toiseen yhtälöön saadaan
  \[
    y=-\left(-\frac13\right)+2=\frac73.
  \]
  Suorien ainoa leikkauspiste on siis
  \[
    \boxed{(x,y)=\left(-\frac13,\frac73\right).}
  \]
\end{example}

\subsubsection*{Suoran ja tason yleinen yhtälö}

Muoto $y=ax+b$ ei kata pystysuoria suoria $x=c$. Kaikki tason suorat
voidaan esittää muodossa
\[
  \boxed{a_1x_1+a_2x_2=b,\qquad (a_1,a_2)\ne(0,0),}
\]
missä $a_1,a_2,b\in\RR$. Esimerkiksi $x_1=c$ saadaan valitsemalla
$a_1=1$, $a_2=0$ ja $b=c$. Suoran pistejoukko on
\[
  L=\{(x_1,x_2)\in\RR^2:a_1x_1+a_2x_2=b\}.
\]

\begin{remark}[Yhtälön kertominen vakiolla]
  Kun $\lambda\ne0$, yhtälön molemmat puolet voidaan kertoa luvulla
  $\lambda$ muuttamatta ratkaisujoukkoa:
  \[
    a_1x_1+a_2x_2=b
    \quad\Longleftrightarrow\quad
    \lambda a_1x_1+\lambda a_2x_2=\lambda b.
  \]
\end{remark}

Vastaavasti avaruuden $\RR^3$ tason yleinen yhtälö on
\[
  a_1x_1+a_2x_2+a_3x_3=b,
  \qquad (a_1,a_2,a_3)\ne(0,0,0).
\]

\subsubsection*{Yhtälöryhmä, ratkaisu ja ratkaisujoukko}

\begin{definition}[Lineaarinen yhtälöryhmä]
  Olkoot $m,n$ positiivisia kokonaislukuja ja
  $a_{ij},b_i\in\RR$, missä $i=1,\ldots,m$ ja $j=1,\ldots,n$.
  Ryhmää
  \begin{equation}\label{eq:linear-system}
    \left\{
      \begin{aligned}
        a_{11}x_1+\cdots+a_{1n}x_n&=b_1,\\
        &\ \vdots\\
        a_{m1}x_1+\cdots+a_{mn}x_n&=b_m
      \end{aligned}\right.
      \tag{YR}
    \end{equation}
    kutsutaan \emph{lineaariseksi yhtälöryhmäksi}.
    Siinä on $m$ yhtälöä ja $n$ tuntematonta.
    Kerroin $a_{ij}$ on yhtälön $i$ muuttujan $x_j$ kerroin.
  \end{definition}

  \begin{definition}[Ratkaisu ja ratkaisujoukko]
    Vektori $(x_1,\ldots,x_n)\in\RR^n$ on ryhmän
    \eqref{eq:linear-system} \emph{ratkaisu}, jos se toteuttaa ryhmän
    kaikki yhtälöt. Kaikkien ratkaisujen joukko on
    \[
      R=\left\{(x_1,\ldots,x_n)\in\RR^n:
      \sum_{j=1}^n a_{ij}x_j=b_i\ \text{kaikilla }i=1,\ldots,m\right\}.
    \]
  \end{definition}

  \noindent\textbf{Geometrinen tulkinta.}
  Jos $L_i$ on yhtälön $i$ ratkaisujoukko, koko ryhmän ratkaisujoukko on
  $R=L_1\cap\cdots\cap L_m$.

  \newpage
  \subsubsection*{Avaruuden \texorpdfstring{$\RR^n$}{Rⁿ} laskutoimitukset}

  Avaruuden $\RR^n$ alkiot ovat vektoreita $(x_1,\ldots,x_n)$.
  Vektorien yhteenlasku ja skalaarilla kertominen tehdään
  \emph{koordinaateittain}.

  \begin{definition}[Yhteenlasku ja skalaarilla kertominen]
    Yhteenlasku $+\colon\RR^n\times\RR^n\to\RR^n$ määritellään kaavalla
    \[
      \boxed{(x_1,\ldots,x_n)+(y_1,\ldots,y_n)
      =(x_1+y_1,\ldots,x_n+y_n).}
    \]
    Skalaarilla kertominen $\RR\times\RR^n\to\RR^n$ määritellään kaavalla
    \[
      \boxed{c(x_1,\ldots,x_n)=(cx_1,\ldots,cx_n).}
    \]
    Tässä $c\in\RR$ on \emph{skalaari}.
  \end{definition}

  \begin{example}
    Kun $u=(1,-2)$ ja $v=(3,4)$, saadaan
    \[
      u+v=(4,2),\qquad 2u=(2,-4).
    \]
  \end{example}

  \subsubsection*{Vektorien laskusäännöt}
  Kaikilla $u,v,w\in\RR^n$ ja $c,d\in\RR$ pätevät
  \begin{align*}
    u+v&=v+u, & (u+v)+w&=u+(v+w),\\
    u+0&=u, & u+(-u)&=0,\\
    c(u+v)&=cu+cv, & (c+d)u&=cu+du,\\
    c(du)&=(cd)u, & 1u&=u.
  \end{align*}
  Näiden laskutoimitusten kanssa $\RR^n$ on vektoriavaruus kunnan
  $\RR$ yli. Yleisemmin puhutaan vektoriavaruudesta kunnan $K$ yli;
  renkaan yli vastaava käsite on $K$-moduuli.

  \subsubsection*{Yhtälöt $0=0$ ja $0=b_i$}

  Yhtälöryhmässä voi esiintyä rivi, jonka kaikki kertoimet ovat nollia:
  \[
    0x_1+0x_2+\cdots+0x_n=b_i.
  \]
  Tällöin on erotettava kaksi tapausta.
  \begin{description}[style=nextline,leftmargin=0pt]
    \item[$b_i=0$: yhtälö $0=0$.]
      Rivi toteutuu kaikilla vektoreilla eikä aseta ratkaisuille mitään
      ehtoa. Sen voi poistaa muuttamatta ryhmän ratkaisujoukkoa.
    \item[$b_i\ne0$: yhtälö $0=b_i$.]
      Rivi ei toteudu millään vektorilla. Siksi koko yhtälöryhmällä
      ei ole ratkaisuja: $\boxed{R=\varnothing}$.
  \end{description}

  \newpage
  \subsection{29.10.2026: rivioperaatiot ja Gauss--Jordanin menetelmä}

  Rivioperaatioilla yhtälöryhmä muutetaan helpommin luettavaan muotoon
  säilyttäen sen ratkaisujoukko. Tavoitteena on selvittää, onko ratkaisuja,
  ja esittää kaikki ratkaisut sidottujen ja vapaiden muuttujien avulla.

  \subsubsection*{Sallitut rivioperaatiot ja niiden perustelut}

  \begin{definition}[Ekvivalentit yhtälöryhmät]
    Kaksi yhtälöryhmää, joissa on samat muuttujat, ovat
    \emph{ekvivalentit}, jos niiden ratkaisujoukot ovat samat.
  \end{definition}

  Merkitään ryhmän $i$:ttä yhtälöä eli riviä symbolilla $R_i$ ja sen
  vasenta puolta $L_i(x)=\sum_{j=1}^n a_{ij}x_j$. Rivi on siis
  $L_i(x)=b_i$. Jokainen operaatio tehdään \emph{yhtälön molemmille puolille}.

  \begin{lemma}[Rivin kertominen]
    Kun $\lambda\ne0$, operaatio $R_i\leftarrow\lambda R_i$
    säilyttää ratkaisujoukon.
  \end{lemma}
  \begin{proof}
    Koska $\lambda\ne0$, jokaisella $x\in\RR^n$ pätee
    \[
      L_i(x)=b_i\quad\Longleftrightarrow\quad
      \lambda L_i(x)=\lambda b_i.
    \]
    Suunta oikealle saadaan kertomalla luvulla $\lambda$ ja suunta
    vasemmalle jakamalla sillä. Muut yhtälöt pysyvät samoina.
  \end{proof}

  \begin{lemma}[Toisen rivin monikerran lisääminen]
    Kun $i\ne j$ ja $c\in\RR$, operaatio
    $R_i\leftarrow R_i+cR_j$ säilyttää ratkaisujoukon.
    Rivi $R_j$ säilytetään ennallaan.
  \end{lemma}
  \begin{proof}
    Jos $x$ toteuttaa alkuperäiset yhtälöt, niin
    $L_i(x)+cL_j(x)=b_i+cb_j$, joten se toteuttaa myös uuden ryhmän.
    Kääntäen uuden ryhmän ratkaisulle pätevät
    \[
      L_i(x)+cL_j(x)=b_i+cb_j,\qquad L_j(x)=b_j.
    \]
    Vähentämällä ensimmäisestä yhtälöstä $c$ kertaa toinen saadaan
    $L_i(x)=b_i$. Muut yhtälöt eivät muutu, joten ratkaisujoukot ovat samat.
  \end{proof}

  \begin{lemma}[Rivien järjestyksen vaihtaminen]
    Yhtälöiden järjestyksen vaihtaminen säilyttää ratkaisujoukon.
    Erityisesti voidaan vaihtaa kaksi riviä: $R_i\leftrightarrow R_j$.
  \end{lemma}
  \begin{proof}
    Olkoon $S_i$ yhtälön $i$ ratkaisujoukko ja $\sigma$ joukon
    $\{1,\ldots,m\}$ permutaatio. Tällöin
    \[
      \bigcap_{i=1}^m S_i=\bigcap_{i=1}^m S_{\sigma(i)},
    \]
    sillä kummassakin leikkauksessa vaaditaan täsmälleen samojen
    yhtälöiden toteutumista.
  \end{proof}

  \noindent\textbf{Muista.}
  Kaikki kolme operaatiota voidaan perua: kertoimella $1/\lambda$,
  operaatiolla $R_i\leftarrow R_i-cR_j$ tai vaihtamalla rivit takaisin.
  Siksi myös niiden äärellinen jono säilyttää ratkaisujoukon.
  Nollalla kertominen ei ole sallittua: esimerkiksi $x=1$ muuttuisi
  ehdoksi $0=0$. Myöskään kahden yhtälön korvaaminen pelkällä niiden
  summalla ei yleensä säilytä ratkaisujoukkoa.

  \subsubsection*{Matriisimerkintä ja porrasmuodot}

  Laskemista tiivistää \emph{laajennettu matriisi}
  \[
    [A\mid b]=\left[
      \begin{array}{ccc|c}
        a_{11}&\cdots&a_{1n}&b_1\\
        \vdots&&\vdots&\vdots\\
        a_{m1}&\cdots&a_{mn}&b_m
    \end{array}\right].
  \]
  Sarakkeet ennen pystyviivaa vastaavat muuttujia $x_1,\ldots,x_n$;
  viimeinen sarake sisältää oikeat puolet. Rivioperaatio koskee aina
  koko riviä, myös viimeistä saraketta. Merkintä $M\sim N$ tarkoittaa,
  että matriisit saadaan toisistaan rivioperaatioilla.

  \noindent\textbf{Rivioperaatioiden merkintä.}
  Rivin vieressä $\rowop{-2R_1}$ tarkoittaa, että kyseisestä rivistä
  vähennetään kaksi kertaa ensimmäinen rivi. Merkintä
  $\rowop{\cdot(1/2)}$ tarkoittaa kyseisen rivin kertomista luvulla $1/2$.
  Merkinnät koskevat siirtymää seuraavaan matriisiin; samalla kertaa
  merkityissä operaatioissa käytetään lähtömatriisin rivejä.
  Rivinvaihto merkitään $\underset{R_i\leftrightarrow R_j}{\sim}$.

  \begin{definition}[Porrasmuoto]
    Rivin ensimmäinen nollasta poikkeava alkio on sen \emph{johtava alkio}
    eli \emph{pivot}. Matriisi on tässä käytetyn määritelmän mukaan
    \emph{porrasmuodossa}, jos
    \begin{enumerate}
      \item jokaisen nollasta poikkeavan rivin johtava alkio on $1$;
      \item kaikki nollarivit ovat alimpina;
      \item alemman rivin johtava alkio on aina ylemmän rivin johtavan
        alkion oikealla puolella.
    \end{enumerate}
    Se on \emph{supistetussa porrasmuodossa}, jos lisäksi jokainen
    johtava $1$ on sarakkeensa ainoa nollasta poikkeava alkio.
  \end{definition}

  Ehto 3 takaa nollat johtavien alkioiden alapuolelle; supistetussa
  muodossa myös niiden yläpuolella on nollia. Joissakin lähteissä
  porrasmuodon johtavilta alkioilta ei vaadita arvoa $1$. Tässä
  noudatetaan luennon määritelmää.
  Esimerkiksi
  \[
    \underbrace{\left[
        \begin{array}{ccc|c}
          1&2&0&3\\0&0&1&4\\0&0&0&0
    \end{array}\right]}_{\text{supistettu porrasmuoto}},\qquad
    \underbrace{\left[
        \begin{array}{ccc|c}
          1&2&3&4\\0&1&5&6\\0&0&0&0
    \end{array}\right]}_{\text{porrasmuoto, ei supistettu}}.
  \]

  \begin{remark}[Nollarivi ja ristiriitarivi]
    Laajennetun matriisin nollarivi $[0\ \cdots\ 0\mid0]$ tarkoittaa
    yhtälöä $0=0$. Rivi $[0\ \cdots\ 0\mid d]$, missä $d\ne0$,
    ei ole nollarivi, vaan ristiriita $0=d$. Siinä johtava alkio on
    oikean puolen sarakkeessa, eikä se vastaa mitään muuttujaa.
  \end{remark}

  \newpage
  \subsubsection*{Miksi eliminointi onnistuu aina?}

  \begin{theorem}[Gauss--Jordan]
    Jokainen laajennettu matriisi voidaan saattaa rivioperaatioilla
    supistettuun porrasmuotoon. Näin saatu yhtälöryhmä on ekvivalentti
    alkuperäisen kanssa.
  \end{theorem}
  \begin{proof}
    Nollamatriisi on jo supistetussa porrasmuodossa. Muuten
    etsitään vasemmanpuoleisin sarake, jossa on nollasta poikkeava
    alkio. Vaihdetaan tällainen alkio ensimmäiselle riville,
    jaetaan rivi tällä alkiolla ja nollataan sen alapuoliset alkiot
    lisäämällä sopivia ensimmäisen rivin monikertoja.

    Toistetaan sama alemmille riveille ja johtavan alkion oikealla
    puolella oleville sarakkeille. Jos jäljellä olevat rivit ovat
    nollia, lopetetaan. Aiemmat johtavat sarakkeet säilyvät alemmilla
    riveillä nollina. Jokainen vaihe siirtyy yhden rivin alas ja
    vähintään yhden sarakkeen oikealle, joten vaiheita on äärellisesti.
    Tuloksena on porrasmuoto.

    Käydään sitten johtavat ykköset läpi alhaalta ylöspäin ja
    nollataan kunkin yläpuoliset alkiot kyseisen rivin monikerroilla.
    Jo käsitellyt johtavat sarakkeet eivät muutu, sillä käsiteltävällä
    rivillä on niissä nollat. Saadaan supistettu porrasmuoto.
    Kaikki käytetyt operaatiot säilyttävät ratkaisujoukon.
  \end{proof}

  \noindent\textbf{Gauss vai Gauss--Jordan?}
  Gaussin eliminoinnissa riittää porrasmuoto, josta muuttujat ratkaistaan
  alimmasta yhtälöstä alkaen \emph{takaisinsijoituksella}.
  Gauss--Jordanissa jatketaan supistettuun porrasmuotoon, josta ratkaisut
  voidaan lukea suoraan.

  \subsubsection*{Sidotut muuttujat, vapaat muuttujat ja ratkaisujen määrä}

  \begin{definition}
    Porrasmuodossa muuttuja on \emph{sidottu}, jos sen sarakkeessa
    on johtava alkio. Muut muuttujat ovat \emph{vapaita}.
  \end{definition}

  \begin{theorem}[Ratkaisujen lukumäärä reaaliluvuilla]
    Tarkastellaan rivioperaatioilla saatua supistettua porrasmuotoa.
    \begin{enumerate}
      \item Jos siinä on rivi $[0\ \cdots\ 0\mid d]$, $d\ne0$,
        ratkaisuja ei ole.
      \item Jos ristiriitariviä ei ole ja kaikki muuttujat ovat
        sidottuja, ratkaisuja on täsmälleen yksi.
      \item Jos ristiriitariviä ei ole ja ainakin yksi muuttuja on
        vapaa, ratkaisuja on äärettömän monta.
    \end{enumerate}
  \end{theorem}
  \begin{proof}
    Ristiriitarivi vaatisi $0=d\ne0$, mikä on mahdotonta.
    Ilman ristiriitariviä jokainen ei-nollarivi voidaan kirjoittaa muodossa
    \[
      x_{p_i}=b_i-\sum_{j\in F}a_{ij}x_j,
    \]
    missä $x_{p_i}$ on rivin sidottu muuttuja ja $F$ on vapaiden
    muuttujien indeksijoukko. Muita sidottuja muuttujia ei esiinny
    tällä rivillä, koska niiden johtavissa sarakkeissa on muualla nollat.
    Vapaat muuttujat voidaan siis valita mielivaltaisesti, minkä jälkeen
    jokainen sidottu muuttuja määräytyy yksikäsitteisesti.
    Jos $F=\varnothing$, saadaan yksi ratkaisu. Jos $F\ne\varnothing$,
    jo yhden vapaan muuttujan vaihteleminen reaaliluvuissa antaa
    äärettömän monta eri ratkaisua.
  \end{proof}

  \newpage
  \subsubsection*{Esimerkkejä rivioperaatioiden käytöstä}

  \begin{example}[Yksi ratkaisu: kaikki kolme rivioperaatiota]
    Ratkaistaan
    \[
      \begin{cases}
        2y+2z=10,\\
        x+y+z=6,\\
        2x+3y+4z=20.
      \end{cases}
    \]
    Vaihdetaan ensin rivit, jotta ensimmäinen johtava alkio on $1$:
    \begin{align*}
      \left[
        \begin{array}{ccc|c}0&2&2&10\\1&1&1&6\\2&3&4&20
      \end{array}\right]
      &\underset{R_1\leftrightarrow R_2}{\sim}
      \left[
        \begin{array}{ccc|c}1&1&1&6\\0&2&2&10\\2&3&4&20
      \end{array}\right]
      \begin{array}{@{}l@{}}\vphantom{0}\\\vphantom{0}\\\rowop{-2R_1}
      \end{array}\\[3pt]
      &\sim
      \left[
        \begin{array}{ccc|c}1&1&1&6\\0&2&2&10\\0&1&2&8
      \end{array}\right]
      \begin{array}{@{}l@{}}\vphantom{0}\\\rowop{\cdot(1/2)}\\\vphantom{0}
      \end{array}\\[3pt]
      &\sim
      \left[
        \begin{array}{ccc|c}1&1&1&6\\0&1&1&5\\0&1&2&8
      \end{array}\right]
      \begin{array}{@{}l@{}}\vphantom{0}\\\vphantom{0}\\\rowop{-R_2}
      \end{array}\\[3pt]
      &\sim
      \left[
        \begin{array}{ccc|c}1&1&1&6\\0&1&1&5\\0&0&1&3
      \end{array}\right].
    \end{align*}
    Nyt ollaan porrasmuodossa. Takaisinsijoitus antaa
    $z=3$, $y=5-z=2$ ja $x=6-y-z=1$.
    Gauss--Jordanin menetelmällä jatkettaisiin vielä:
    \begin{align*}
      \left[
        \begin{array}{ccc|c}1&1&1&6\\0&1&1&5\\0&0&1&3
      \end{array}\right]
      \begin{array}{@{}l@{}}\rowop{-R_3}\\\rowop{-R_3}\\\vphantom{0}
      \end{array}
      &\sim
      \left[
        \begin{array}{ccc|c}1&1&0&3\\0&1&0&2\\0&0&1&3
      \end{array}\right]
      \begin{array}{@{}l@{}}\rowop{-R_2}\\\vphantom{0}\\\vphantom{0}
      \end{array}
      \\[3pt]
      &\sim
      \left[
        \begin{array}{ccc|c}1&0&0&1\\0&1&0&2\\0&0&1&3
      \end{array}\right].
    \end{align*}
    Siis $\boxed{(x,y,z)=(1,2,3)}$.
    Tarkistus alkuperäisiin yhtälöihin: $4+6=10$, $1+2+3=6$ ja
    $2+6+12=20$.
  \end{example}

  \begin{example}[Ei ratkaisuja: ristiriita tulee näkyviin]
    \[
      \begin{cases}x+2y=3,\\2x+4y=7
      \end{cases}
      \quad\longrightarrow\quad
      \left[
        \begin{array}{cc|c}1&2&3\\2&4&7
      \end{array}\right]
      \begin{array}{@{}l@{}}\vphantom{0}\\\rowop{-2R_1}
      \end{array}
      \sim
      \left[
        \begin{array}{cc|c}1&2&3\\0&0&1
      \end{array}\right].
    \]
    Viimeinen rivi tarkoittaa $0=1$, joten ratkaisujoukko on
    $\boxed{\varnothing}$. Eliminoinnin voi lopettaa heti ristiriitaan;
    supistettua porrasmuotoa ei tarvitse laskea loppuun.
  \end{example}

  \newpage
  \begin{example}[Äärettömän monta ratkaisua: kaksi vapaata muuttujaa]
    Ratkaistaan neljän muuttujan yhtälöryhmä
    \[
      \begin{cases}
        x_1+2x_2-x_3+x_4=3,\\
        2x_1+4x_2-x_3+3x_4=8,\\
        3x_1+6x_2-2x_3+4x_4=11.
      \end{cases}
    \]
    Nollataan ensimmäisen johtavan alkion alapuoli:
    \[
      \left[
        \begin{array}{rrrr|r}
          1&2&-1&1&3\\2&4&-1&3&8\\3&6&-2&4&11
      \end{array}\right]
      \begin{array}{@{}l@{}}\vphantom{0}\\\rowop{-2R_1}\\\rowop{-3R_1}
      \end{array}
      \sim
      \left[
        \begin{array}{rrrr|r}
          1&2&-1&1&3\\0&0&1&1&2\\0&0&1&1&2
      \end{array}\right].
    \]
    Toisen sarakkeen jäljellä olevat alkiot ovat nollia, joten seuraava
    johtava alkio tulee kolmanteen sarakkeeseen:
    \begin{align*}
      \left[
        \begin{array}{rrrr|r}
          1&2&-1&1&3\\0&0&1&1&2\\0&0&1&1&2
      \end{array}\right]
      \begin{array}{@{}l@{}}\vphantom{0}\\\vphantom{0}\\\rowop{-R_2}
      \end{array}
      &\sim
      \left[
        \begin{array}{rrrr|r}
          1&2&-1&1&3\\0&0&1&1&2\\0&0&0&0&0
      \end{array}\right]
      \begin{array}{@{}l@{}}\rowop{+R_2}\\\vphantom{0}\\\vphantom{0}
      \end{array}\\[3pt]
      &\sim
      \left[
        \begin{array}{rrrr|r}
          1&2&0&2&5\\0&0&1&1&2\\0&0&0&0&0
      \end{array}\right].
    \end{align*}
    Sidotut muuttujat ovat $x_1,x_3$ ja vapaat muuttujat $x_2,x_4$.
    Valitaan $x_2=s$ ja $x_4=t$, missä $s,t\in\RR$. Tällöin
    \[
      x_1=5-2s-2t,\qquad x_3=2-t,
    \]
    joten koko ratkaisujoukko on
    \[
      \boxed{R=\{(5-2s-2t,\ s,\ 2-t,\ t):s,t\in\RR\}.}
    \]
    Vektorimuodossa sama ratkaisu on
    \[
      (x_1,x_2,x_3,x_4)
      =(5,0,2,0)+s(-2,1,0,0)+t(-2,0,-1,1).
    \]
    Esimerkiksi $s=t=0$ antaa ratkaisun $(5,0,2,0)$ ja $s=1,t=0$
    ratkaisun $(3,1,2,0)$. Vain nämä esimerkit luettelemalla ei kuitenkaan
    saataisi kaikkia ratkaisuja: siihen tarvitaan vapaat parametrit.
  \end{example}

  \subsubsection*{Kertaus: miten luet lopputuloksen?}
  \begin{enumerate}
    \item Tarkista ensin ristiriitarivit. Yksikin $0=d\ne0$ tarkoittaa,
      ettei ratkaisuja ole, vaikka mukana olisi vapaita muuttujia.
    \item Jos ristiriitaa ei ole, etsi muuttujasarakkeiden johtavat alkiot.
      Muut muuttujat saavat omat vapaat parametrinsa.
    \item Ratkaise sidotut muuttujat ja kirjoita \emph{koko} ratkaisujoukko.
      Jos vapaita muuttujia ei ole, ratkaisu on yksikäsitteinen.
    \item Tarkista tulos sijoittamalla alkuperäisiin yhtälöihin.
      Nollarivi $0=0$ ei itsessään tarkoita äärettömän montaa ratkaisua:
      ratkaisevaa on vapaiden muuttujien olemassaolo.
  \end{enumerate}

  \subsection{3.11.2026: matriisit ja matriisikertolasku}

  % Lähdekuvat: IMG_1999.HEIC ja IMG_2001.HEIC--IMG_2005.HEIC.
  Matriisi on suorakulmainen lukutaulukko. Sen avulla voidaan esittää
  sekä lineaarinen yhtälöryhmä että kuvaus, joka vie vektorin toiseksi
  vektoriksi. Aloitetaan merkinnöistä ja laskutoimituksista.

  \subsubsection*{Matriisi, rivit ja sarakkeet}

  \begin{definition}[Matriisi]
    Olkoot $m,n$ positiivisia kokonaislukuja. Reaalinen
    \emph{$m\times n$-matriisi} on funktio
    \[
      A\colon\{1,\ldots,m\}\times\{1,\ldots,n\}\to\RR.
    \]
    Merkitään $a_{ij}=A(i,j)$. Matriisi kirjoitetaan taulukkona
    \[
      A=[a_{ij}]=
      \begin{bmatrix}
        a_{11}&\cdots&a_{1n}\\
        \vdots&&\vdots\\
        a_{m1}&\cdots&a_{mn}
      \end{bmatrix}\in\RR^{m\times n}.
    \]
    Siinä on $m$ riviä ja $n$ saraketta. Alkion $a_{ij}$ ensimmäinen
    indeksi kertoo rivin ja toinen sarakkeen. Lukua $a_{ij}$ kutsutaan
    matriisin \emph{alkioksi} eli \emph{kertoimeksi}.
  \end{definition}

  Kaksi matriisia ovat samat, jos ne ovat samankokoiset ja niiden
  vastinalkiot ovat samat. Tämä antaa myös tavallisen todistustavan:
  matriisiyhtälö todistetaan tarkistamalla sen jokainen alkio.

  Matriisin $i$:s \emph{rivivektori} ja $j$:s \emph{sarakevektori} ovat
  \[
    A_{i\cdot}=[a_{i1}\ \cdots\ a_{in}]\in\RR^{1\times n},
    \qquad
    A_{\cdot j}=
    \begin{bmatrix}a_{1j}\\\vdots\\a_{mj}
    \end{bmatrix}
    \in\RR^{m\times1}.
  \]
  Siten matriisin voi koota joko riveistä tai sarakkeista:
  \[
    A=
    \begin{bmatrix}A_{1\cdot}\\\vdots\\A_{m\cdot}
    \end{bmatrix}
    =[A_{\cdot1}\ \cdots\ A_{\cdot n}].
  \]
  Kun kirjoitamme rivejä tai sarakkeita vierekkäin tai allekkain,
  tarkoitamme näin muodostettua matriisia.

  \begin{remark}[Vektorit ja skalaarit matriiseina]
    Samaistamme vektorin $(x_1,\ldots,x_n)\in\RR^n$ sarakematriisiin
    $[x_1\ \cdots\ x_n]^{\mathsf T}\in\RR^{n\times1}$; merkintä
    $\mathsf T$ kääntää tässä rivin sarakkeeksi.
    Vastaavasti $1\times1$-matriisi $[a]$ samaistetaan lukuun $a\in\RR$.
    Rivi- ja sarakevektori on silti erotettava toisistaan, sillä niiden
    koot ovat erilaiset.
  \end{remark}

  \newpage
  \subsubsection*{Yhteenlasku ja skalaarilla kertominen}

  \begin{definition}
    Olkoot $A=[a_{ij}],B=[b_{ij}]\in\RR^{m\times n}$ ja
    $\lambda\in\RR$. Määritellään \emph{summa} ja
    \emph{skalaarilla kertominen} alkioittain:
    \[
      \boxed{(A+B)_{ij}=a_{ij}+b_{ij},\qquad
      (\lambda A)_{ij}=\lambda a_{ij}.}
    \]
    Summa on määritelty vain samankokoisille matriiseille.
    Molemmat tulokset kuuluvat joukkoon $\RR^{m\times n}$.
  \end{definition}

  \begin{lemma}[Summan ja skalaarikertolaskun laskusäännöt]
    Olkoot $A,B,C\in\RR^{m\times n}$ ja $\lambda,\mu\in\RR$.
    Tällöin
    \begin{align*}
      (A+B)+C&=A+(B+C), & A+B&=B+A,\\
      \lambda(A+B)&=\lambda A+\lambda B,
      & (\lambda\mu)A&=\lambda(\mu A),\\
      (\lambda+\mu)A&=\lambda A+\mu A.
    \end{align*}
  \end{lemma}
  \begin{proof}
    Kaikki yhtälöissä esiintyvät matriisit ovat kokoa $m\times n$.
    Kiinnitetään $i,j$. Reaalilukujen laskusäännöt antavat
    \begin{align*}
      ((A+B)+C)_{ij}
      &=(a_{ij}+b_{ij})+c_{ij}
      =a_{ij}+(b_{ij}+c_{ij})=(A+(B+C))_{ij},\\
      (A+B)_{ij}&=a_{ij}+b_{ij}=b_{ij}+a_{ij}=(B+A)_{ij},\\
      (\lambda(A+B))_{ij}
      &=\lambda(a_{ij}+b_{ij})
      =\lambda a_{ij}+\lambda b_{ij}=(\lambda A+\lambda B)_{ij},\\
      ((\lambda\mu)A)_{ij}
      &=(\lambda\mu)a_{ij}=\lambda(\mu a_{ij})
      =(\lambda(\mu A))_{ij},\\
      ((\lambda+\mu)A)_{ij}
      &=(\lambda+\mu)a_{ij}=\lambda a_{ij}+\mu a_{ij}
      =(\lambda A+\mu A)_{ij}.
    \end{align*}
    Koska vastinalkiot ovat samat kaikilla $i,j$, matriisit ovat samat.
  \end{proof}

  \emph{Nollamatriisin} $0_{m\times n}$ kaikki alkiot ovat nollia ja
  \emph{vastamatriisi} on $-A=[-a_{ij}]$. Alkioittain nähdään myös, että
  \[
    A+0_{m\times n}=A,\qquad A+(-A)=0_{m\times n},\qquad 1A=A.
  \]
  Näin $\RR^{m\times n}$ on reaalinen vektoriavaruus samoilla
  laskusäännöillä kuin $\RR^n$. Erotus tarkoittaa summaa $A-B=A+(-B)$.

  \subsubsection*{Matriisikertolaskun koko ja määritelmä}

  \begin{definition}[Matriisitulo]
    Olkoot $A=[a_{ij}]\in\RR^{m\times n}$ ja
    $B=[b_{j\ell}]\in\RR^{n\times k}$. Niiden \emph{tulo}
    $AB\in\RR^{m\times k}$ määritellään kaavalla
    \[
      \boxed{(AB)_{i\ell}=\sum_{j=1}^n a_{ij}b_{j\ell}
      =a_{i1}b_{1\ell}+\cdots+a_{in}b_{n\ell}.}
    \]
    Alkio $(i,\ell)$ saadaan siis $A$:n rivin $i$ ja $B$:n sarakkeen
    $\ell$ pistetulona: vastinalkiot kerrotaan ja tulot lasketaan yhteen.
  \end{definition}

  \newpage
  Tulon määrittely edellyttää, että vasemman matriisin sarakkeita on
  yhtä monta kuin oikean matriisin rivejä. Tuloksen koon määräävät
  ulommat luvut:
  \[
    \underbrace{(m\times n)(n\times k)}_{\text{sisemmat koot samat}}
    \quad\longrightarrow\quad m\times k.
  \]
  Matriisitulo ei siis ole alkioittainen kertolasku. Myös tekijöiden
  järjestyksellä on väliä: esimerkiksi seuraavan esimerkin $AB$ on
  kokoa $2\times2$, mutta $BA$ olisi kokoa $3\times3$.

  \begin{example}[Luennon laskutehtävä: $AB+2C$]
    Olkoot
    \[
      A=
      \begin{bmatrix}1&0&-1\\-2&1&2
      \end{bmatrix},\qquad
      B=
      \begin{bmatrix}3&2\\-1&1\\0&1
      \end{bmatrix},\qquad
      C=
      \begin{bmatrix}-1&2\\3&4
      \end{bmatrix}.
    \]
    Koska $A\in\RR^{2\times3}$ ja $B\in\RR^{3\times2}$,
    tulo $AB$ on kokoa $2\times2$, kuten $C$. Summa on siis määritelty.
    Lasketaan jokainen alkio rivin ja sarakkeen avulla:
    \begin{align*}
      (AB)_{11}&=1\cdot3+0\cdot(-1)+(-1)\cdot0=3,\\
      (AB)_{12}&=1\cdot2+0\cdot1+(-1)\cdot1=1,\\
      (AB)_{21}&=(-2)\cdot3+1\cdot(-1)+2\cdot0=-7,\\
      (AB)_{22}&=(-2)\cdot2+1\cdot1+2\cdot1=-1.
    \end{align*}
    Siis
    \[
      AB=
      \begin{bmatrix}3&1\\-7&-1
      \end{bmatrix},\qquad
      2C=
      \begin{bmatrix}-2&4\\6&8
      \end{bmatrix},
    \]
    ja siten
    \[
      \boxed{AB+2C=
        \begin{bmatrix}1&5\\-1&7
      \end{bmatrix}.}
    \]
  \end{example}

  \subsubsection*{Matriisin ja vektorin tulo; yhteys yhtälöryhmiin}

  Erityisesti, kun $x=[x_1\ \cdots\ x_n]^{\mathsf T}\in\RR^{n\times1}$,
  matriisikertolaskun määritelmä antaa
  \[
    Ax=
    \begin{bmatrix}
      a_{11}x_1+\cdots+a_{1n}x_n\\
      \vdots\\
      a_{m1}x_1+\cdots+a_{mn}x_n
    \end{bmatrix}\in\RR^{m\times1}.
  \]
  Näin yhtälöryhmä \eqref{eq:linear-system} voidaan kirjoittaa lyhyesti
  \[
    \boxed{Ax=b,\qquad b=
      \begin{bmatrix}b_1\\\vdots\\b_m
    \end{bmatrix}.}
  \]
  Vektorien yhtäsuuruus tarkoittaa koordinaattien yhtäsuuruutta, joten
  tämä matriisiyhtälö ja alkuperäinen yhtälöryhmä ovat täsmälleen sama
  ehto. Sen \emph{laajennettu matriisi} on $[A\mid b]$.

  \newpage
  \subsubsection*{Tulon sarakkeet ja rivit}

  \begin{lemma}[Matriisitulo sarakkeittain ja riveittäin]
    Olkoot $A\in\RR^{m\times n}$ ja $B\in\RR^{n\times k}$.
    Kirjoitetaan $B=[b_1\ \cdots\ b_k]$, missä $b_\ell$ ovat sen
    sarakkeet. Tällöin
    \[
      \boxed{AB=[Ab_1\ \cdots\ Ab_k].}
    \]
    Jos $a_i=A_{i\cdot}$ ovat $A$:n rivit, niin vastaavasti
    \[
      \boxed{AB=
        \begin{bmatrix}a_1B\\\vdots\\a_mB
      \end{bmatrix}.}
    \]
  \end{lemma}
  \begin{proof}
    Sarakkeen $\ell$ rivillä $i$ molemmat ensimmäisen yhtälön puolet
    saavat arvon
    \[
      (Ab_\ell)_i=\sum_{j=1}^n a_{ij}(b_\ell)_j
      =\sum_{j=1}^n a_{ij}b_{j\ell}=(AB)_{i\ell}.
    \]
    Toisessa yhtälössä rivin $i$ sarakkeessa $\ell$ on
    \[
      (a_iB)_{1\ell}=\sum_{j=1}^n(a_i)_{1j}b_{j\ell}
      =\sum_{j=1}^n a_{ij}b_{j\ell}=(AB)_{i\ell}.
    \]
    Myös koot täsmäävät: $Ab_\ell\in\RR^{m\times1}$ ja
    $a_iB\in\RR^{1\times k}$.
  \end{proof}

  \begin{lemma}[Matriisin ja vektorin tulo sarakkeiden summana]
    Kun $A\in\RR^{m\times n}$ ja $x\in\RR^n$, pätee
    \[
      \boxed{Ax=x_1A_{\cdot1}+\cdots+x_nA_{\cdot n}.}
    \]
    Siis $Ax$ on $A$:n sarakkeiden lineaarikombinaatio, jonka
    kertoimet ovat vektorin $x$ koordinaatit.
  \end{lemma}
  \begin{proof}
    Oikean puolen koordinaatti $i$ on
    $\sum_{j=1}^n x_ja_{ij}=\sum_{j=1}^n a_{ij}x_j=(Ax)_i$.
    Tämä pätee jokaiselle $i=1,\ldots,m$.
  \end{proof}

  Näin $Ax=b$ kysyy myös, millä kertoimilla $x_1,\ldots,x_n$
  matriisin $A$ sarakkeista saadaan vektori $b$.
  Esimerkin matriisille $A$ tämä tarkoittaa kaavaa
  \[
    Ax=x_1
    \begin{bmatrix}1\\-2
    \end{bmatrix}
    +x_2
    \begin{bmatrix}0\\1
    \end{bmatrix}
    +x_3
    \begin{bmatrix}-1\\2
    \end{bmatrix}
    =
    \begin{bmatrix}x_1-x_3\\-2x_1+x_2+2x_3
    \end{bmatrix}.
  \]

  \subsubsection*{Koordinaattivektorit}

  \begin{definition}
    Avaruuden $\RR^n$ $j$:s \emph{koordinaattivektori} $e_j$ on
    sarakevektori, jonka $j$:s koordinaatti on $1$ ja muut ovat $0$:
    \[
      (e_j)_i=
      \begin{cases}1,&i=j,\\0,&i\ne j.
      \end{cases}
    \]
    Esimerkiksi avaruudessa $\RR^3$
    \[
      e_1=
      \begin{bmatrix}1\\0\\0
      \end{bmatrix},\qquad
      e_2=
      \begin{bmatrix}0\\1\\0
      \end{bmatrix},\qquad
      e_3=
      \begin{bmatrix}0\\0\\1
      \end{bmatrix}.
    \]
  \end{definition}

  \subsubsection*{Matriisi kuvauksena}

  Matriisi $A\in\RR^{m\times n}$ määrittää kuvauksen
  \[
    f_A\colon\RR^n\to\RR^m,\qquad x\mapsto Ax.
  \]
  Tässä $A$:n $n$ saraketta vastaavat lähtövektorin $n$:ää
  koordinaattia ja $m$ riviä tulosvektorin $m$:ää koordinaattia.
  Kuvaus $f_A$ on eri funktio kuin matriisin määritelmässä esiintyvä
  indeksiparin kuvaus $(i,j)\mapsto a_{ij}$.

  \begin{lemma}[Matriisikuvauksen lineaarisuus ja sarakkeet]
    Olkoot $A\in\RR^{m\times n}$, $x,y\in\RR^n$ ja
    $\lambda,\mu\in\RR$. Tällöin
    \[
      f_A(\lambda x+\mu y)=\lambda f_A(x)+\mu f_A(y).
    \]
    Tätä ominaisuutta kutsutaan \emph{lineaarisuudeksi}.
    Lisäksi $f_A(e_j)=Ae_j=A_{\cdot j}$ jokaisella $j=1,\ldots,n$.
  \end{lemma}
  \begin{proof}
    Jokaiselle tuloskoordinaatille $i$ pätee
    \begin{align*}
      (A(\lambda x+\mu y))_i
      &=\sum_{j=1}^n a_{ij}(\lambda x_j+\mu y_j)\\
      &=\lambda\sum_{j=1}^n a_{ij}x_j
      +\mu\sum_{j=1}^n a_{ij}y_j
      =(\lambda Ax+\mu Ay)_i.
    \end{align*}
    Sarakkeiden lineaarikombinaation kaavassa $Ae_j$:n kaikki
    kertoimet ovat nollia lukuun ottamatta sarakkeen $j$ kerrointa $1$.
    Siis $Ae_j=A_{\cdot j}$.
  \end{proof}

  Jokainen vektori voidaan kirjoittaa muodossa
  $x=x_1e_1+\cdots+x_ne_n$. Siksi lineaarinen kuvaus määräytyy
  koordinaattivektorien kuvista: jos $f\colon\RR^n\to\RR^m$ on
  lineaarinen ja $A=[f(e_1)\ \cdots\ f(e_n)]$, niin
  \[
    f(x)=\sum_{j=1}^n x_jf(e_j)=Ax.
  \]
  Tämä perustelee, miksi matriiseilla voidaan esittää kaikki
  näiden avaruuksien väliset lineaariset kuvaukset.

  \begin{example}[Sarakkeet kertovat koordinaattivektorien kuvat]
    Edellä käytetylle matriisille
    $A=\left[
      \begin{smallmatrix}1&0&-1\\-2&1&2
    \end{smallmatrix}\right]$
    kuvaus on $f_A\colon\RR^3\to\RR^2$, ja
    \[
      f_A(e_1)=
      \begin{bmatrix}1\\-2
      \end{bmatrix},\qquad
      f_A(e_2)=
      \begin{bmatrix}0\\1
      \end{bmatrix},\qquad
      f_A(e_3)=
      \begin{bmatrix}-1\\2
      \end{bmatrix}.
    \]
    Esimerkiksi $f_A(1,2,3)=(-2,6)$, kun tulokset esitetään
    koordinaattipareina.
  \end{example}
