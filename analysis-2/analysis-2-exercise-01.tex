\begin{exercise}
  Olkoon \(f\colon \RR \to \RR\),
  \[
    f(x) =
    \begin{cases}
      1, & \text{kun } x < 2,\\
      2, & \text{kun } x \geqslant 2.
    \end{cases}
  \]
  Osoita jatkuvuuden \((\varepsilon,\delta)\)-määritelmän
  perusteella, että funktio \(f\)
  \begin{enumerate}[label=(\alph*), nosep]
    \item on jatkuva pisteessä \(x = -2\),
    \item ei ole jatkuva pisteessä \(x = 2\).
  \end{enumerate}
\end{exercise}

\begin{solution}
  \leavevmode\par
  \begin{enumerate}[label=(\alph*), nosep]
    \item Olkoon \(\varepsilon > 0\). Valitaan \(\delta = 1\). Kun
      \(\abs{x - (-2)} < \delta = 1\), niin \(x \in (-3, -1)\). Täten
      \[
        \abs{f(x) - f(-2)} = \abs{1 - 1} = 0 < \varepsilon,
      \]
      joten \(f\) on jatkuva pisteessä \(-2\).
    \item Oletetaan, että \(f\) on jatkuva pisteessä \(2\).
      Valitaan \(\varepsilon_0 = \frac{1}{2}\). Tällöin on olemassa
      sellainen \(\delta > 0\), että kaikilla \(x \in \RR\)
      \[
        \abs{f(x) - f(2)} < \varepsilon_0 = \frac{1}{2},
      \]
      kun \(\abs{x - 2} < \delta\). Olkoon \(t = 2 - \frac{\delta}{2}\). Tällöin
      \(\abs{t - 2} = \frac{\delta}{2} < \delta\) ja \(t < 2\), joten
      \[
        \abs{f(t) - f(2)} = \abs{1 - 2} = 1 > \frac{1}{2} = \varepsilon_0.
      \]
      Tämä on ristiriita, joten \(f\) ei ole jatkuva pisteessä \(2\).
  \end{enumerate}
\end{solution}

\begin{exercise}
  Olkoon \(f\colon \RR \to \RR\),
  \[
    f(x) =
    \begin{cases}
      x, & \text{kun } x < 0,\\
      x^2, & \text{kun } x \geqslant 0.
    \end{cases}
  \]
  Osoita määritelmän perusteella, että funktio \(f\) on jatkuva origossa.
\end{exercise}

\begin{solution}
  Olkoon \(\varepsilon > 0\). Valitaan \(\delta = \minset{1, \varepsilon}\).
  Kun \(\abs{x - 0} < \delta \leq 1\), niin \(\abs{x^2} \leq \abs{x}\).
  Täten
  \[
    \abs{f(x) - f(0)} = \abs{f(x)} \leq \abs{x} < \delta \leq \varepsilon,
  \]
  kun \(\abs{x - 0} = \abs{x} < \delta\). Täten \(f\) on jatkuva origossa.
\end{solution}

\begin{exercise}
  Todista oppikirjan Lause 4.1.7.

  \medskip
  \noindent\textsc{Lause 4.1.7:} Olkoon \((x_n)\) suppeneva lukujono, jonka
  raja-arvo on \(x \in (a,b)\), ja olkoon \(f\colon (a,b) \to \RR\)
  funktio, joka on jatkuva pisteessä \(x\). Jos
  \(\{x_n : n \in \NNone\} \subset (a,b)\), niin lukujono \((f(x_n))\)
  suppenee kohti lukua \(f(x)\).
\end{exercise}

\begin{solution}
  Olkoon \(\varepsilon > 0\). Koska \(f\) on jatkuva pisteessä \(x \in (a, b)\),
  on olemassa \(\delta > 0\) siten, että
  \[
    \abs{f(t) - f(x)} < \varepsilon
  \]
  kaikilla \(t \in (a, b)\), joilla \(\abs{t - x} < \delta\).
  Koska \(x_n \to x\), on olemassa \(N \in \NNone\) siten, että
  \[
    \abs{x_n - x} < \delta,
  \]
  kun \(n > N\). Koska lisäksi \(x_n \in (a, b)\) kaikilla \(n \in
  \NNone\), saadaan
  \[
    \abs{f(x_n) - f(x)} < \varepsilon,
  \]
  kun \(n > N\). Siis \(\lim_{n \to \infty} f(x_n) = f(x)\).
\end{solution}

\begin{exercise}
  Olkoon \(f\colon \RR \to \RR\) jatkuva.
  Määritellään funktio \(g\colon \RR \to \RR\) asettamalla, että
  \(g(x) = \abs{f(x)}\). Osoita, että \(g\) on jatkuva.
\end{exercise}

\begin{solution}
  Olkoon \(t \in \RR\) mielivaltainen ja \(\varepsilon > 0\).
  Koska \(f\) on jatkuva pisteessä \(t\), on olemassa \(\delta > 0\) siten, että
  \[
    \abs{f(x) - f(t)} < \varepsilon
  \]
  kaikilla \(x \in \RR\), joilla \(\abs{x - t} < \delta\).
  Tällöin
  \[
    \begin{aligned}
      \abs{g(x) - g(t)} &= \abs*{\abs{f(x)} - \abs{f(t)}} \\
      &\leq \abs{f(x) - f(t)} < \varepsilon
    \end{aligned}
  \]
  kaikilla \(x \in \RR\), joilla \(\abs{x - t} < \delta\).
  Siis \(g\) on jatkuva pisteessä \(t\). Koska \(t \in \RR\) oli mielivaltainen,
  \(g\) on jatkuva.
\end{solution}

\begin{exercise}
  Olkoon \(f\colon \RR \setminus \set{0} \to \RR\),
  \(f(x) = \frac{1}{x}\). Osoita määritelmän perusteella, että funktio
  \(f\) on jatkuva.
\end{exercise}

\begin{solution}
  Olkoon \(A = \RR \setminus \set{0}\) ja \(g \colon A \to \RR\), \(g(x) = x\).
  Funktio \(g\) on jatkuva, sillä jokaisella \(t \in A\) ja
  \(\varepsilon > 0\) valinta \(\delta = \varepsilon\) antaa
  \[
    \abs{g(x) - g(t)} = \abs{x - t} < \varepsilon
  \]
  kaikilla \(x \in A\), joilla \(\abs{x - t} < \delta\).

  Olkoon nyt \(t \in A\) mielivaltainen ja \(\varepsilon > 0\).
  Koska \(g(t) = t \neq 0\) ja \(g\) on jatkuva pisteessä \(t\),
  on olemassa \(\delta_1 > 0\) siten, että
  \[
    \abs{g(x) - g(t)} < \frac{\abs{g(t)}}{2}
  \]
  kaikilla \(x \in A\), joilla \(\abs{x - t} < \delta_1\). Tällöin
  \[
    \abs{g(x)} \geq \abs{g(t)} - \abs{g(x) - g(t)}
    > \frac{\abs{g(t)}}{2}.
  \]
  Jatkuvuuden perusteella on myös olemassa \(\delta_2 > 0\) siten, että
  \[
    \abs{g(x) - g(t)} < \frac{\varepsilon\abs{g(t)}^2}{2}
  \]
  kaikilla \(x \in A\), joilla \(\abs{x - t} < \delta_2\). Valitaan
  \(\delta = \minset{\delta_1, \delta_2} > 0\). Nyt kaikilla \(x \in A\),
  joilla \(\abs{x - t} < \delta\), pätee
  \[
    \begin{aligned}
      \abs{f(x) - f(t)} &= \abs*{\frac{1}{g(x)} - \frac{1}{g(t)}} \\
      &= \abs{g(x) - g(t)}\frac{1}{\abs{g(x)}\abs{g(t)}} \\
      &\leq \abs{g(x) - g(t)}\frac{2}{\abs{g(t)}^2} \\
      &< \frac{\varepsilon\abs{g(t)}^2}{2}\frac{2}{\abs{g(t)}^2}
      = \varepsilon.
    \end{aligned}
  \]
  Täten \(f\) on jatkuva pisteessä \(t\). Koska \(t \in A\) oli
  mielivaltainen, \(f\) on jatkuva.
\end{solution}

\begin{exercise}
  \leavevmode\par
  \begin{enumerate}[label=(\alph*), nosep]
    \item Olkoon \(f\colon \RR \to \RR\) funktio, jolle
      \[
        f(x+y) = f(x) + f(y)
      \]
      kaikilla \(x,y \in \RR\). Osoita: jos funktio \(f\) on jatkuva
      origossa, niin \(f\) on jatkuva kaikkialla.
    \item Anna esimerkki funktiosta \(f\colon \RR \to \RR\), joka toteuttaa
      (a)-kohdan ehdot mutta ei saa jokaisessa pisteessä arvoa \(0\).
  \end{enumerate}
\end{exercise}

\begin{solution}
  \leavevmode\par
  \begin{enumerate}[label=(\alph*), nosep]
    \item Koska \(f(x + y) = f(x) + f(y)\) kaikilla \(x,y \in \RR\), niin
      \[
        f(x) = f((x - y) + y) = f(x - y) + f(y),
      \]
      eli \(f(x - y) = f(x) - f(y)\). Lisäksi
      \[
        f(0) = f(0 + 0) = 2f(0),
      \]
      eli \(f(0) = 0\).

      Olkoon \(y \in \RR\) mielivaltainen ja \(\varepsilon > 0\). Koska
      \(f\) on jatkuva origossa, on olemassa sellainen \(\delta > 0\), että
      \[
        \abs{f(t) - f(0)} < \varepsilon
      \]
      kaikilla \(t \in \RR\), joilla \(\abs{t} < \delta\).
      Soveltamalla tätä lukuun \(t = x - y\) saadaan
      \[
        \begin{aligned}
          \abs{f(x) - f(y)} &= \abs{f(x - y)} \\
          &= \abs{f(x - y) - f(0)} \\
          &< \varepsilon
        \end{aligned}
      \]
      kaikilla \(x \in \RR\), joilla \(\abs{x - y} < \delta\).
      Siis \(f\) on jatkuva pisteessä \(y\). Koska \(y \in \RR\) oli
      mielivaltainen, \(f\) on jatkuva kaikkialla.
    \item Olkoon \(g \colon \RR \to \RR\), \(g(x) = x\). Tällöin
      \[
        g(x + y) = x + y = g(x) + g(y)
      \]
      kaikilla \(x,y \in \RR\). Olkoon \(\varepsilon > 0\).
      Valitaan \(\delta = \varepsilon\). Tällöin
      \[
        \abs{g(x) - g(0)} = \abs{x} < \varepsilon
      \]
      kaikilla \(x \in \RR\), joilla \(\abs{x} < \delta\). Siis \(g\) on
      jatkuva origossa ja täyttää kohdan (a) ehdot. Lisäksi
      \(g(1) = 1 \neq 0\), joten \(g\) ei ole kaikkialla \(0\).
  \end{enumerate}
\end{solution}
