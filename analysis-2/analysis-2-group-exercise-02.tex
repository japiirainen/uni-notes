\begin{exercise}[Jensenin epäyhtälö]
  Olkoot \(N,m\in\NNone\), \(S\subset\RR^N\) konveksi joukko ja
  \(f\colon S\to\RR\) konveksi funktio. Olkoot
  \(x_1,\ldots,x_m\in S\) ja \(\lambda_1,\ldots,\lambda_m\geq 0\)
  siten, että \(\sum_{i=1}^m\lambda_i=1\).
  Käy huolellisesti läpi epäyhtälön
  \[
    f\!\left(\sum_{i=1}^m\lambda_i x_i\right)
    \leq \sum_{i=1}^m\lambda_i f(x_i)
  \]
  induktiotodistus. Kirjoita muistiinpanot ja valmistaudu perustelemaan
  todistuksen vaiheet suullisesti.
\end{exercise}

\begin{exercise}[Konveksisuus ja toinen derivaatta]
  Olkoot \(a,b\in\RR\), \(a<b\), ja \(f\colon[a,b]\to\RR\) jatkuva
  välillä \([a,b]\) ja kahdesti derivoituva välillä \((a,b)\).
  Oletetaan, että \(f''(x)\geq 0\) kaikilla \(x\in(a,b)\).
  Osoita, että \(f\) on konveksi välillä \([a,b]\).
  \begin{enumerate}[label=(\alph*)]
    \item Olkoot \(a\leq x_1<c<x_2\leq b\). Mitä väliarvolause kertoo
      osamääristä
      \[
        \frac{f(c)-f(x_1)}{c-x_1}
        \qquad\text{ja}\qquad
        \frac{f(x_2)-f(c)}{x_2-c}?
      \]
    \item Johda näiden avulla haluttu konveksisuusväite.
  \end{enumerate}
\end{exercise}

\begin{solution}
  \leavevmode\par
  \begin{enumerate}[label=(\alph*)]
    \item Väliarvolauseen nojalla on olemassa pisteet
      \(u\in(x_1,c)\) ja \(v\in(c,x_2)\), joille
      \[
        \frac{f(c)-f(x_1)}{c-x_1}=f'(u)
        \qquad\text{ja}\qquad
        \frac{f(x_2)-f(c)}{x_2-c}=f'(v).
      \]
      Koska \(f\) on kahdesti derivoituva välillä \((a,b)\),
      \(f'\) on jatkuva välillä \([u,v]\) ja derivoituva välillä
      \((u,v)\). Soveltamalla väliarvolausetta funktioon \(f'\)
      saadaan piste \(w\in(u,v)\), jolle
      \[
        f'(v)-f'(u)=f''(w)(v-u)\geq 0.
      \]
      Siis
      \[
        \frac{f(c)-f(x_1)}{c-x_1}
        \leq \frac{f(x_2)-f(c)}{x_2-c}.
      \]
    \item Olkoot \(x_1,x_2\in[a,b]\) ja \(t\in[0,1]\).
      Jos \(x_1=x_2=x\), niin
      \[
        f\bigl((1-t)x+tx\bigr)=f(x)=(1-t)f(x)+tf(x),
      \]
      joten konveksisuusehto toteutuu yhtäsuuruutena ilman jakamista
      luvulla \(x_2-x_1\). Myös tapauksissa \(t \in \set{0, 1}\)
      ehto toteutuu yhtäsuuruutena.

      Voidaan siis olettaa, että \(x_1\neq x_2\) ja \(0<t<1\).
      Vaihtamalla tarvittaessa pisteiden nimet ja korvaamalla
      \(t\) luvulla \(1-t\) voidaan lisäksi olettaa \(x_1<x_2\).
      Valitaan \(c=(1-t)x_1+tx_2\in(x_1,x_2)\).
      Koska \(c-x_1>0\) ja \(x_2-c>0\), kohdan (a)
      epäyhtälöstä seuraa
      \[
        (x_2-c)\bigl(f(c)-f(x_1)\bigr)
        \leq (c-x_1)\bigl(f(x_2)-f(c)\bigr).
      \]
      Kokoamalla termin \(f(c)\) sisältävät lausekkeet vasemmalle
      ja jakamalla luvulla \(x_2-x_1>0\) saadaan
      \[
        f(c)\leq
        \frac{x_2-c}{x_2-x_1}f(x_1)
        +\frac{c-x_1}{x_2-x_1}f(x_2).
      \]
      Valitun pisteen \(c\) määritelmän nojalla
      \[
        \begin{aligned}
          \frac{x_2-c}{x_2-x_1}
          &=\frac{x_2-\bigl((1-t)x_1+tx_2\bigr)}{x_2-x_1}\\
          &=\frac{(1-t)x_2-(1-t)x_1}{x_2-x_1}
          =\frac{(1-t)(x_2-x_1)}{x_2-x_1}=1-t
        \end{aligned}
      \]
      ja
      \[
        \begin{aligned}
          \frac{c-x_1}{x_2-x_1}
          &=\frac{(1-t)x_1+tx_2-x_1}{x_2-x_1}\\
          &=\frac{-tx_1+tx_2}{x_2-x_1}
          =\frac{t(x_2-x_1)}{x_2-x_1}=t.
        \end{aligned}
      \]
      Siten edellinen epäyhtälö antaa
      \[
        f\bigl((1-t)x_1+tx_2\bigr)
        \leq (1-t)f(x_1)+tf(x_2).
      \]
      Tämä on konveksisuuden määritelmä, joten \(f\) on konveksi
      välillä \([a,b]\).
  \end{enumerate}
\end{solution}

\begin{exercise}[Potenssisarjan konveksisuus]
  Olkoon \(a_n\geq 0\) kaikilla \(n\geq 0\), ja oletetaan, että
  potenssisarja
  \[
    f(x)=\sum_{n=0}^{\infty}a_n x^n
  \]
  suppenee jokaisessa pisteessä \(x\in[0,1]\).
  Osoita, että \(f\colon[0,1]\to\RR\) on konveksi.
\end{exercise}

\begin{solution}
  Koska potenssisarja suppenee pisteessä \(x=1\), sen suppenemissäde
  \(R\) toteuttaa ehdon \(R\geq 1\). Jokainen piste \(x\in(0,1)\)
  on siis suppenemisvälin sisäpiste, joten potenssisarjaa voidaan
  derivoida tällä välillä kahdesti termi termiltä. Saadaan
  \[
    \begin{aligned}
      f'(x)&=\sum_{n=1}^{\infty}na_nx^{n-1},\\
      f''(x)&=\sum_{n=2}^{\infty}n(n-1)a_nx^{n-2}\geq 0,
      \qquad x\in(0,1),
    \end{aligned}
  \]
  sillä \(a_n\geq 0\) ja \(x>0\), joten jälkimmäisen sarjan kaikki
  termit ovat epänegatiivisia.

  Tarkistetaan vielä jatkuvuus koko välillä \([0,1]\).
  Pisteessä \(x=1\) suppenemisen perusteella
  \[
    \sum_{n=0}^{\infty}a_n=f(1)<\infty.
  \]
  Olkoot \(x_0\in[0,1]\) ja \(\varepsilon>0\) mielivaltaisia.
  Koska sarja \(\sum_{n=0}^{\infty}a_n\) suppenee, voidaan valita
  \(N\geq 0\) siten, että
  \[
    \sum_{n=N+1}^{\infty}a_n<\frac{\varepsilon}{3}.
  \]
  Merkitään \(S_N(x)=\sum_{n=0}^{N}a_nx^n\). Koska
  \(0\leq x^n\leq 1\) kaikilla \(x\in[0,1]\), saadaan
  \[
    |f(x)-S_N(x)|
    =\sum_{n=N+1}^{\infty}a_nx^n
    \leq\sum_{n=N+1}^{\infty}a_n
    <\frac{\varepsilon}{3}
    \qquad\text{kaikilla }x\in[0,1].
  \]
  Sama osasumma \(S_N\) siis approksimoi funktiota \(f\) näin tarkasti
  koko välillä \([0,1]\). Tämä on tasaisen suppenemisen ydin.

  Koska \(S_N\) on polynomina jatkuva pisteessä \(x_0\), on olemassa
  \(\delta>0\) siten, että
  \[
    |S_N(x)-S_N(x_0)|<\frac{\varepsilon}{3},
    \qquad\text{kun }|x-x_0|<\delta.
  \]
  Kun \(x\in[0,1]\) ja \(|x-x_0|<\delta\), kolmioepäyhtälö antaa
  \[
    \begin{aligned}
      |f(x)-f(x_0)|
      &\leq |f(x)-S_N(x)|+|S_N(x)-S_N(x_0)|\\
      &\quad+|S_N(x_0)-f(x_0)|\\
      &<\frac{\varepsilon}{3}+\frac{\varepsilon}{3}
      +\frac{\varepsilon}{3}=\varepsilon.
    \end{aligned}
  \]
  Siis \(f\) on jatkuva pisteessä \(x_0\). Koska \(x_0\in[0,1]\)
  oli mielivaltainen, \(f\) on jatkuva koko välillä \([0,1]\),
  myös päätepisteissä.

  Siis \(f\) on jatkuva välillä \([0,1]\), kahdesti derivoituva
  välillä \((0,1)\) ja \(f''(x)\geq 0\) kaikilla \(x\in(0,1)\).
  Tehtävän 3 nojalla \(f\) on siten konveksi välillä \([0,1]\).
\end{solution}

\begin{exercise}[Aritmeettis-geometrinen epäyhtälö]
  Olkoot \(n\in\NNone\) ja \(x_1,\ldots,x_n\geq 0\).
  Todista, että
  \[
    (x_1\cdots x_n)^{1/n}\leq\frac{x_1+\cdots+x_n}{n}.
  \]
\end{exercise}

\begin{solution}
  Olkoot \(n\in\NNone\) ja \(x_1,\ldots,x_n\geq 0\).
  Jos jokin luvuista \(x_i\) on nolla, niin
  \[
    (x_1\cdots x_n)^{1/n}=0\leq\frac{x_1+\cdots+x_n}{n},
  \]
  joten väite pätee. Voidaan siis olettaa, että \(x_i>0\) kaikilla
  \(i=1,\ldots,n\). Tällöin kaikki seuraavassa esiintyvät logaritmit
  ovat määriteltyjä.

  Olkoon \(f\colon(0,\infty)\to\RR\), \(f(x)=-\ln x\).
  Derivoimalla saadaan
  \[
    f'(x)=-\frac{1}{x},
    \qquad f''(x)=\frac{1}{x^2}>0
    \quad\text{kaikilla }x\in(0,\infty).
  \]
  Tehtävän 3 nojalla \(f\) on konveksi jokaisella suljetulla välillä
  \([a,b]\subset(0,\infty)\), missä \(a<b\). Koska mitkä tahansa kaksi
  positiivista lukua sisältyvät tällaiseen väliin, \(f\) on konveksi
  koko välillä \((0,\infty)\).

  Logaritmin laskusääntöjä soveltamalla saadaan
  \[
    f\left((x_1\cdots x_n)^{1/n}\right)
    =-\frac{1}{n}\left(\ln x_1+\cdots+\ln x_n\right).
  \]
  Valitaan painot \(\lambda_i=1/n\) kaikilla \(i=1,\ldots,n\).
  Ne ovat epänegatiivisia ja niiden summa on \(1\), joten Jensenin
  epäyhtälön (tehtävä 2) nojalla
  \[
    -\frac{1}{n}\left(\ln(x_1) + \cdots + \ln(x_n)\right) \geq
    -\ln\left(\frac{1}{n}(x_1 + \cdots + x_n)\right).
  \]
  Kertomalla puolittain \(-1\):llä saadaan
  \[
    \ln\left((x_1 \cdots x_n)^{1/n}\right) \leq
    \ln\left(\frac{1}{n}(x_1 + \cdots + x_n)\right).
  \]
  Koska eksponenttifunktio on aidosti kasvava, sen soveltaminen
  epäyhtälön molemmille puolille säilyttää epäyhtälön suunnan. Näin saadaan
  \[
    (x_1 \cdots x_n)^{1/n} \leq
    \frac{1}{n}(x_1 + \cdots + x_n).
  \]
\end{solution}
